   385→function ExpertiseIcon({ id, className }: { id: string; className?: string }) {
   386→  const opt = EXPERTISE_OPTIONS.find((o) => o.id === id)
   387→  if (!opt) return <Sparkles className={className} />
   388→  const Icon = opt.icon
   389→  return <Icon className={className} />
   390→}

   392→function ExpertiseLabel({ id }: { id: string }) {

   397→function InstructorCard({ instructor: inst, onViewProfile, onDelete, index }: { instructor: Instructor; onViewProfile: () => void; onDelete: () => void; index: number }) {
   398→  const initials = inst.name
   399→    .split(" ")
   400→    .map((p) => p[0])
   401→    .slice(0, 2)
   402→    .join("")
   403→    .toUpperCase()
   404→  const workloadPct = inst.profile ? Math.min(100, Math.round((inst.currentBatches / inst.profile.maxBatches) * 100)) : 0
   405→  const workloadColor = workloadPct >= 100 ? "bg-rose-500" : workloadPct >= 66 ? "bg-amber-500" : "bg-emerald-500"
   406→  const workloadText = workloadPct >= 100 ? "text-rose-300 border-rose-500/30" : "text-emerald-300 border-emerald-500/30"
   407→
   408→  return (
   409→    <motion.div
   410→      initial={{ opacity: 0, y: 10 }}
   411→      animate={{ opacity: 1, y: 0 }}
   412→      exit={{ opacity: 0, y: -10 }}
   413→      transition={{ duration: 0.3, delay: index * 0.05 }}
   414→      className="card-premium rounded-xl p-5"
   415→    >
   416→      {/* Header */}
   417→      <div className="flex items-start gap-3 mb-3">
   418→        {/* Avatar */}
   419→        {inst.avatar ? (
   420→          <img src={inst.avatar} alt={inst.name} className="size-12 rounded-full object-cover border border-border/60" />
   421→        ) : (
   422→          <div className="size-12 rounded-full bg-gradient-to-br from-violet-500/30 to-cyan-500/30 border border-border/60 flex items-center justify-center text-sm font-bold text-violet-100 shrink-0">
   423→            {initials}
   424→          </div>
   425→        )}
   426→        <div className="min-w-0 flex-1">
   427→          <div className="flex items-center gap-1.5">
   428→            <h3 className="font-semibold text-sm truncate">{inst.name}</h3>
   429→            <BadgeCheck className="h-3.5 w-3.5 text-violet-400 shrink-0" />
   430→          </div>
   431→          <p className="text-xs text-muted-foreground truncate">{inst.title || "Security Instructor"}</p>
   432→          <p className="text-[10px] text-muted-foreground truncate">{inst.email}</p>
   433→        </div>
   434→      </div>
   435→
   436→      {/* Bio */}
   437→      {inst.bio && <p className="text-xs text-muted-foreground mb-3 line-clamp-2">{inst.bio}</p>}
   438→
   439→      {/* Expertise tags */}
   440→      {inst.profile?.expertise && inst.profile.expertise.length > 0 && (
   441→        <div className="flex flex-wrap gap-1 mb-3">
   442→          {inst.profile.expertise.map((e) => (
   443→            <Badge key={e} variant="outline" className="text-[9px] bg-violet-500/10 text-violet-300 border-violet-500/30">
   444→              <ExpertiseIcon id={e} className="h-2.5 w-2.5 mr-1" />
   445→              <ExpertiseLabel id={e} />
   446→            </Badge>
   447→          ))}
   448→        </div>
   449→      )}
   450→
   451→      {/* Years of experience */}
   452→      {inst.profile?.yearsExperience != null && (
   453→        <div className="flex items-center gap-1.5 text-xs text-muted-foreground mb-2">
   454→          <Briefcase className="h-3 w-3" /> {inst.profile.yearsExperience}+ years experience
   455→        </div>
   456→      )}
   457→
   458→      {/* Certifications */}
   459→      {inst.profile?.certifications && inst.profile.certifications.length > 0 && (
   460→        <div className="flex flex-wrap gap-1 mb-3">
   461→          {inst.profile.certifications.map((c) => (
   462→            <Badge key={c} className="text-[9px] bg-amber-500/10 text-amber-300 border border-amber-500/30">
   463→              <Award className="h-2.5 w-2.5 mr-1" /> {c}
   464→            </Badge>
   465→          ))}
   466→        </div>
   467→      )}
   468→
   469→      {/* Workload */}
   470→      <div className="mb-3">
   471→        <div className="flex items-center justify-between mb-1">
   472→          <span className="text-[10px] uppercase tracking-wider text-muted-foreground">Workload</span>
   473→          <Badge variant="outline" className={cn("text-[9px]", workloadText)}>
   474→            {inst.currentBatches}/{inst.profile?.maxBatches ?? 3} batches
   475→          </Badge>
   476→        </div>
   477→        <div className="w-full h-1.5 bg-muted rounded-full overflow-hidden">
   478→          <div className={cn("h-full rounded-full transition-all", workloadColor)} style={{ width: `${workloadPct}%` }} />
   479→        </div>
   480→      </div>
   481→
   482→      {/* Action buttons */}
