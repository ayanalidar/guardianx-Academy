   207→   207→// ============ Live Room (WebRTC) ============
   208→   208→function LiveRoom({ session, onLeave, userName, userId, isHost }: {
   209→   209→  session: LiveSessionItem; onLeave: () => void; userName: string; userId: string; isHost: boolean
   210→   210→}) {
   211→   211→  const qc = useQueryClient()
   212→   212→  const webrtcRef = React.useRef<WebRTCSession | null>(null)
   213→   213→  const [connected, setConnected] = React.useState(false)
   214→   214→  const [peers, setPeers] = React.useState<PeerMeta[]>([])
   215→   215→  const [micOn, setMicOn] = React.useState(false)
   216→   216→  const [sharing, setSharing] = React.useState(false)
   217→   217→  const [presentRequests, setPresentRequests] = React.useState<any[]>([])
   218→   218→  const [chat, setChat] = React.useState<{ userId: string; userName: string; message: string; timestamp: number }[]>([])
   219→   219→  const [chatInput, setChatInput] = React.useState("")
   220→   220→  const [showChat, setShowChat] = React.useState(true)
   221→   221→  const [activePresenterName, setActivePresenterName] = React.useState<string | null>(null)
   222→   222→  const [stageMode, setStageMode] = React.useState<"screen" | "whiteboard">("screen")
   223→   223→
   224→   224→  // Video Recording state (records the screen-share stream via MediaRecorder)
   225→   225→  const [recording, setRecording] = React.useState(false)
   226→   226→  const [recordSeconds, setRecordSeconds] = React.useState(0)
   227→   227→  const mediaRecorderRef = React.useRef<MediaRecorder | null>(null)
   228→   228→  const recordedChunksRef = React.useRef<Blob[]>([])
   229→   229→  const recordTimerRef = React.useRef<ReturnType<typeof setInterval> | null>(null)
   230→   230→
   231→   231→  // remote video elements
   232→   232→  const screenVideoRef = React.useRef<HTMLVideoElement>(null)
   233→   233→  const remoteVoicePeers = React.useRef<Map<string, HTMLAudioElement>>(new Map())
   234→   234→  const voiceContainerRef = React.useRef<HTMLDivElement>(null)
   235→   235→
   236→   236→  React.useEffect(() => {
   237→   237→    const session_ = new WebRTCSession(session.roomId, userId, userName, isHost ? "host" : "viewer")
   238→   238→    webrtcRef.current = session_
   239→   239→
   240→   240→    session_.onPeersChange = (members) => setPeers(members)
   241→   241→    session_.onRemoteStream = (socketId, stream, kind) => {
   242→   242→      if (kind === "screen") {
   243→   243→        // this peer is presenting their screen
   244→   244→        if (screenVideoRef.current) {
   245→   245→          screenVideoRef.current.srcObject = stream
   246→   246→          screenVideoRef.current.play().catch(() => {})
   247→   247→        }
   248→   248→      } else {
   249→   249→        // voice — attach to a hidden audio element
   250→   250→        let el = remoteVoicePeers.current.get(socketId)
   251→   251→        if (!el) {
   252→   252→          el = document.createElement("audio")
   253→   253→          el.autoplay = true
   254→   254→          el.setAttribute("playsinline", "")
   255→   255→          voiceContainerRef.current?.appendChild(el)
   256→   256→          remoteVoicePeers.current.set(socketId, el)
   257→   257→        }
   258→   258→        el.srcObject = stream
   259→   259→        el.play().catch(() => {})
   260→   260→      }
   261→   261→    }
   262→   262→    session_.onStreamRemoved = (socketId) => {
   263→   263→      if (screenVideoRef.current && screenVideoRef.current.srcObject) {
   264→   264→        const tracks = (screenVideoRef.current.srcObject as MediaStream).getTracks()
   265→   265→        // only clear if no peers are presenting
   266→   266→      }
   267→   267→      const el = remoteVoicePeers.current.get(socketId)
   268→   268→      if (el) { el.srcObject = null; el.remove(); remoteVoicePeers.current.delete(socketId) }
   269→   269→    }
   270→   270→    session_.onChat = (msg) => setChat((c) => [...c, msg])
   271→   271→    session_.onPresenterChange = (_socketId, name) => {
   272→   272→      setActivePresenterName(name)
   273→   273→      toast.info(`${name} is now presenting`)
   274→   274→    }
   275→   275→    session_.onPresentRequest = (req) => {
   276→   276→      setPresentRequests((r) => [...r, req])
   277→   277→      toast.info(`${req.userName} requested to present`)
   278→   278→    }
   279→   279→
   280→   280→    session_.connect().then(() => setConnected(true))
   281→   281→
   282→   282→    return () => {
   283→   283→      session_.disconnect()
   284→   284→      // cleanup audio els
   285→   285→      remoteVoicePeers.current.forEach((el) => { el.srcObject = null; el.remove() })
   286→   286→      remoteVoicePeers.current.clear()
   287→   287→    }
   288→   288→  }, [session.id])
   289→   289→
   290→   290→  async function toggleMic() {
   291→   291→    const next = !micOn
   292→   292→    await webrtcRef.current?.toggleMic(next)
   293→   293→    setMicOn(next)
   294→   294→    toast.success(next ? "Microphone on" : "Microphone muted")
   295→   295→  }
   296→   296→
   297→   297→  async function toggleShare() {
   298→   298→    if (sharing) {
   299→   299→      webrtcRef.current?.stopScreenShare()
   300→   300→      setSharing(false)
   301→   301→      if (screenVideoRef.current) screenVideoRef.current.srcObject = null
   302→   302→    } else {
   303→   303→      const ok = await webrtcRef.current?.startScreenShare()
   304→   304→      if (ok) {
   305→   305→        setSharing(true)
   306→   306→        // show own screen preview
   307→   307→        if (screenVideoRef.current && webrtcRef.current?.screenStream) {
   308→   308→          screenVideoRef.current.srcObject = webrtcRef.current.screenStream
   309→   309→          screenVideoRef.current.play().catch(() => {})
   310→   310→        }
   311→   311→        toast.success("Screen sharing started")
   312→   312→      } else {
   313→   313→        toast.error("Screen share failed or was denied")
   314→   314→      }
   315→   315→    }
   316→   316→  }
   317→   317→
   318→   318→  function requestPresent() {
   319→   319→    webrtcRef.current?.requestPresent()
   320→   320→    toast.success("Requested to present — waiting for host approval")
   321→   321→  }
   322→   322→
   323→   323→  function approveRequest(req: any) {
   324→   324→    webrtcRef.current?.grantPresent(req.socketId)
   325→   325→    setPresentRequests((r) => r.filter((x) => x.socketId !== req.socketId))
   326→   326→  }
   327→   327→
   328→   328→  function sendChat() {
   329→   329→    if (!chatInput.trim()) return
   330→   330→    webrtcRef.current?.sendChat(chatInput)
   331→   331→    setChat((c) => [...c, { userId, userName: "You", message: chatInput, timestamp: Date.now() }])
   332→   332→    setChatInput("")
   333→   333→  }
   334→   334→
   335→   335→  async function leaveCall() {
   336→   336→    // Stop recording if active
   337→   337→    if (recording) {
   338→   338→      stopRecording()
   339→   339→    }
   340→   340→    await api(`/api/live-sessions/${session.id}`, { method: "POST", body: JSON.stringify({ action: "leave" }) })
   341→   341→    if (isHost) {
   342→   342→      await api(`/api/live-sessions/${session.id}`, { method: "POST", body: JSON.stringify({ action: "end" }) })
   343→   343→    }
   344→   344→    qc.invalidateQueries({ queryKey: ["live-sessions"] })
   345→   345→    onLeave()
   346→   346→  }
   347→   347→
   348→   348→  // ===== Video Recording (MediaRecorder API) =====
   349→   349→  function startRecording() {
   350→   350→    const videoEl = screenVideoRef.current
   351→   351→    const stream = videoEl?.srcObject as MediaStream | null
   352→   352→    if (!stream || stream.getTracks().length === 0) {
   353→   353→      toast.error("Start screen sharing first to record the session.")
   354→   354→      return
   355→   355→    }
   356→   356→    try {
   357→   357→      recordedChunksRef.current = []
   358→   358→      // Prefer video/webm; fall back to whatever the browser supports
   359→   359→      const mimeType = MediaRecorder.isTypeSupported("video/webm;codecs=vp9")
   360→   360→        ? "video/webm;codecs=vp9"
   361→   361→        : MediaRecorder.isTypeSupported("video/webm")
   362→   362→          ? "video/webm"
   363→   363→          : ""
   364→   364→      const recorder = new MediaRecorder(stream, mimeType ? { mimeType } : undefined)
   365→   365→      recorder.ondataavailable = (e) => {
   366→   366→        if (e.data.size > 0) recordedChunksRef.current.push(e.data)
   367→   367→      }
   368→   368→      recorder.onstop = () => {
   369→   369→        const blob = new Blob(recordedChunksRef.current, { type: "video/webm" })
   370→   370→        const url = URL.createObjectURL(blob)
   371→   371→        const a = document.createElement("a")
   372→   372→        a.href = url
   373→   373→        const stamp = new Date().toISOString().replace(/[:.]/g, "-")
   374→   374→        a.download = `guardianx-${session.title.replace(/[^a-z0-9]/gi, "-").toLowerCase()}-${stamp}.webm`
   375→   375→        document.body.appendChild(a)
   376→   376→        a.click()
   377→   377→        document.body.removeChild(a)
   378→   378→        URL.revokeObjectURL(url)
   379→   379→        toast.success("Recording saved to your downloads.")
   380→   380→      }
   381→   381→      recorder.start(1000) // collect data every 1s
   382→   382→      mediaRecorderRef.current = recorder
   383→   383→      setRecording(true)
   384→   384→      setRecordSeconds(0)
   385→   385→      recordTimerRef.current = setInterval(() => {
   386→   386→        setRecordSeconds((s) => s + 1)
   387→   387→      }, 1000)
   388→   388→      toast.success("Recording started")
   389→   389→    } catch (err) {
   390→   390→      console.error(err)
   391→   391→      toast.error("Recording failed to start.")
   392→   392→    }
   393→   393→  }
   394→   394→
