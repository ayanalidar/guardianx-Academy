    47→function SessionList({ onJoin }: { onJoin: (s: LiveSessionItem) => void }) {

   133→            </div>

   136→          {/* Ended */}

   207→// ============ Live Room (WebRTC) ============
   208→function LiveRoom({ session, onLeave, userName, userId, isHost }: {
   209→  session: LiveSessionItem; onLeave: () => void; userName: string; userId: string; isHost: boolean
   210→}) {
   211→  const qc = useQueryClient()
   212→  const webrtcRef = React.useRef<WebRTCSession | null>(null)
   213→  const [connected, setConnected] = React.useState(false)
   214→  const [peers, setPeers] = React.useState<PeerMeta[]>([])
   215→  const [micOn, setMicOn] = React.useState(false)
   216→  const [sharing, setSharing] = React.useState(false)
   217→  const [presentRequests, setPresentRequests] = React.useState<any[]>([])
   218→  const [chat, setChat] = React.useState<{ userId: string; userName: string; message: string; timestamp: number }[]>([])
   219→  const [chatInput, setChatInput] = React.useState("")
   220→  const [showChat, setShowChat] = React.useState(true)
   221→  const [activePresenterName, setActivePresenterName] = React.useState<string | null>(null)
   222→  const [stageMode, setStageMode] = React.useState<"screen" | "whiteboard">("screen")
   223→
   224→  // Video Recording state (records the screen-share stream via MediaRecorder)
   225→  const [recording, setRecording] = React.useState(false)
   226→  const [recordSeconds, setRecordSeconds] = React.useState(0)
   227→  const mediaRecorderRef = React.useRef<MediaRecorder | null>(null)
   228→  const recordedChunksRef = React.useRef<Blob[]>([])
   229→  const recordTimerRef = React.useRef<ReturnType<typeof setInterval> | null>(null)
   230→
   231→  // remote video elements
   232→  const screenVideoRef = React.useRef<HTMLVideoElement>(null)
   233→  const remoteVoicePeers = React.useRef<Map<string, HTMLAudioElement>>(new Map())
   234→  const voiceContainerRef = React.useRef<HTMLDivElement>(null)
   235→
   236→  React.useEffect(() => {
   237→    const session_ = new WebRTCSession(session.roomId, userId, userName, isHost ? "host" : "viewer")
   238→    webrtcRef.current = session_
   239→
   240→    session_.onPeersChange = (members) => setPeers(members)
   241→    session_.onRemoteStream = (socketId, stream, kind) => {
   242→      if (kind === "screen") {
   243→        // this peer is presenting their screen
   244→        if (screenVideoRef.current) {
   245→          screenVideoRef.current.srcObject = stream
   246→          screenVideoRef.current.play().catch(() => {})
   247→        }
   248→      } else {
   249→        // voice — attach to a hidden audio element
   250→        let el = remoteVoicePeers.current.get(socketId)
   251→        if (!el) {
   252→          el = document.createElement("audio")
   253→          el.autoplay = true
   254→          el.setAttribute("playsinline", "")
   255→          voiceContainerRef.current?.appendChild(el)
   256→          remoteVoicePeers.current.set(socketId, el)
   257→        }
   258→        el.srcObject = stream
   259→        el.play().catch(() => {})
   260→      }
   261→    }
   262→    session_.onStreamRemoved = (socketId) => {
   263→      if (screenVideoRef.current && screenVideoRef.current.srcObject) {
   264→        const tracks = (screenVideoRef.current.srcObject as MediaStream).getTracks()
   265→        // only clear if no peers are presenting
   266→      }
   267→      const el = remoteVoicePeers.current.get(socketId)
   268→      if (el) { el.srcObject = null; el.remove(); remoteVoicePeers.current.delete(socketId) }
   269→    }
   270→    session_.onChat = (msg) => setChat((c) => [...c, msg])
   271→    session_.onPresenterChange = (_socketId, name) => {
   272→      setActivePresenterName(name)
   273→      toast.info(`${name} is now presenting`)
   274→    }
   275→    session_.onPresentRequest = (req) => {
   276→      setPresentRequests((r) => [...r, req])
   277→      toast.info(`${req.userName} requested to present`)
   278→    }
   279→
   280→    session_.connect().then(() => setConnected(true))
   281→
   282→    return () => {
   283→      session_.disconnect()
   284→      // cleanup audio els
   285→      remoteVoicePeers.current.forEach((el) => { el.srcObject = null; el.remove() })
   286→      remoteVoicePeers.current.clear()
   287→    }
   288→  }, [session.id])
   289→
   290→  async function toggleMic() {
   291→    const next = !micOn
   292→    await webrtcRef.current?.toggleMic(next)
   293→    setMicOn(next)
   294→    toast.success(next ? "Microphone on" : "Microphone muted")
   295→  }
   296→
   297→  async function toggleShare() {
   298→    if (sharing) {
   299→      webrtcRef.current?.stopScreenShare()
   300→      setSharing(false)
   301→      if (screenVideoRef.current) screenVideoRef.current.srcObject = null
   302→    } else {
   303→      const ok = await webrtcRef.current?.startScreenShare()
   304→      if (ok) {
   305→        setSharing(true)
   306→        // show own screen preview
   307→        if (screenVideoRef.current && webrtcRef.current?.screenStream) {
   308→          screenVideoRef.current.srcObject = webrtcRef.current.screenStream
   309→          screenVideoRef.current.play().catch(() => {})
   310→        }
   311→        toast.success("Screen sharing started")
   312→      } else {
   313→        toast.error("Screen share failed or was denied")
   314→      }
   315→    }
   316→  }
   317→
   318→  function requestPresent() {
   319→    webrtcRef.current?.requestPresent()
   320→    toast.success("Requested to present — waiting for host approval")
   321→  }
   322→
   323→  function approveRequest(req: any) {
   324→    webrtcRef.current?.grantPresent(req.socketId)
   325→    setPresentRequests((r) => r.filter((x) => x.socketId !== req.socketId))
   326→  }
   327→
   328→  function sendChat() {
   329→    if (!chatInput.trim()) return
   330→    webrtcRef.current?.sendChat(chatInput)
   331→    setChat((c) => [...c, { userId, userName: "You", message: chatInput, timestamp: Date.now() }])
   332→    setChatInput("")
   333→  }
   334→
   335→  async function leaveCall() {
   336→    // Stop recording if active
   337→    if (recording) {
   338→      stopRecording()
   339→    }
   340→    await api(`/api/live-sessions/${session.id}`, { method: "POST", body: JSON.stringify({ action: "leave" }) })
   341→    if (isHost) {
   342→      await api(`/api/live-sessions/${session.id}`, { method: "POST", body: JSON.stringify({ action: "end" }) })
   343→    }
   344→    qc.invalidateQueries({ queryKey: ["live-sessions"] })
   345→    onLeave()
   346→  }
   347→
   348→  // ===== Video Recording (MediaRecorder API) =====
   349→  function startRecording() {
   350→    const videoEl = screenVideoRef.current
   351→    const stream = videoEl?.srcObject as MediaStream | null
   352→    if (!stream || stream.getTracks().length === 0) {
   353→      toast.error("Start screen sharing first to record the session.")
   354→      return
   355→    }
   356→    try {
   357→      recordedChunksRef.current = []
   358→      // Prefer video/webm; fall back to whatever the browser supports
   359→      const mimeType = MediaRecorder.isTypeSupported("video/webm;codecs=vp9")
   360→        ? "video/webm;codecs=vp9"
   361→        : MediaRecorder.isTypeSupported("video/webm")
   362→          ? "video/webm"
   363→          : ""
   364→      const recorder = new MediaRecorder(stream, mimeType ? { mimeType } : undefined)
   365→      recorder.ondataavailable = (e) => {
   366→        if (e.data.size > 0) recordedChunksRef.current.push(e.data)
   367→      }
   368→      recorder.onstop = () => {
   369→        const blob = new Blob(recordedChunksRef.current, { type: "video/webm" })
   370→        const url = URL.createObjectURL(blob)
   371→        const a = document.createElement("a")
   372→        a.href = url
   373→        const stamp = new Date().toISOString().replace(/[:.]/g, "-")
   374→        a.download = `guardianx-${session.title.replace(/[^a-z0-9]/gi, "-").toLowerCase()}-${stamp}.webm`
   375→        document.body.appendChild(a)
   376→        a.click()
   377→        document.body.removeChild(a)
   378→        URL.revokeObjectURL(url)
   379→        toast.success("Recording saved to your downloads.")
   380→      }
   381→      recorder.start(1000) // collect data every 1s
   382→      mediaRecorderRef.current = recorder
   383→      setRecording(true)
   384→      setRecordSeconds(0)
   385→      recordTimerRef.current = setInterval(() => {
   386→        setRecordSeconds((s) => s + 1)
   387→      }, 1000)
   388→      toast.success("Recording started")
   389→    } catch (err) {
   390→      console.error(err)
   391→      toast.error("Recording failed to start.")
   392→    }
   393→  }
   394→
   395→  function stopRecording() {
   396→    const recorder = mediaRecorderRef.current
   397→    if (recorder && recorder.state !== "inactive") {
   398→      recorder.stop()
   399→    }
   400→    mediaRecorderRef.current = null
   401→    setRecording(false)
   402→    if (recordTimerRef.current) {
   403→      clearInterval(recordTimerRef.current)
   404→      recordTimerRef.current = null
   405→    }
   406→  }
   407→
   408→  function formatRecordTime(s: number) {
   409→    const m = Math.floor(s / 60)
   410→    const sec = s % 60
   411→    return `${m.toString().padStart(2, "0")}:${sec.toString().padStart(2, "0")}`
   412→  }
   413→
   414→  return (
   415→    <div className="space-y-4">
   416→      <div className="flex items-center justify-between gap-3 flex-wrap">
   417→        <div className="flex items-center gap-3">
   418→          <Button variant="ghost" size="sm" onClick={leaveCall} className="text-muted-foreground">
   419→            <ChevronLeft className="h-4 w-4 mr-1" /> Leave
   420→          </Button>
   421→          <div>
   422→            <h1 className="text-xl font-bold flex items-center gap-2">
   423→              <span className="h-2 w-2 rounded-full bg-red-500 pulse-dot" /> {session.title}
   424→            </h1>
   425→            <p className="text-xs text-muted-foreground">{peers.length + 1} participant{(peers.length + 1) !== 1 ? "s" : ""} · {connected ? "Connected" : "Connecting..."}</p>
   426→          </div>
   427→        </div>
   428→        {activePresenterName && (
   429→          <Badge className="bg-cyan-500/15 text-cyan-400 border border-cyan-500/30">
   430→            <Monitor className="h-3 w-3 mr-1" /> Presenting: {activePresenterName}
   431→          </Badge>
   432→        )}
   433→      </div>
   434→
   435→      <div className="grid lg:grid-cols-4 gap-4">
   436→        {/* Main stage */}
   437→        <div className="lg:col-span-3 space-y-4">
   438→          {/* Stage mode switcher (Screen Share / Whiteboard) */}
   439→          <div className="flex items-center gap-2">
   440→            <Button
   441→              variant={stageMode === "screen" ? "default" : "outline"}
   442→              size="sm"
   443→              onClick={() => setStageMode("screen")}
   444→              className={stageMode === "screen" ? "bg-emerald-500 text-emerald-950 hover:bg-emerald-400" : ""}
   445→            >
   446→              <Monitor className="h-3.5 w-3.5 mr-1.5" /> Screen Share
   447→            </Button>
   448→            <Button
   449→              variant={stageMode === "whiteboard" ? "default" : "outline"}
   450→              size="sm"
   451→              onClick={() => setStageMode("whiteboard")}
   452→              className={stageMode === "whiteboard" ? "bg-cyan-500 text-cyan-950 hover:bg-cyan-400" : ""}
   453→            >
   454→              <PenLine className="h-3.5 w-3.5 mr-1.5" /> Whiteboard
   455→            </Button>
   456→            <div className="ml-auto flex items-center gap-2 text-xs text-muted-foreground">
   457→              {stageMode === "whiteboard" && (
   458→                <Badge variant="outline" className="text-[10px] text-cyan-400 border-cyan-500/30 bg-cyan-500/10">
   459→                  <Presentation className="h-2.5 w-2.5 mr-1" /> Live collaborative whiteboard
   460→                </Badge>
   461→              )}
   462→            </div>
   463→          </div>
   464→
   465→          {stageMode === "whiteboard" ? (
   466→            <Whiteboard
   467→              roomId={session.roomId}
   468→              userId={userId}
   469→              userName={userName}
   470→              role={isHost ? "host" : "viewer"}
   471→              height={500}
   472→            />
   473→          ) : (
   474→            <Card className="overflow-hidden relative bg-black/40 border-border min-h-[400px] flex items-center justify-center">
   475→              <div className="absolute top-3 left-3 z-10 flex items-center gap-2 px-3 py-1.5 rounded-full bg-black/60 backdrop-blur text-xs">
   476→                {recording ? (
   477→                  <>
   478→                    <span className="h-2 w-2 rounded-full bg-red-500 pulse-dot" />
   479→                    <span className="font-mono text-red-400">REC {formatRecordTime(recordSeconds)}</span>
   480→                  </>
   481→                ) : (
   482→                  <>
   483→                    <span className="h-1.5 w-1.5 rounded-full bg-red-400/60" />
   484→                    <span className="font-mono text-red-400/70">REC</span>
   485→                  </>
   486→                )}
   487→                <span className="text-muted-foreground">·</span>
   488→                <Video className="h-3 w-3 text-emerald-400" />
   489→                <span className="text-emerald-400">{sharing ? "Presenting" : activePresenterName ? "Viewing" : "No presentation"}</span>
   490→              </div>
   491→              <video
   492→                ref={screenVideoRef}
   493→                autoPlay
   494→                playsInline
   495→                muted={sharing} // mute local preview to avoid echo
   496→                className={cn("w-full h-full max-h-[500px] object-contain", !sharing && !activePresenterName && "hidden")}
   497→              />
   498→              {!sharing && !activePresenterName && (
   499→                <div className="text-center p-8">
   500→                  <Monitor className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   501→                  <p className="text-sm text-muted-foreground mb-2">No one is presenting yet.</p>
   502→                  {isHost ? (
   503→                    <Button onClick={toggleShare} variant="outline"><ScreenShare className="h-4 w-4 mr-1.5" /> Share Your Screen</Button>
   504→                  ) : (
   505→                    <Button onClick={requestPresent} variant="outline"><Hand className="h-4 w-4 mr-1.5" /> Request to Present</Button>
   506→                  )}
   507→                </div>
   508→              )}
   509→              {/* hidden voice container */}
   510→              <div ref={voiceContainerRef} className="hidden" />
   511→            </Card>
   512→          )}
   513→
   514→          {/* Controls */}
   515→          <Card className="p-3 flex items-center justify-center gap-2">
   516→            <Button
   517→              variant={micOn ? "default" : "outline"}
   518→              size="sm"
   519→              onClick={toggleMic}
   520→              className={micOn ? "bg-emerald-500 text-emerald-950 hover:bg-emerald-400" : ""}
   521→            >
   522→              {micOn ? <Mic className="h-4 w-4" /> : <MicOff className="h-4 w-4" />}
   523→              <span className="ml-1.5">{micOn ? "Mute" : "Unmute"}</span>
   524→            </Button>
   525→            <Button variant={sharing ? "destructive" : "outline"} size="sm" onClick={toggleShare}>
   526→              {sharing ? <ScreenShareOff className="h-4 w-4" /> : <ScreenShare className="h-4 w-4" />}
   527→              <span className="ml-1.5">{sharing ? "Stop Share" : "Share Screen"}</span>
   528→            </Button>
   529→            {!isHost && !sharing && (
   530→              <Button variant="outline" size="sm" onClick={requestPresent}>
   531→                <Hand className="h-4 w-4" /><span className="ml-1.5">Raise Hand</span>
   532→              </Button>
   533→            )}
   534→            <div className="h-6 w-px bg-border mx-1" />
   535→            {recording ? (
   536→              <Button variant="destructive" size="sm" onClick={stopRecording} className="gap-1.5">
   537→                <Square className="h-3.5 w-3.5 fill-current" />
   538→                <span className="font-mono text-xs">{formatRecordTime(recordSeconds)}</span>
   539→                <span className="ml-1">Stop Rec</span>
   540→              </Button>
   541→            ) : (
   542→              <Button
   543→                variant="outline"
   544→                size="sm"
   545→                onClick={startRecording}
   546→                disabled={stageMode !== "screen" || (!sharing && !activePresenterName)}
   547→                title={stageMode !== "screen" ? "Switch to Screen Share to record" : (!sharing && !activePresenterName) ? "Waiting for a screen share to record" : "Record the screen share"}
   548→              >
   549→                <Circle className="h-3.5 w-3.5 text-red-500 fill-red-500" />
   550→                <span className="ml-1.5">Record</span>
   551→              </Button>
   552→            )}
   553→            <div className="h-6 w-px bg-border mx-1" />
   554→            <Button variant="ghost" size="sm" onClick={() => setShowChat((s) => !s)}>
   555→              <MessageSquare className="h-4 w-4" /><span className="ml-1.5">Chat</span>
   556→            </Button>
   557→            <Button variant="ghost" size="sm" className="text-red-400 hover:text-red-500" onClick={leaveCall}>
   558→              <PhoneOff className="h-4 w-4" /><span className="ml-1.5">Leave</span>
   559→            </Button>
   560→          </Card>
   561→
   562→          {/* Present requests (host only) */}
   563→          {isHost && presentRequests.length > 0 && (
   564→            <Card className="p-4 border-amber-500/30 bg-amber-500/5">
   565→              <h3 className="text-sm font-semibold flex items-center gap-2 mb-3"><Hand className="h-4 w-4 text-amber-400" /> Presentation Requests</h3>
   566→              <div className="space-y-2">
   567→                {presentRequests.map((req) => (
   568→                  <div key={req.socketId} className="flex items-center justify-between p-2 rounded-lg bg-background/50">
   569→                    <span className="text-sm">{req.userName} wants to present</span>
   570→                    <div className="flex gap-2">
   571→                      <Button size="sm" onClick={() => approveRequest(req)}>Approve</Button>
   572→                      <Button size="sm" variant="ghost" onClick={() => setPresentRequests((r) => r.filter((x) => x.socketId !== req.socketId))}>Dismiss</Button>
   573→                    </div>
   574→                  </div>
   575→                ))}
   576→              </div>
   577→            </Card>
   578→          )}
   579→        </div>
   580→
   581→        {/* Sidebar: participants + chat */}
   582→        <div className="space-y-4">
   583→          {/* Participants */}
   584→          <Card className="p-4">
   585→            <h3 className="font-semibold text-sm flex items-center gap-2 mb-3">
   586→              <Users className="h-4 w-4 text-emerald-400" /> Participants
   587→              <Badge variant="outline" className="text-[10px] ml-auto">{peers.length + 1}</Badge>
   588→            </h3>
   589→            <ScrollArea className="h-48 pr-2">
   590→              <div className="space-y-2">
   591→                <ParticipantRow name={userName + " (You)"} role={isHost ? "Host" : "You"} micOn={micOn} presenting={sharing} isYou />
   592→                {peers.map((p) => (
   593→                  <ParticipantRow key={p.socketId} name={p.userName} role={p.role} micOn={p.micOn} presenting={p.isPresenting} />
   594→                ))}
   595→                {peers.length === 0 && <p className="text-xs text-muted-foreground text-center py-4">Waiting for others to join...</p>}
   596→              </div>
   597→            </ScrollArea>
   598→          </Card>
   599→
   600→          {/* Chat */}
   601→          {showChat && (
   602→            <Card className="p-4 flex flex-col h-[300px]">
   603→              <h3 className="font-semibold text-sm flex items-center gap-2 mb-3">
   604→                <MessageSquare className="h-4 w-4 text-cyan-400" /> Live Chat
   605→              </h3>
   606→              <ScrollArea className="flex-1 pr-2">
   607→                <div className="space-y-2">
   608→                  {chat.length === 0 && <p className="text-xs text-muted-foreground text-center py-4">No messages yet. Say hi!</p>}
   609→                  {chat.map((m, i) => (
   610→                    <div key={i} className="text-xs">
   611→                      <span className="font-medium text-emerald-400">{m.userName}: </span>
   612→                      <span className="text-muted-foreground">{m.message}</span>
   613→                    </div>
   614→                  ))}
   615→                </div>
   616→              </ScrollArea>
   617→              <div className="flex gap-1 mt-2">
   618→                <Input placeholder="Message..." value={chatInput} onChange={(e) => setChatInput(e.target.value)} onKeyDown={(e) => e.key === "Enter" && sendChat()} className="h-8 text-xs" />
   619→                <Button size="icon" className="h-8 w-8 shrink-0" onClick={sendChat}><Send className="h-3.5 w-3.5" /></Button>
   620→              </div>
   621→            </Card>
   622→          )}
   623→
   624→          <Card className="p-3 border-cyan-500/20 bg-cyan-500/5">
   625→            <div className="flex items-start gap-2">
   626→              <ShieldAlert className="h-4 w-4 text-cyan-400 shrink-0 mt-0.5" />
   627→              <p className="text-[11px] text-muted-foreground leading-relaxed">
   628→                Media is peer-to-peer via WebRTC. Grant mic/screen permissions when your browser prompts. Echo cancellation is on by default.
   629→              </p>
   630→            </div>
   631→          </Card>
   632→        </div>
   633→      </div>
   634→    </div>
   635→  )
   636→}

   638→function ParticipantRow({ name, role, micOn, presenting, isYou }: { name: string; role: string; micOn: boolean; presenting: boolean; isYou?: boolean }) {

   652→    </div>
