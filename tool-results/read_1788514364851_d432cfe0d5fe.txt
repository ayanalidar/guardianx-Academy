   190→function relativeTime(iso: string): string {
   191→  const d = new Date(iso)
   192→  const diff = Date.now() - d.getTime()
   193→  const min = Math.floor(diff / 60000)
   194→  if (min < 1) return "just now"
   195→  if (min < 60) return `${min}m ago`
   196→  const hr = Math.floor(min / 60)
   197→  if (hr < 24) return `${hr}h ago`
   198→  const day = Math.floor(hr / 24)
   199→  if (day < 7) return `${day}d ago`
   200→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" })
   201→}

   203→function formatClock(d: Date): string {
   204→  return d.toLocaleTimeString(undefined, {
   205→    hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false,
   206→  })
   207→}

   209→function deriveServices(lab: LabListItem): string[] {
   210→  const t = (lab.tags || "").toLowerCase()
   211→  const out: string[] = []
   212→  if (t.includes("ssh") || lab.commands?.includes("ssh")) out.push("SSH")
   213→  if (t.includes("http") || t.includes("web")) out.push("HTTP")
   214→  if (t.includes("ftp") || t.includes("smb")) out.push("FTP")
   215→  if (t.includes("dns") || t.includes("smtp")) out.push("DNS")
   216→  if (t.includes("mysql") || t.includes("sql")) out.push("MySQL")
   217→  if (out.length === 0) out.push("SSH", "HTTP")
   218→  return out.slice(0, 3)
   219→}

   221→function pseudoIp(seed: string): string {
   222→  let h = 0
   223→  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) | 0
   224→  const a = (Math.abs(h) % 200) + 10
   225→  const b = (Math.abs(h >> 8) % 200) + 10
   226→  return `10.10.${a % 250}.${b % 250}`
   227→}
