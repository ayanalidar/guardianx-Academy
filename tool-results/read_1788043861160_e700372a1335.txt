   201→function MyCoursesTab() {
   202→  const { navigate } = useAppStore()
   203→  const [createOpen, setCreateOpen] = React.useState(false)
   204→  const [editingCourseId, setEditingCourseId] = React.useState<string | null>(null)
   205→
   206→  const { data, isLoading } = useQuery<{ courses: InstructorCourse[]; totals: any }>({
   207→    queryKey: ["instructor", "courses"],
   208→    queryFn: () => api("/api/instructor/courses"),
   209→  })
   210→  const courses = data?.courses ?? []
   211→
   212→  if (isLoading) {
   213→    return (
   214→      <div className="space-y-4">
   215→        {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-32" />)}
   216→      </div>
   217→    )
   218→  }
   219→
   220→  if (editingCourseId) {
   221→    return <CourseEditor courseId={editingCourseId} onBack={() => setEditingCourseId(null)} />
   222→  }
   223→
   224→  return (
   225→    <div className="space-y-4">
   226→      <div className="flex items-center justify-between">
   227→        <div>
   228→          <h2 className="text-xl font-bold flex items-center gap-2">
   229→            <BookOpen className="h-5 w-5 text-cyan-400" /> My Courses
   230→          </h2>
   231→          <p className="text-sm text-muted-foreground">Create and manage your course catalog</p>
   232→        </div>
   233→        <Button onClick={() => setCreateOpen(true)} className="bg-cyan-500 text-cyan-950 hover:bg-cyan-400">
   234→          <Plus className="h-4 w-4 mr-1.5" /> Create Course
   235→        </Button>
   236→      </div>
   237→
   238→      {courses.length === 0 ? (
   239→        <Card className="p-12 text-center border-dashed">
   240→          <BookOpen className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   241→          <p className="font-medium mb-1">No courses yet</p>
   242→          <p className="text-sm text-muted-foreground mb-4">Create your first course to get started.</p>
   243→          <Button onClick={() => setCreateOpen(true)} className="bg-cyan-500 text-cyan-950 hover:bg-cyan-400">
   244→            <Plus className="h-4 w-4 mr-1.5" /> Create Your First Course
   245→          </Button>
   246→        </Card>
   247→      ) : (
   248→        <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   249→          {courses.map((c) => (
   250→            <InstructorCourseCard
   251→              key={c.id}
   252→              course={c}
   253→              onOpen={() => navigate({ name: "course", courseId: c.id })}
   254→              onEdit={() => setEditingCourseId(c.id)}
   255→            />
   256→          ))}
   257→        </div>
   258→      )}
   259→
   260→      <CreateCourseDialog open={createOpen} onOpenChange={setCreateOpen} onCreated={(id) => {
   261→        setCreateOpen(false)
   262→        setEditingCourseId(id)
   263→      }} />
   264→    </div>
   265→  )
   266→}

   268→function InstructorCourseCard({ course, onOpen, onEdit }: {
   269→  course: InstructorCourse
   270→  onOpen: () => void
   271→  onEdit: () => void
   272→}) {
   273→  const qc = useQueryClient()
   274→  const col = colorFor(course.color)
   275→
   276→  const deleteCourse = useMutation({
   277→    mutationFn: () => api(`/api/instructor/courses/${course.id}/edit`, { method: "DELETE" }),
   278→    onSuccess: () => {
   279→      qc.invalidateQueries({ queryKey: ["instructor", "courses"] })
   280→      toast.success("Course deleted")
   281→    },
   282→    onError: (e: any) => toast.error(e.message),
   283→  })
   284→
   285→  return (
   286→    <Card className="overflow-hidden card-hover h-full flex flex-col">
   287→      <div className={cn("relative h-32 bg-gradient-to-br", col.gradient, "flex items-center justify-center overflow-hidden")}>
   288→        {course.thumbnail ? (
   289→          <img
   290→            src={course.thumbnail}
   291→            alt={course.title}
   292→            className="absolute inset-0 w-full h-full object-cover"
   293→            onError={(e) => { (e.target as HTMLImageElement).style.display = "none" }}
   294→          />
   295→        ) : (
   296→          <>
   297→            <div className="absolute inset-0 bg-grid opacity-40" />
   298→            <span className={cn("relative font-mono font-bold text-3xl", col.text)}>{course.shortName}</span>
   299→          </>
   300→        )}
   301→        <div className="absolute top-2 right-2 flex gap-1">
   302→          <Badge variant="outline" className={cn("text-[10px]", LEVEL_COLORS[course.level])}>{course.level}</Badge>
   303→        </div>
   304→        <div className="absolute bottom-2 left-2">
   305→          <Badge variant="outline" className="text-[10px] bg-background/60 backdrop-blur">{course.certBody || course.category}</Badge>
   306→        </div>
   307→      </div>
   308→      <div className="p-4 flex-1 flex flex-col">
   309→        <h3 className="font-semibold text-sm mb-1 line-clamp-1">{course.title}</h3>
   310→        <p className="text-xs text-muted-foreground line-clamp-2 mb-3 flex-1">{course.description}</p>
   311→        <div className="grid grid-cols-3 gap-2 text-center mb-3 pt-3 border-t border-border">
   312→          <div>
   313→            <div className="text-sm font-bold text-cyan-400 tabular-nums">{course.enrollmentCount}</div>
   314→            <div className="text-[9px] text-muted-foreground uppercase">Students</div>
   315→          </div>
   316→          <div>
   317→            <div className="text-sm font-bold text-emerald-400 tabular-nums">{course.completedStudents}</div>
   318→            <div className="text-[9px] text-muted-foreground uppercase">Done</div>
   319→          </div>
   320→          <div>
   321→            <div className="text-sm font-bold text-violet-400 tabular-nums">{course.avgProgress}%</div>
   322→            <div className="text-[9px] text-muted-foreground uppercase">Avg</div>
   323→          </div>
   324→        </div>
   325→        <div className="flex gap-1.5">
   326→          <Button size="sm" variant="outline" className="flex-1" onClick={onEdit}>
   327→            <Pencil className="h-3 w-3 mr-1" /> Edit
   328→          </Button>
   329→          <Button size="sm" variant="ghost" onClick={onOpen} title="View course">
   330→            <Eye className="h-3.5 w-3.5" />
   331→          </Button>
   332→          <button
   333→            onClick={() => { if (confirm(`Delete "${course.title}"? This will remove all modules, lessons, and quizzes.`)) deleteCourse.mutate() }}
   334→            className="p-1.5 rounded-lg text-muted-foreground hover:text-red-400 hover:bg-red-500/10 transition-colors"
   335→            title="Delete course"
   336→          >
   337→            <Trash2 className="h-3.5 w-3.5" />
   338→          </button>
   339→        </div>
   340→      </div>
   341→    </Card>
   342→  )
   343→}
