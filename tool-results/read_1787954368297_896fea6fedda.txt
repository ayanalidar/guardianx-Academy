   490→   490→function timeAgo(date: string): string {
   491→   491→  const diff = Date.now() - new Date(date).getTime()
   492→   492→  const m = Math.floor(diff / 60000)
   493→   493→  if (m < 1) return "just now"
   494→   494→  if (m < 60) return `${m}m ago`
   495→   495→  const h = Math.floor(m / 60)
   496→   496→  if (h < 24) return `${h}h ago`
   497→   497→  const d = Math.floor(h / 24)
   498→   498→  if (d < 7) return `${d}d ago`
   499→   499→  return new Date(date).toLocaleDateString()
   500→   500→}
