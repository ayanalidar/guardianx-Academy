   185→function ThemeToggle() {
   186→  const { theme, setTheme } = useTheme()
   187→  const [mounted, setMounted] = React.useState(false)
   188→  React.useEffect(() => setMounted(true), [])
   189→  if (!mounted) return <div className="h-9 w-9" />
   190→  return (
   191→    <Button
   192→      variant="ghost"
   193→      size="icon"
   194→      className="h-9 w-9"
   195→      onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   196→      title="Toggle theme"
   197→      aria-label="Toggle dark/light theme"
   198→    >
   199→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   200→    </Button>
   201→  )

   533→function timeAgo(date: string): string {
   534→  const diff = Date.now() - new Date(date).getTime()
   535→  const m = Math.floor(diff / 60000)
   536→  if (m < 1) return "just now"
   537→  if (m < 60) return `${m}m ago`
   538→  const h = Math.floor(m / 60)
   539→  if (h < 24) return `${h}h ago`
   540→  const d = Math.floor(h / 24)
   541→  if (d < 7) return `${d}d ago`
   542→  return new Date(date).toLocaleDateString()
   543→}
