    77→  async function buildCourse(data: {
    78→    slug: string
    79→    title: string
    80→    shortName: string
    81→    description: string
    82→    longDescription: string
    83→    category: string
    84→    level: string
    85→    durationHours: number
    86→    price: number
    87→    rating: number
    88→    studentsCount: number
    89→    color: string
    90→    tags: string
    91→    certBody: string
    92→    instructorId: string
    93→    modules: ModuleInput[]
    94→  }) {
    95→    const existing = await db.course.findUnique({ where: { slug: data.slug } })
    96→    if (existing) {
    97→      // update meta only
    98→      return db.course.update({
    99→        where: { id: existing.id },
   100→        data: {
   101→          title: data.title,
   102→          shortName: data.shortName,
   103→          description: data.description,
   104→          longDescription: data.longDescription,
   105→          category: data.category,
   106→          level: data.level,
   107→          durationHours: data.durationHours,
   108→          price: data.price,
   109→          rating: data.rating,
   110→          studentsCount: data.studentsCount,
   111→          color: data.color,
   112→          tags: data.tags,
   113→          certBody: data.certBody,
   114→          instructorId: data.instructorId,
   115→        },
   116→      })
   117→    }
   118→    const course = await db.course.create({
   119→      data: {
   120→        slug: data.slug,
   121→        title: data.title,
   122→        shortName: data.shortName,
   123→        description: data.description,
   124→        longDescription: data.longDescription,
   125→        category: data.category,
   126→        level: data.level,
   127→        durationHours: data.durationHours,
   128→        price: data.price,
   129→        rating: data.rating,
   130→        studentsCount: data.studentsCount,
   131→        color: data.color,
   132→        tags: data.tags,
   133→        certBody: data.certBody,
   134→        instructorId: data.instructorId,
   135→      },
   136→    })
   137→    for (let mi = 0; mi < data.modules.length; mi++) {
   138→      const m = data.modules[mi]
   139→      const module_ = await db.module.create({
   140→        data: { courseId: course.id, title: m.title, description: m.description, order: mi },
   141→      })
   142→      for (let li = 0; li < m.lessons.length; li++) {
   143→        const l = m.lessons[li]
   144→        const lesson = await db.lesson.create({
   145→          data: {
   146→            moduleId: module_.id,
   147→            title: l.title,
   148→            type: l.type,
   149→            content: l.content,
   150→            pdfPages: l.pdfPages ?? 0,
   151→            durationMin: l.durationMin ?? 15,
   152→            order: li,
   153→            preview: l.preview ?? false,
   154→          },
   155→        })
   156→        if (l.quiz) {
   157→          const quiz = await db.quiz.create({
   158→            data: { lessonId: lesson.id, title: l.quiz.title, description: "Test your knowledge" },
   159→          })
   160→          for (const q of l.quiz.questions) {
   161→            await db.question.create({
   162→              data: {
   163→                quizId: quiz.id,
   164→                text: q.text,
   165→                options: q.options.join("|"),
   166→                answerIndex: q.answerIndex,
   167→                explanation: q.explanation,
   168→              },
   169→            })
   170→          }
   171→        }
   172→      }
   173→    }
   174→    return course
   175→  }
