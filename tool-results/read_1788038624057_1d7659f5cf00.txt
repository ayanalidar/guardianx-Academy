   370→interface Review {
   371→  id: string
   372→  rating: number
   373→  title: string
   374→  content: string
   375→  createdAt: string
   376→  user: { id: string; name: string; title: string | null; avatar: string | null }

   378→
   379→function ReviewsSection({ courseId, isEnrolled }: { courseId: string; isEnrolled: boolean }) {
   380→  const { user } = useUser()
   381→  const qc = useQueryClient()
   382→  const [showForm, setShowForm] = React.useState(false)
   383→  const [rating, setRating] = React.useState(5)
   384→  const [hoverRating, setHoverRating] = React.useState(0)
   385→  const [title, setTitle] = React.useState("")
   386→  const [content, setContent] = React.useState("")
   387→
   388→  const { data, isLoading } = useQuery<{ reviews: Review[]; avgRating: number; totalReviews: number; distribution: { star: number; count: number }[] }>({
   389→    queryKey: ["reviews", courseId],
   390→    queryFn: () => api(`/api/courses/${courseId}/reviews`),
   391→  })
   392→
   393→  const submitMutation = useMutation({
   394→    mutationFn: () => api(`/api/courses/${courseId}/reviews`, {
   395→      method: "POST",
   396→      body: JSON.stringify({ rating, title, content }),
   397→    }),
   398→    onSuccess: () => {
   399→      qc.invalidateQueries({ queryKey: ["reviews", courseId] })
   400→      setShowForm(false); setTitle(""); setContent(""); setRating(5)
   401→      toast.success("Review submitted! Thanks for your feedback.")
   402→    },
   403→    onError: (e: any) => toast.error(e.message),
   404→  })
   405→
   406→  const reviews = data?.reviews ?? []
   407→  const avg = data?.avgRating ?? 0
   408→  const total = data?.totalReviews ?? 0
   409→  const distribution = data?.distribution ?? []
   410→  const myReview = reviews.find((r) => r.user.id === user?.id)
   411→
   412→  return (
   413→    <Card className="p-6">
   414→      <div className="flex items-center justify-between mb-4">
   415→        <h3 className="font-semibold flex items-center gap-2"><Star className="h-4 w-4 text-amber-400" fill="currentColor" /> Student Reviews</h3>
   416→        {isEnrolled && !myReview && (
   417→          <Button size="sm" variant="outline" onClick={() => setShowForm((s) => !s)}>
   418→            <PenLine className="h-3.5 w-3.5 mr-1.5" /> {showForm ? "Cancel" : "Write a Review"}
   419→          </Button>
   420→        )}
   421→        {myReview && (
   422→          <Badge variant="outline" className="text-[10px] text-emerald-400 border-emerald-500/30 bg-emerald-500/10">
   423→            <CheckCircle2 className="h-3 w-3 mr-1" /> You reviewed this
   424→          </Badge>
   425→        )}
   426→      </div>
   427→
   428→      {/* Rating summary */}
   429→      <div className="flex flex-col sm:flex-row items-center gap-6 mb-5 p-4 rounded-lg bg-muted/30">
   430→        <div className="text-center">
   431→          <div className="text-4xl font-bold text-amber-400 tabular-nums">{avg.toFixed(1)}</div>
   432→          <div className="flex gap-0.5 justify-center my-1">
   433→            {[1, 2, 3, 4, 5].map((s) => (
   434→              <Star key={s} className={cn("h-3.5 w-3.5", s <= Math.round(avg) ? "text-amber-400 fill-amber-400" : "text-muted-foreground/40")} />
   435→            ))}
   436→          </div>
   437→          <div className="text-[10px] text-muted-foreground">{total} review{total !== 1 ? "s" : ""}</div>
   438→        </div>
   439→        <div className="flex-1 w-full space-y-1">
   440→          {distribution.slice().reverse().map((d) => (
   441→            <div key={d.star} className="flex items-center gap-2 text-xs">
   442→              <span className="w-3 text-muted-foreground font-mono">{d.star}</span>
   443→              <Star className="h-2.5 w-2.5 text-amber-400 fill-amber-400" />
   444→              <div className="flex-1 h-1.5 rounded-full bg-muted overflow-hidden">
   445→                <div className="h-full bg-amber-400 rounded-full transition-all" style={{ width: `${total ? (d.count / total) * 100 : 0}%` }} />
   446→              </div>
   447→              <span className="w-6 text-right text-muted-foreground font-mono">{d.count}</span>
   448→            </div>
   449→          ))}
   450→        </div>
   451→      </div>
   452→
   453→      {/* Review form */}
   454→      {showForm && (
   455→        <div className="mb-5 p-4 rounded-lg border border-emerald-500/20 bg-emerald-500/[0.03] space-y-3 animate-fade-in-up">
   456→          <div className="flex items-center gap-2">
   457→            <span className="text-sm font-medium">Your rating:</span>
   458→            <div className="flex gap-1">
   459→              {[1, 2, 3, 4, 5].map((s) => (
   460→                <button
   461→                  key={s}
   462→                  type="button"
   463→                  onMouseEnter={() => setHoverRating(s)}
   464→                  onMouseLeave={() => setHoverRating(0)}
   465→                  onClick={() => setRating(s)}
   466→                  className="transition-transform hover:scale-110"
   467→                >
   468→                  <Star className={cn("h-5 w-5", s <= (hoverRating || rating) ? "text-amber-400 fill-amber-400" : "text-muted-foreground/40")} />
   469→                </button>
   470→              ))}
   471→            </div>
   472→          </div>
   473→          <Input placeholder="Review title (optional)..." value={title} onChange={(e) => setTitle(e.target.value)} />
   474→          <Textarea placeholder="Share your experience with this course..." value={content} onChange={(e) => setContent(e.target.value)} className="min-h-[100px]" />
   475→          <Button size="sm" onClick={() => submitMutation.mutate()} disabled={submitMutation.isPending}>
   476→            {submitMutation.isPending ? "Submitting..." : "Submit Review"}
   477→          </Button>
   478→        </div>
   479→      )}
   480→
   481→      {/* Reviews list */}
   482→      {isLoading ? (
   483→        <div className="space-y-3">
   484→          {Array.from({ length: 2 }).map((_, i) => <Skeleton key={i} className="h-24" />)}
   485→        </div>
   486→      ) : reviews.length === 0 ? (
   487→        <div className="text-center py-8">
   488→          <Star className="h-8 w-8 text-muted-foreground/50 mx-auto mb-2" />
   489→          <p className="text-sm text-muted-foreground">No reviews yet. Be the first to review!</p>
   490→        </div>
   491→      ) : (
   492→        <div className="space-y-3">
   493→          {reviews.slice(0, 5).map((r) => (
   494→            <div key={r.id} className="p-3 rounded-lg border border-border hover:border-emerald-500/20 transition-colors">
   495→              <div className="flex items-start gap-3">
   496→                <Avatar className="h-8 w-8 shrink-0">
   497→                  <AvatarFallback className="bg-amber-500/10 text-amber-400 text-[10px]">
   498→                    {r.user.name.split(" ").map((n) => n[0]).join("").toUpperCase().slice(0, 2)}
   499→                  </AvatarFallback>
   500→                </Avatar>
   501→                <div className="flex-1 min-w-0">
   502→                  <div className="flex items-center gap-2 mb-0.5 flex-wrap">
   503→                    <span className="text-sm font-medium">{r.user.name}</span>
   504→                    {r.user.title && <span className="text-[10px] text-muted-foreground">· {r.user.title}</span>}
   505→                    <div className="flex gap-0.5 ml-auto">
   506→                      {[1, 2, 3, 4, 5].map((s) => (
   507→                        <Star key={s} className={cn("h-3 w-3", s <= r.rating ? "text-amber-400 fill-amber-400" : "text-muted-foreground/40")} />
   508→                      ))}
   509→                    </div>
   510→                  </div>
   511→                  {r.title && <p className="text-sm font-medium mb-0.5">{r.title}</p>}
   512→                  {r.content && <p className="text-xs text-muted-foreground leading-relaxed">{r.content}</p>}
   513→                  <div className="flex items-center gap-3 mt-1.5 text-[10px] text-muted-foreground">
   514→                    <span>{new Date(r.createdAt).toLocaleDateString()}</span>
   515→                    {r.user.id === user?.id && <span className="text-emerald-400">· Your review</span>}
   516→                  </div>
   517→                </div>
   518→              </div>
   519→            </div>
   520→          ))}
   521→          {reviews.length > 5 && (
   522→            <p className="text-xs text-center text-muted-foreground py-2">+ {reviews.length - 5} more review{reviews.length - 5 !== 1 ? "s" : ""}</p>
   523→          )}
   524→        </div>
   525→      )}
   526→    </Card>
   527→  )
   528→}
