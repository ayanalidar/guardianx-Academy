    44→
    45→interface Batch {
    46→  id: string
    47→  certification: string
    48→  certGroup: string
    49→  name: string
    50→  schedule: string
    51→  startDate: string
    52→  mode: string
    53→  instructor: string
    54→  seats: number
    55→  enrolled: number
    56→  almostFull: boolean
    57→  level: string
    58→  // visual styling per batch (driven by level)
    59→  certColor: string
    60→  certTint: string
    61→  certBorder: string
    62→  levelColor: string
    63→  levelTint: string
    64→  levelBorder: string
    65→  borderColor: string
    66→  btnClass: string
    67→  // for the schedule filter
    68→  scheduleType: "weekday" | "weekend" | "morning" | "evening" | "late-night"
    69→}
    70→
    71→/* Derive the filter `certGroup` from a certification name. Matches the
    72→ * existing static BATCHES values: "Security+", "CEH", "CCNA", "CISSP".
    73→ * Falls back to the first word of the certification string for unknown
    74→ * certs (so new DB-driven batches still get a useful filter group). */
    75→function deriveCertGroup(cert: string): string {
    76→  const s = cert.toLowerCase()
    77→  if (s.includes("security")) return "Security+"
    78→  if (s.includes("ceh") || s.includes("ethical hacker")) return "CEH"
    79→  if (s.includes("ccna")) return "CCNA"
    80→  if (s.includes("cissp")) return "CISSP"
    81→  return cert.split(/\s+/)[0] || cert
    82→}
    83→
    84→/* Derive the `scheduleType` filter value from a human-readable schedule
    85→ * string. Heuristic: weekend > late-night > morning > evening > weekday. */
    86→function deriveScheduleType(schedule: string): Batch["scheduleType"] {
    87→  const s = schedule.toLowerCase()
    88→  if (s.includes("sat") || s.includes("sun")) return "weekend"
    89→  // Look for "10 pm" / "11 pm" / "12 am" → late-night
    90→  const pmMatch = s.match(/(\d{1,2})\s*pm/)
    91→  if (pmMatch && Number(pmMatch[1]) >= 10) return "late-night"
    92→  if (s.includes(" am")) return "morning"
    93→  if (pmMatch) return "evening"
    94→  return "weekday"
    95→}

    97→/* Normalize a raw DB row (from the API) or a static `UPCOMING_BATCHES`
    98→ * item into the local `Batch` shape used by the JSX + filters. */
    99→function normalizeBatch(raw: {
   100→  id?: string
   101→  certification: string
   102→  name: string
   103→  schedule: string
   104→  startDate: string
   105→  mode: string
   106→  instructor: string
   107→  seats: number
   108→  enrolled?: number
   109→  status?: string
   110→  almostFull?: boolean
   111→  level: string
   112→  certColor: string
   113→  certTint: string
   114→  certBorder: string
   115→  levelColor: string
   116→  levelTint: string
   117→  levelBorder: string
   118→  borderColor: string
   119→  btnClass: string
   120→}): Batch {
   121→  const almostFull =
   122→    raw.almostFull ?? ((raw.seats - (raw.enrolled ?? 0)) <= 2 || raw.status === "Almost Full")
   123→  return {
   124→    id: raw.id ?? `batch-${raw.name}`,
   125→    certification: raw.certification,
   126→    certGroup: deriveCertGroup(raw.certification),
   127→    name: raw.name,
   128→    schedule: raw.schedule,
   129→    startDate: raw.startDate,
   130→    mode: raw.mode,
   131→    instructor: raw.instructor,
   132→    seats: raw.seats,
   133→    enrolled: raw.enrolled ?? 0,
   134→    almostFull,
   135→    level: raw.level,
   136→    certColor: raw.certColor,
   137→    certTint: raw.certTint,
   138→    certBorder: raw.certBorder,
   139→    levelColor: raw.levelColor,
   140→    levelTint: raw.levelTint,
   141→    levelBorder: raw.levelBorder,
   142→    borderColor: raw.borderColor,
   143→    btnClass: raw.btnClass,
   144→    scheduleType: deriveScheduleType(raw.schedule),
   145→  }
   146→}
