   408→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   409→  const { navigate } = useAppStore()
   410→  const [q, setQ] = React.useState("")
   411→  const [results, setResults] = React.useState<{ courses: any[]; labs: any[]; notes: any[] }>({ courses: [], labs: [], notes: [] })
   412→
   413→  React.useEffect(() => {
   414→    if (!q.trim()) {
   415→      setResults({ courses: [], labs: [], notes: [] })
   416→      return
   417→    }
   418→    const t = setTimeout(async () => {
   419→      try {
   420→        const r = await fetch(`/api/search?q=${encodeURIComponent(q)}`)
   421→        const data = await r.json()
   422→        setResults(data)
   423→      } catch {}
   424→    }, 150)
   425→    return () => clearTimeout(t)
   426→  }, [q])
   427→
   428→  React.useEffect(() => {
   429→    if (!open) setQ("")
   430→  }, [open])

   563→function timeAgo(date: string): string {
   564→  const diff = Date.now() - new Date(date).getTime()
   565→  const m = Math.floor(diff / 60000)
   566→  if (m < 1) return "just now"
   567→  if (m < 60) return `${m}m ago`
   568→  const h = Math.floor(m / 60)
   569→  if (h < 24) return `${h}h ago`
   570→  const d = Math.floor(h / 24)
   571→  if (d < 7) return `${d}d ago`
   572→  return new Date(date).toLocaleDateString()
   573→}
