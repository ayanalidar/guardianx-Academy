   138→function CoursesTab() {
   139→  const { data, isLoading } = useQuery<{ courses: AdminCourse[] }>({
   140→    queryKey: ["admin", "courses"],
   141→    queryFn: () => api("/api/admin/courses"),
   142→  })
   143→  const [showNew, setShowNew] = React.useState(false)
   144→  const [editingId, setEditingId] = React.useState<string | null>(null)
   145→  const [managingId, setManagingId] = React.useState<string | null>(null)
   146→  const [search, setSearch] = React.useState("")
   147→
   148→  const courses = data?.courses ?? []
   149→  const filtered = courses.filter(
   150→    (c) => c.title.toLowerCase().includes(search.toLowerCase()) || c.shortName.toLowerCase().includes(search.toLowerCase())
   151→  )
   152→
   153→  if (isLoading) {
   154→    return (
   155→      <div className="space-y-4">
   156→        {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-20" />)}
   157→      </div>
   158→    )
   159→  }
   160→
   161→  return (
   162→    <div className="space-y-4">
   163→      {/* Toolbar */}
   164→      <div className="flex flex-col sm:flex-row gap-3 items-stretch sm:items-center">
   165→        <div className="relative flex-1">
   166→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   167→          <Input
   168→            value={search}
   169→            onChange={(e) => setSearch(e.target.value)}
   170→            placeholder="Search courses by title or short name..."
   171→            className="pl-9"
   172→          />
   173→        </div>
   174→        <Button onClick={() => setShowNew((s) => !s)} className="bg-emerald-500 text-emerald-950 hover:bg-emerald-400">
   175→          <Plus className="h-4 w-4 mr-1" /> {showNew ? "Cancel" : "Add Course"}
   176→        </Button>
   177→      </div>
   178→
   179→      {/* New course form */}
   180→      {showNew && (
   181→        <CourseForm
   182→          mode="create"
   183→          onClose={() => setShowNew(false)}
   184→        />
   185→      )}
   186→
   187→      {/* Course list */}
   188→      {filtered.length === 0 ? (
   189→        <Card className="p-12 text-center border-dashed">
   190→          <BookOpen className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   191→          <p className="font-medium mb-1">No courses found</p>
   192→          <p className="text-sm text-muted-foreground">Add a course to get started.</p>
   193→        </Card>
   194→      ) : (
   195→        <div className="space-y-3">
   196→          {filtered.map((c) => {
   197→            const col = colorFor(c.color)
   198→            return (
   199→              <Card key={c.id} className="overflow-hidden">
   200→                <div className="p-4">
   201→                  <div className="flex items-center gap-4">
   202→                    <div className={cn("flex h-12 w-12 shrink-0 items-center justify-center rounded-xl font-mono font-bold border", col.bg, col.border, col.text)}>
   203→                      {c.shortName.slice(0, 4)}
   204→                    </div>
   205→                    <div className="flex-1 min-w-0">
   206→                      <div className="flex items-center gap-2 mb-1 flex-wrap">
   207→                        <h3 className="font-semibold truncate">{c.title}</h3>
   208→                        <Badge variant="outline" className={cn("text-[10px]", LEVEL_COLORS[c.level])}>{c.level}</Badge>
   209→                        <Badge variant="outline" className="text-[10px]">{c.category}</Badge>
   210→                        {!c.published && <Badge variant="outline" className="text-[10px] text-amber-400 border-amber-500/30">Draft</Badge>}
   211→                      </div>
   212→                      <div className="flex items-center gap-4 text-xs text-muted-foreground">
   213→                        <span className="flex items-center gap-1"><FolderOpen className="h-3 w-3" />{c.moduleCount} modules</span>
   214→                        <span className="flex items-center gap-1"><FileText className="h-3 w-3" />{c.lessonCount} lessons</span>
   215→                        <span className="flex items-center gap-1"><Users className="h-3 w-3" />{c.enrollmentCount} enrolled</span>
   216→                        <span className="flex items-center gap-1"><Star className="h-3 w-3 text-amber-400 fill-amber-400" />{c.rating}</span>
   217→                        <span className="flex items-center gap-1"><DollarSign className="h-3 w-3" />${c.price}</span>
   218→                      </div>
   219→                    </div>
   220→                    <div className="flex items-center gap-1">
   221→                      <Button
   222→                        size="sm"
   223→                        variant="outline"
   224→                        onClick={() => setManagingId(managingId === c.id ? null : c.id)}
   225→                      >
   226→                        <FolderOpen className="h-3.5 w-3.5 mr-1" /> Modules
   227→                      </Button>
   228→                      <Button
   229→                        size="sm"
   230→                        variant="outline"
   231→                        onClick={() => setEditingId(editingId === c.id ? null : c.id)}
   232→                      >
   233→                        <Pencil className="h-3.5 w-3.5 mr-1" /> Edit
   234→                      </Button>
   235→                      <DeleteCourseButton course={c} />
   236→                    </div>
   237→                  </div>
   238→                </div>
   239→                {editingId === c.id && (
   240→                  <div className="border-t border-border">
   241→                    <CourseForm mode="edit" course={c} onClose={() => setEditingId(null)} />
   242→                  </div>
   243→                )}
   244→              </Card>
   245→            )
   246→          })}
   247→        </div>
   248→      )}
   249→
   250→      {/* Modules manager (rendered as expanded panel below the list) */}
   251→      {managingId && (
   252→        <ModulesManager courseId={managingId} onClose={() => setManagingId(null)} />
   253→      )}
   254→    </div>
   255→  )
   256→}

   258→// Delete course with confirmation
   259→function DeleteCourseButton({ course }: { course: AdminCourse }) {
   260→  const qc = useQueryClient()
   261→  const [open, setOpen] = React.useState(false)
   262→  const del = useMutation({
   263→    mutationFn: () => api(`/api/admin/courses/${course.id}`, { method: "DELETE" }),
   264→    onSuccess: () => {
   265→      qc.invalidateQueries({ queryKey: ["admin", "courses"] })
   266→      qc.invalidateQueries({ queryKey: ["courses"] })
   267→      qc.invalidateQueries({ queryKey: ["courses", "all"] })
   268→      qc.invalidateQueries({ queryKey: ["stats"] })
   269→      toast.success(`Course "${course.title}" deleted`)
   270→      setOpen(false)
   271→    },
   272→    onError: (e: any) => toast.error(e.message),
   273→  })
   274→  return (
   275→    <AlertDialog open={open} onOpenChange={setOpen}>
   276→      <button
   277→        onClick={() => setOpen(true)}
   278→        className="p-2 rounded-lg text-muted-foreground hover:text-red-400 hover:bg-red-500/10 transition-colors"
   279→        title="Delete course"
   280→        aria-label="Delete course"
   281→      >
   282→        <Trash2 className="h-4 w-4" />
   283→      </button>
   284→      <AlertDialogContent>
   285→        <AlertDialogHeader>
   286→          <AlertDialogTitle>Delete course "{course.title}"?</AlertDialogTitle>
   287→          <AlertDialogDescription>
   288→            This will permanently delete the course and ALL its modules, lessons, enrollments,
   289→            certificates, discussions, and bookmarks. This action cannot be undone.
   290→          </AlertDialogDescription>
   291→        </AlertDialogHeader>
   292→        <AlertDialogFooter>
   293→          <AlertDialogCancel>Cancel</AlertDialogCancel>
   294→          <AlertDialogAction
   295→            className="bg-red-500 text-white hover:bg-red-600"
   296→            onClick={() => del.mutate()}
   297→            disabled={del.isPending}
   298→          >
   299→            {del.isPending ? "Deleting..." : "Delete permanently"}
   300→          </AlertDialogAction>
   301→        </AlertDialogFooter>
   302→      </AlertDialogContent>
   303→    </AlertDialog>
   304→  )
   305→}

   307→// Course form (create/edit)
   308→function CourseForm({ mode, course, onClose }: {
   309→  mode: "create" | "edit"
   310→  course?: AdminCourse
   311→  onClose: () => void
   312→}) {
   313→  const qc = useQueryClient()
   314→  const [form, setForm] = React.useState({
   315→    title: course?.title ?? "",
   316→    shortName: course?.shortName ?? "",
   317→    slug: course?.slug ?? "",
   318→    description: course?.description ?? "",
   319→    longDescription: course?.longDescription ?? "",
   320→    category: course?.category ?? "Certification",
   321→    level: course?.level ?? "Beginner",
   322→    durationHours: course?.durationHours ?? 40,
   323→    price: course?.price ?? 0,
   324→    color: course?.color ?? "emerald",
   325→    tags: course?.tags ?? "",
   326→    certBody: course?.certBody ?? "",
   327→    published: course?.published ?? true,
   328→    instructorId: course?.instructorId ?? "",
   329→  })
   330→
   331→  const set = (k: keyof typeof form, v: any) => setForm((f) => ({ ...f, [k]: v }))
   332→
   333→  const save = useMutation({
   334→    mutationFn: async () => {
   335→      if (mode === "create") {
   336→        return api("/api/admin/courses", { method: "POST", body: JSON.stringify(form) })
   337→      } else {
   338→        return api(`/api/admin/courses/${course!.id}`, { method: "PATCH", body: JSON.stringify(form) })
   339→      }
   340→    },
   341→    onSuccess: () => {
   342→      qc.invalidateQueries({ queryKey: ["admin", "courses"] })
   343→      qc.invalidateQueries({ queryKey: ["courses"] })
   344→      qc.invalidateQueries({ queryKey: ["courses", "all"] })
   345→      toast.success(mode === "create" ? "Course created" : "Course updated")
   346→      onClose()
   347→    },
   348→    onError: (e: any) => toast.error(e.message),
   349→  })
   350→
   351→  return (
   352→    <div className="p-4 bg-emerald-500/[0.03] space-y-4 animate-fade-in-up">
   353→      <div className="flex items-center gap-2">
   354→        {mode === "create" ? <Plus className="h-4 w-4 text-emerald-400" /> : <Pencil className="h-4 w-4 text-emerald-400" />}
   355→        <span className="text-sm font-semibold">{mode === "create" ? "New Course" : "Edit Course"}</span>
   356→        <Button size="sm" variant="ghost" className="ml-auto h-7" onClick={onClose}><X className="h-3.5 w-3.5" /></Button>
   357→      </div>
   358→      <div className="grid sm:grid-cols-2 gap-3">
   359→        <div className="space-y-1.5">
   360→          <label className="text-xs text-muted-foreground">Title *</label>
   361→          <Input value={form.title} onChange={(e) => set("title", e.target.value)} placeholder="e.g. Certified Ethical Hacker" />
   362→        </div>
   363→        <div className="space-y-1.5">
   364→          <label className="text-xs text-muted-foreground">Short Name *</label>
   365→          <Input value={form.shortName} onChange={(e) => set("shortName", e.target.value)} placeholder="e.g. CEH" />
   366→        </div>
   367→        <div className="space-y-1.5">
   368→          <label className="text-xs text-muted-foreground">Slug (optional — auto-generated)</label>
   369→          <Input value={form.slug} onChange={(e) => set("slug", e.target.value)} placeholder="certified-ethical-hacker" />
   370→        </div>
   371→        <div className="space-y-1.5">
   372→          <label className="text-xs text-muted-foreground">Certification Body</label>
   373→          <Input value={form.certBody} onChange={(e) => set("certBody", e.target.value)} placeholder="e.g. EC-Council" />
   374→        </div>
   375→        <div className="space-y-1.5">
   376→          <label className="text-xs text-muted-foreground">Category</label>
   377→          <Select value={form.category} onValueChange={(v) => set("category", v)}>
   378→            <SelectTrigger><SelectValue /></SelectTrigger>
   379→            <SelectContent>
   380→              {COURSE_CATEGORIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   381→            </SelectContent>
   382→          </Select>
   383→        </div>
   384→        <div className="space-y-1.5">
   385→          <label className="text-xs text-muted-foreground">Level</label>
   386→          <Select value={form.level} onValueChange={(v) => set("level", v)}>
   387→            <SelectTrigger><SelectValue /></SelectTrigger>
   388→            <SelectContent>
   389→              {COURSE_LEVELS.map((l) => <SelectItem key={l} value={l}>{l}</SelectItem>)}
   390→            </SelectContent>
   391→          </Select>
   392→        </div>
   393→        <div className="space-y-1.5">
   394→          <label className="text-xs text-muted-foreground">Duration (hours)</label>
   395→          <Input type="number" value={form.durationHours} onChange={(e) => set("durationHours", Number(e.target.value))} />
   396→        </div>
   397→        <div className="space-y-1.5">
   398→          <label className="text-xs text-muted-foreground">Price (USD)</label>
   399→          <Input type="number" value={form.price} onChange={(e) => set("price", Number(e.target.value))} />
   400→        </div>
   401→        <div className="space-y-1.5">
   402→          <label className="text-xs text-muted-foreground">Theme Color</label>
   403→          <Select value={form.color} onValueChange={(v) => set("color", v)}>
   404→            <SelectTrigger><SelectValue /></SelectTrigger>
   405→            <SelectContent>
   406→              {COURSE_COLORS.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   407→            </SelectContent>
   408→          </Select>
   409→        </div>
   410→        <div className="space-y-1.5">
   411→          <label className="text-xs text-muted-foreground">Tags (comma separated)</label>
   412→          <Input value={form.tags} onChange={(e) => set("tags", e.target.value)} placeholder="pentesting, ethics, oscp" />
   413→        </div>
   414→      </div>
   415→      <div className="space-y-1.5">
   416→        <label className="text-xs text-muted-foreground">Short Description</label>
   417→        <Input value={form.description} onChange={(e) => set("description", e.target.value)} placeholder="1-2 sentence summary" />
   418→      </div>
   419→      <div className="space-y-1.5">
   420→        <label className="text-xs text-muted-foreground">Long Description (Markdown supported)</label>
   421→        <Textarea
   422→          value={form.longDescription}
   423→          onChange={(e) => set("longDescription", e.target.value)}
   424→          placeholder="Full course description, what students will learn, prerequisites..."
   425→          className="min-h-[120px] font-mono text-xs"
   426→        />
   427→      </div>
   428→      <label className="flex items-center gap-2 text-sm cursor-pointer">
   429→        <input
   430→          type="checkbox"
   431→          checked={form.published}
   432→          onChange={(e) => set("published", e.target.checked)}
   433→          className="rounded border-border"
   434→        />
   435→        Published (visible to students)
   436→      </label>
   437→      <div className="flex items-center justify-end gap-2">
   438→        <Button size="sm" variant="ghost" onClick={onClose}>Cancel</Button>
   439→        <Button
   440→          size="sm"
   441→          onClick={() => save.mutate()}
   442→          disabled={!form.title.trim() || !form.shortName.trim() || save.isPending}
   443→          className="bg-emerald-500 text-emerald-950 hover:bg-emerald-400"
   444→        >
   445→          <Save className="h-3.5 w-3.5 mr-1" /> {save.isPending ? "Saving..." : mode === "create" ? "Create Course" : "Save Changes"}
   446→        </Button>
   447→      </div>
   448→    </div>
   449→  )
   450→}

   452→// Modules manager (with lessons)
   453→function ModulesManager({ courseId, onClose }: { courseId: string; onClose: () => void }) {
   454→  const { data } = useQuery<any>({
