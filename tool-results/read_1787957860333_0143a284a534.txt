   324→// Announcement composer
   325→function AnnouncementComposer() {
   326→  const qc = useQueryClient()
   327→  const [open, setOpen] = React.useState(false)
   328→  const [title, setTitle] = React.useState("")
   329→  const [message, setMessage] = React.useState("")
   330→
   331→  const post = useMutation({
   332→    mutationFn: () => api("/api/school/announcements", {
   333→      method: "POST",
   334→      body: JSON.stringify({ title, message }),
   335→    }),
   336→    onSuccess: (r: any) => {
   337→      qc.invalidateQueries({ queryKey: ["notifications"] })
   338→      toast.success(`Announcement sent to ${r.sent} students`)
   339→      setTitle("")
   340→      setMessage("")
   341→      setOpen(false)
   342→    },
   343→    onError: (e: any) => toast.error(e.message),
   344→  })
   345→
   346→  if (!open) {
   347→    return (
   348→      <div className="flex flex-col justify-center">
   349→        <Button variant="outline" onClick={() => setOpen(true)} className="border-amber-500/30 text-amber-400 hover:bg-amber-500/10">
   350→          <Megaphone className="h-4 w-4 mr-1.5" /> Post Announcement
   351→        </Button>
   352→      </div>
   353→    )
   354→  }
   355→
   356→  return (
   357→    <div className="w-full lg:max-w-md p-4 rounded-xl border border-amber-500/20 bg-amber-500/[0.03] space-y-3">
   358→      <div className="flex items-center gap-2">
   359→        <Megaphone className="h-4 w-4 text-amber-400" />
   360→        <span className="text-sm font-semibold">Post Announcement</span>
   361→        <Button size="sm" variant="ghost" className="ml-auto h-7" onClick={() => setOpen(false)}><X className="h-3.5 w-3.5" /></Button>
   362→      </div>
   363→      <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Announcement title" />
   364→      <Textarea
   365→        value={message}
   366→        onChange={(e) => setMessage(e.target.value)}
   367→        placeholder="Your message to all students..."
   368→        className="min-h-[80px] text-xs"
   369→      />
   370→      <Button size="sm" onClick={() => post.mutate()} disabled={!title.trim() || !message.trim() || post.isPending} className="bg-amber-500 text-amber-950 hover:bg-amber-400">
   371→        <Megaphone className="h-3.5 w-3.5 mr-1" /> {post.isPending ? "Sending..." : "Send to all students"}
   372→      </Button>
   373→    </div>
   374→  )
   375→}

   813→function AttendanceMarker({ batchId, date }: { batchId: string; date: string }) {
   814→  const qc = useQueryClient()
   815→  const { data: batchData } = useQuery<any>({
   816→    queryKey: ["school", "batch", batchId],
   817→    queryFn: () => api(`/api/school/batches/${batchId}`),
   818→  })
   819→  const { data: existingAtt } = useQuery<any>({
   820→    queryKey: ["school", "attendance", batchId, date],
   821→    queryFn: () => api(`/api/school/attendance?batchId=${batchId}&date=${date}`),
   822→    enabled: !!batchId && !!date,
   823→  })
   824→
   825→  const batch = batchData?.batch
   826→  const students: any[] = batch?.students ?? []
   827→  const existingMap: Record<string, any> = {}
   828→  for (const a of (existingAtt?.attendance ?? [])) {
   829→    existingMap[a.batchStudentId] = a
   830→  }
   831→
   832→  // Local status map (batchStudentId -> status)
   833→  const [statusMap, setStatusMap] = React.useState<Record<string, string>>({})
   834→
   835→  React.useEffect(() => {
   836→    const m: Record<string, string> = {}
   837→    for (const s of students) {
   838→      m[s.id] = existingMap[s.id]?.status ?? "present"
   839→    }
   840→    setStatusMap(m)
   841→  }, [students.length, existingAtt?.attendance?.length, date])
   842→
   843→  const save = useMutation({
   844→    mutationFn: () => {
   845→      const records = students.map((s) => ({
   846→        batchStudentId: s.id,
   847→        status: statusMap[s.id] ?? "present",
   848→      }))
   849→      return api("/api/school/attendance", {
   850→        method: "POST",
   851→        body: JSON.stringify({ batchId, date, records }),
   852→      })
   853→    },
   854→    onSuccess: () => {
