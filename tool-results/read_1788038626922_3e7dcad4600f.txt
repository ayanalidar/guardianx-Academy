   365→   365→// ---- PDF-style paginated document viewer ----
   366→   366→function PdfStyleViewer({ content, title, totalPages, page, onPageChange }: {
   367→   367→  content: string; title: string; totalPages: number; page: number; onPageChange: (p: number) => void
   368→   368→}) {
   369→   369→  // Split content into pseudo-pages by headings or chunks
   370→   370→  const pages = React.useMemo(() => {
   371→   371→    const blocks = content.split(/\n(?=#{1,3}\s)/).filter(Boolean)
   372→   372→    if (blocks.length <= 1) {
   373→   373→      // split by double newline into chunks of ~3 paragraphs
   374→   374→      const paras = content.split(/\n\n+/)
   375→   375→      const chunks: string[] = []
   376→   376→      for (let i = 0; i < paras.length; i += 3) chunks.push(paras.slice(i, i + 3).join("\n\n"))
   377→   377→      return chunks.length ? chunks : [content]
   378→   378→    }
   379→   379→    return blocks
   380→   380→  }, [content])
   381→   381→
   382→   382→  const numPages = Math.max(pages.length, totalPages, 1)
   383→   383→  const current = Math.min(page, numPages)
   384→   384→  const pageContent = pages[current - 1] ?? pages[0] ?? content
   385→   385→
   386→   386→  return (
   387→   387→    <div className="space-y-3">
   388→   388→      {/* Toolbar */}
   389→   389→      <div className="flex items-center justify-between gap-2 p-2 rounded-lg border border-border bg-card/50">
   390→   390→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   391→   391→          <FileText className="h-4 w-4 text-emerald-400" />
   392→   392→          <span className="font-mono truncate max-w-[200px]">{title}.pdf</span>
   393→   393→        </div>
   394→   394→        <div className="flex items-center gap-1">
   395→   395→          <Button variant="ghost" size="icon" className="h-7 w-7" disabled={current <= 1} onClick={() => onPageChange(current - 1)}>
   396→   396→            <ChevronUp className="h-3.5 w-3.5" />
   397→   397→          </Button>
   398→   398→          <span className="text-xs font-mono px-2">{current} / {numPages}</span>
   399→   399→          <Button variant="ghost" size="icon" className="h-7 w-7" disabled={current >= numPages} onClick={() => onPageChange(current + 1)}>
   400→   400→            <ChevronDown className="h-3.5 w-3.5" />
   401→   401→          </Button>
   402→   402→        </div>
   403→   403→      </div>
   404→   404→
   405→   405→      {/* Page */}
   406→   406→      <div className="relative rounded-lg border border-border bg-[oklch(0.97_0.005_150)] dark:bg-[oklch(0.13_0.015_200)] overflow-hidden shadow-xl">

   484→   484→function QuizPanel({ quiz, lessonId }: { quiz: NonNullable<LessonData["quiz"]>; lessonId: string }) {
   485→   485→  const qc = useQueryClient()
   486→   486→  const [answers, setAnswers] = React.useState<Record<string, number>>({})
   487→   487→  const [submitted, setSubmitted] = React.useState(false)
   488→   488→  const [result, setResult] = React.useState<any>(null)
   489→   489→
   490→   490→  const submitMutation = useMutation({
   491→   491→    mutationFn: () => api(`/api/quizzes/${quiz.id}/attempt`, { method: "POST", body: JSON.stringify({ answers }) }),
   492→   492→    onSuccess: (data) => {
   493→   493→      setResult(data)
   494→   494→      setSubmitted(true)
   495→   495→      qc.invalidateQueries({ queryKey: ["me"] })

   497→   497→      else toast.error(`Quiz not passed. Score: ${data.score}%`)
   498→   498→    },
   499→   499→    onError: (e: any) => toast.error(e.message),
   500→   500→  })
   501→   501→
   502→   502→  const allAnswered = quiz.questions.every((q) => answers[q.id] !== undefined)
   503→   503→
   504→   504→  return (
   505→   505→    <Card className="p-6">
   506→   506→      <div className="flex items-center justify-between mb-4">
   507→   507→        <div>

   509→   509→            <Award className="h-5 w-5 text-amber-400" /> {quiz.title}
   510→   510→          </h3>
   511→   511→          <p className="text-xs text-muted-foreground mt-1">{quiz.questions.length} questions · 70% to pass</p>
   512→   512→        </div>
   513→   513→        {result && (
   514→   514→          <Badge className={result.passed ? "bg-emerald-500/20 text-emerald-400" : "bg-red-500/20 text-red-400"}>

   516→   516→          </Badge>
   517→   517→        )}
   518→   518→      </div>
   519→   519→
   520→   520→      <div className="space-y-6">
   521→   521→        {quiz.questions.map((q, qi) => {
   522→   522→          const selected = answers[q.id]
   523→   523→          const correct = result?.breakdown?.find((b: any) => b.questionId === q.id)?.correctIndex
