  4400→  export type AuthUser = NonNullable<SafeUser>
  4401→  export async function requireRole(
  4402→    roles: string[]
  4403→  ): Promise<AuthUser | NextResponse> {

  4405→    if (!user) return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
  4406→    if (!roles.includes(user.role)) return NextResponse.json({ error: "Forbidden" }, { status: 403 })
  4407→    return user
  4408→  }
