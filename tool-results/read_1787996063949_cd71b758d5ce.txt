   152→function ThemeToggle() {
   153→  const { theme, setTheme } = useTheme()
   154→  const [mounted, setMounted] = React.useState(false)
   155→  React.useEffect(() => setMounted(true), [])
   156→  if (!mounted) return <div className="h-9 w-9" />
   157→  return (
   158→    <Button
   159→      variant="ghost"
   160→      size="icon"
   161→      className="h-9 w-9"
   162→      onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   163→      title="Toggle theme"
   164→      aria-label="Toggle dark/light theme"
   165→    >
   166→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   167→    </Button>
   168→  )
   169→}

   328→// ---- Command Palette (⌘K) ----
   329→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   330→  const { navigate } = useAppStore()
   331→  const [q, setQ] = React.useState("")
   332→  const [results, setResults] = React.useState<{ courses: any[]; labs: any[]; notes: any[] }>({ courses: [], labs: [], notes: [] })
   333→
   334→  React.useEffect(() => {
   335→    if (!q.trim()) {
   336→      setResults({ courses: [], labs: [], notes: [] })
   337→      return
   338→    }
   339→    const t = setTimeout(async () => {
   340→      try {
   341→        const r = await fetch(`/api/search?q=${encodeURIComponent(q)}`)
   342→        const data = await r.json()
   343→        setResults(data)
   344→      } catch {}
   345→    }, 150)
   346→    return () => clearTimeout(t)
   347→  }, [q])
   348→
   349→  React.useEffect(() => {
   350→    if (!open) setQ("")
   351→  }, [open])
   352→
   353→  const navItems = [
   354→    { label: "Dashboard", icon: LayoutDashboard, view: { name: "dashboard" } as const },
   355→    { label: "Course Catalog", icon: BookOpen, view: { name: "catalog" } as const },
   356→    { label: "My Learning", icon: GraduationCap, view: { name: "learning" } as const },
   357→    { label: "Cyber Labs", icon: FlaskConical, view: { name: "labs" } as const },
   358→    { label: "Achievements", icon: Trophy, view: { name: "achievements" } as const },
   359→    { label: "Live Sessions", icon: Radio, view: { name: "live" } as const },
   360→    { label: "My Notes", icon: StickyNote, view: { name: "notes" } as const },
   361→    { label: "Certificates", icon: Award, view: { name: "certificates" } as const },

   363→    { label: "Profile", icon: User, view: { name: "profile" } as const },
   364→  ]
   365→
   366→  const go = (view: any) => {
   367→    navigate(view)
   368→    onOpenChange(false)
   369→  }
   370→
   371→  return (
   372→    <Dialog open={open} onOpenChange={onOpenChange}>
   373→      <DialogContent className="p-0 gap-0 max-w-xl overflow-hidden">
   374→        <DialogTitle className="sr-only">Command Palette</DialogTitle>

   377→          <Search className="h-4 w-4 text-muted-foreground" />
   378→          <input
   379→            autoFocus
   380→            value={q}
   381→            onChange={(e) => setQ(e.target.value)}
   382→            placeholder="Search courses, labs, or jump to..."
   383→            className="flex-1 bg-transparent outline-none text-sm placeholder:text-muted-foreground"
   384→          />
   385→          <kbd className="text-[10px] font-mono text-muted-foreground px-1.5 py-0.5 rounded border border-border">ESC</kbd>
   386→        </div>
   387→        <div className="max-h-[400px] overflow-y-auto p-2">
   388→          {!q.trim() ? (
   389→            <div className="p-2">
   390→              <div className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wider px-2 py-1.5">Quick Navigation</div>
   391→              {navItems.map((item) => (
   392→                <button
   393→                  key={item.label}
   394→                  onClick={() => go(item.view)}

   396→                >
   397→                  <item.icon className="h-4 w-4 text-emerald-400" />

   405→              {results.courses.length === 0 && results.labs.length === 0 && results.notes.length === 0 && (
   406→                <div className="text-center py-8 text-sm text-muted-foreground">
   407→                  <Search className="h-6 w-6 mx-auto mb-2 opacity-50" />
   408→                  No results for "{q}"
   409→                </div>
   410→              )}
   411→              {results.courses.length > 0 && (
   412→                <div>
   413→                  <div className="text-[10px] font-semibold text-muted-foreground uppercase tracking-wider px-2 py-1.5">Courses</div>
   414→                  {results.courses.map((c) => (
   415→                    <button
   416→                      key={c.id}
   417→                      onClick={() => go({ name: "course", courseId: c.id })}
   418→                      className="w-full flex items-center gap-3 px-2 py-2 rounded-lg hover:bg-accent/50 text-sm text-left transition-colors"
   419→                    >
   420→                      <BookOpen className="h-4 w-4 text-emerald-400 shrink-0" />

   443→              {results.notes.length > 0 && (

   446→                  {results.notes.map((n) => (

   484→function timeAgo(date: string): string {
   485→  const diff = Date.now() - new Date(date).getTime()
   486→  const m = Math.floor(diff / 60000)
   487→  if (m < 1) return "just now"
   488→  if (m < 60) return `${m}m ago`
   489→  const h = Math.floor(m / 60)
   490→  if (h < 24) return `${h}h ago`
   491→  const d = Math.floor(h / 24)
   492→  if (d < 7) return `${d}d ago`
   493→  return new Date(date).toLocaleDateString()
   494→}
