   108→function formatTimeAgo(iso?: string | null): string {
   109→  if (!iso) return "-"
   110→  const d = new Date(iso)
   111→  const diff = Date.now() - d.getTime()
   112→  const min = Math.floor(diff / 60_000)
   113→  if (min < 1) return "just now"
   114→  if (min < 60) return `${min}m ago`
   115→  const hr = Math.floor(min / 60)
   116→  if (hr < 24) return `${hr}h ago`
   117→  const day = Math.floor(hr / 24)
   118→  if (day < 30) return `${day}d ago`
   119→  return d.toLocaleDateString()
   120→}
   121→
   122→function detectKind(value: any): "string" | "stringArray" | "objectArray" | "object" | "primitive" {
   123→  if (typeof value === "string") return "string"
   124→  if (Array.isArray(value)) {
   125→    if (value.length === 0) return "stringArray"
   126→    if (value.every((v) => typeof v === "string")) return "stringArray"
   127→    if (value.every((v) => typeof v === "object" && v !== null)) return "objectArray"
   128→    if (value.every((v) => typeof v === "number" || typeof v === "boolean")) return "objectArray"
   129→    return "objectArray"
   130→  }
   131→  if (typeof value === "object" && value !== null) return "object"
   132→  if (typeof value === "number" || typeof value === "boolean") return "primitive"
   133→  return "string"
   134→}

   339→interface SectionData {
   340→  [key: string]: any
   341→}
   342→interface PageDataShape {
   343→  page: string
   344→  sections: Record<string, SectionData>
   345→  updatedAt?: string | null
   346→}
   347→
   348→function PageEditor({ page }: { page: PageId }) {
   349→  const qc = useQueryClient()
   350→  const [draft, setDraft] = React.useState<Record<string, SectionData> | null>(null)
   351→  const [openSection, setOpenSection] = React.useState<string | null>(null)
   352→  const [resetDialogOpen, setResetDialogOpen] = React.useState(false)
   353→
   354→  const meta = PAGES.find((p) => p.id === page)!
   355→
   356→  const { data, isLoading, isError, refetch } = useQuery<PageDataShape>({
   357→    queryKey: ["cms-content", page],
   358→    queryFn: async () => {
   359→      try {
   360→        const res = await fetch(`/api/cms/${page}`)
   361→        if (!res.ok) return { page, sections: {}, updatedAt: null }
   362→        return res.json()
   363→      } catch {
   364→        return { page, sections: {}, updatedAt: null }
   365→      }
   366→    },
   367→    staleTime: 30_000,
   368→    retry: 1,
   369→  })
   370→
   371→  // Reset draft whenever data changes (initial load + invalidations)
   372→  React.useEffect(() => {
   373→    if (data?.sections && Object.keys(data.sections).length > 0) {
   374→      setDraft(JSON.parse(JSON.stringify(data.sections)))
   375→    }
   376→  }, [data?.sections])
   377→
   378→  const saveMutation = useMutation({
   379→    mutationFn: async (payload: { items: Array<{ section: string; key: string; value: any }> }) => {
   380→      return api(`/api/cms/${page}`, {
   381→        method: "PUT",
   382→        body: JSON.stringify(payload),
   383→      })
   384→    },
   385→    onSuccess: () => {
   386→      toast.success("Changes saved · content is live")
   387→      qc.invalidateQueries({ queryKey: ["cms-content", page] })
   388→      // Also invalidate other pages (in case shared global content changed)
   389→      if (page === "global") {
   390→        qc.invalidateQueries({ queryKey: ["cms-content"] })
   391→      }
   392→    },
   393→    onError: (e: any) => {
   394→      toast.error(e?.message || "Failed to save")
   395→    },
   396→  })
   397→
   398→  const handleSave = () => {
   399→    if (!draft) return
   400→    const items: Array<{ section: string; key: string; value: any }> = []
   401→    for (const [section, keys] of Object.entries(draft)) {
   402→      for (const [key, value] of Object.entries(keys)) {
   403→        items.push({ section, key, value })
   404→      }
   405→    }
   406→    saveMutation.mutate({ items })
   407→  }
   408→
   409→  const handleReset = () => {
   410→    setResetDialogOpen(false)
   411→    if (data?.sections) {
   412→      setDraft(JSON.parse(JSON.stringify(data.sections)))
   413→      toast.info("Reverted to last saved version")
   414→    }
   415→  }
   416→
   417→  const isDirty = React.useMemo(() => {
   418→    if (!draft || !data?.sections) return false
   419→    return JSON.stringify(draft) !== JSON.stringify(data.sections)
   420→  }, [draft, data])
   421→
   422→  // Compute changes count
   423→  const changesCount = React.useMemo(() => {
   424→    if (!draft || !data?.sections) return 0
   425→    let n = 0
   426→    for (const [section, keys] of Object.entries(draft)) {
   427→      for (const [key, value] of Object.entries(keys)) {
   428→        const orig = data.sections[section]?.[key]
   429→        if (JSON.stringify(orig) !== JSON.stringify(value)) n++
   430→      }
   431→    }
   432→    return n
   433→  }, [draft, data])
   434→
   435→  const sectionList = draft ? Object.keys(draft).sort() : []
   436→
   437→  return (
   438→    <div className="space-y-4">
   439→      {/* === Top bar === */}
   440→      <Card className="bg-card shadow-lg border border-border p-4 lg:p-5">
   441→        <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-3">
   442→          <div className="flex items-start gap-3">
   443→            <div className={cn(
   444→              "inline-flex items-center justify-center h-10 w-10 rounded-xl shrink-0",
   445→              "bg-amber-500/10"
   446→            )}>
   447→              <meta.icon className={cn("h-5 w-5", meta.accent)} />
   448→            </div>
   449→            <div className="min-w-0">
   450→              <div className="flex items-center gap-2 flex-wrap">
   451→                <h2 className="text-lg font-bold tracking-tight">{meta.label}</h2>
   452→                <Badge variant="outline" className="text-[9px] font-mono border-amber-500/30 text-amber-300 bg-amber-500/5">
   453→                  <Hash className="h-2.5 w-2.5 mr-1" />{page}
   454→                </Badge>
   455→                {sectionList.length > 0 && (
   456→                  <Badge variant="outline" className="text-[9px] font-mono">
   457→                    {sectionList.length} sections
   458→                  </Badge>
   459→                )}
   460→              </div>
   461→              <p className="text-xs text-muted-foreground mt-1">
   462→                {meta.description}
   463→              </p>
   464→              <p className="text-[10px] text-muted-foreground/70 font-mono mt-1">
   465→                Last saved: {formatTimeAgo(data?.updatedAt)}
   466→              </p>
   467→            </div>
   468→          </div>
   469→
   470→          <div className="flex items-center gap-2 flex-wrap">
   471→            <Button
   472→              variant="outline"
   473→              size="sm"
   474→              onClick={() => refetch()}
   475→              disabled={isLoading}
   476→              className="h-9"
   477→            >
   478→              <RefreshCw className={cn("h-3.5 w-3.5 mr-1.5", isLoading && "animate-spin")} />
   479→              Refresh
   480→            </Button>
   481→            <Button
   482→              variant="outline"
   483→              size="sm"
   484→              onClick={() => setResetDialogOpen(true)}
   485→              disabled={!isDirty || saveMutation.isPending}
   486→              className="h-9"
   487→            >
   488→              <RotateCcw className="h-3.5 w-3.5 mr-1.5" />
   489→              Reset
   490→            </Button>
   491→            <Button
   492→              size="sm"
   493→              onClick={handleSave}
   494→              disabled={!isDirty || saveMutation.isPending}
   495→              className="bg-amber-500 hover:bg-amber-400 text-amber-950 h-9"
   496→            >
   497→              {saveMutation.isPending ? (
   498→                <><Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" /> Saving...</>
   499→              ) : (
   500→                <><Save className="h-3.5 w-3.5 mr-1.5" /> Save Changes</>
   501→              )}
   502→            </Button>
   503→          </div>
   504→        </div>
   505→
   506→        {/* Dirty indicator */}
   507→        {isDirty && (
   508→          <div className="mt-3 pt-3 border-t border-border flex items-center gap-2 text-xs">
   509→            <span className="inline-flex items-center gap-1.5 px-2 py-0.5 rounded-full bg-amber-500/10 border border-amber-500/30 text-amber-300 font-mono">
   510→              <span className="h-1.5 w-1.5 rounded-full bg-amber-400 pulse-dot" />
   511→              {changesCount} unsaved {changesCount === 1 ? "change" : "changes"}
   512→            </span>
   513→            <span className="text-muted-foreground">
   514→              Click "Save Changes" to publish.
   515→            </span>
   516→          </div>
   517→        )}
   518→      </Card>
   519→
   520→      {/* === Loading state === */}
   521→      {isLoading && <LoadingSkeleton />}
   522→
   523→      {/* === Error state === */}
   524→      {isError && (
   525→        <Card className="bg-card shadow-lg border border-red-500/30 p-6">
   526→          <div className="flex items-center gap-3">
   527→            <AlertCircle className="h-5 w-5 text-red-400" />
   528→            <div>
   529→              <div className="font-semibold text-sm">Failed to load content</div>
   530→              <div className="text-xs text-muted-foreground mt-0.5">
   531→                Check your connection or try refreshing.
   532→              </div>
   533→            </div>
   534→            <Button variant="outline" size="sm" onClick={() => refetch()} className="ml-auto">
   535→              Retry
   536→            </Button>
   537→          </div>
   538→        </Card>
   539→      )}
   540→
   541→      {/* === Sections === */}

   769→
   770→function KeyEditorBody({
   771→  section,
   772→  keyName,
   773→  value,
   774→  kind,
   775→  showRaw,
   776→  onChange,
   777→}: {
   778→  section: string
   779→  keyName: string
   780→  value: any
   781→  kind: ReturnType<typeof detectKind>
   782→  showRaw: boolean
   783→  onChange: (v: any) => void
   784→}) {
   785→  // String
   786→  if (kind === "string") {
   787→    const isLong = (value as string).length > 60 || (value as string).includes("\n")
   788→    return isLong ? (
   789→      <Textarea
   790→        value={value as string}
   791→        onChange={(e) => onChange(e.target.value)}
   792→        rows={3}
   793→        className="resize-y text-xs"
   794→      />
   795→    ) : (
   796→      <Input
   797→        value={value as string}
   798→        onChange={(e) => onChange(e.target.value)}
   799→        className="h-9 text-xs"
   800→      />
   801→    )
   802→  }
   803→
   804→  // Number / boolean primitive
   805→  if (kind === "primitive") {
   806→    return (
   807→      <Input
   808→        value={String(value)}
   809→        onChange={(e) => {
   810→          const v = e.target.value
   811→          if (typeof value === "number") {
   812→            const n = Number(v)
   813→            onChange(Number.isNaN(n) ? 0 : n)
   814→          } else if (typeof value === "boolean") {
   815→            onChange(v === "true" || v === "1")
   816→          } else {
   817→            onChange(v)
   818→          }
   819→        }}
   820→        className="h-9 text-xs font-mono"
   821→      />
   822→    )
   823→  }
   824→
   825→  // String array
   826→  if (kind === "stringArray" && !showRaw) {
   827→    return <StringArrayEditor value={value as string[]} onChange={onChange} />
   828→  }
   829→
   830→  // Object array (with structured editor)
   831→  if (kind === "objectArray" && !showRaw) {
   832→    return <ObjectArrayEditor value={value as any[]} onChange={onChange} />
   833→  }
   834→
   835→  // Object (single)
   836→  if (kind === "object" && !showRaw) {
   837→    return <ObjectEditor value={value as Record<string, any>} onChange={onChange} />
   838→  }
   839→
   840→  // Fallback: raw JSON editor
   841→  return <JsonEditor value={value} onChange={onChange} />
   842→}

  1185→/* ============================================================
  1186→   JSON editor (raw, with validation)
  1187→   ============================================================ */
  1188→
  1189→function JsonEditor({
  1190→  value,
  1191→  onChange,
  1192→  compact = false,
  1193→}: {
  1194→  value: any
  1195→  onChange: (v: any) => void
  1196→  compact?: boolean
  1197→}) {
  1198→  const [text, setText] = React.useState<string>("")
  1199→  const [error, setError] = React.useState<string | null>(null)
  1200→
  1201→  // Sync text whenever value changes externally
  1202→  React.useEffect(() => {
  1203→    try {
  1204→      setText(JSON.stringify(value, null, compact ? 0 : 2))
  1205→      setError(null)
  1206→    } catch {
  1207→      setText(String(value ?? ""))
  1208→    }
  1209→  }, [value, compact])
  1210→
  1211→  const handle = (raw: string) => {
  1212→    setText(raw)
  1213→    if (raw.trim() === "") {
  1214→      setError(null)
  1215→      onChange(null)
  1216→      return
  1217→    }
  1218→    try {
  1219→      const parsed = JSON.parse(raw)
  1220→      setError(null)
  1221→      onChange(parsed)
  1222→    } catch (e: any) {
  1223→      setError(e?.message ?? "Invalid JSON")
  1224→      // Don't propagate the invalid value to parent
  1225→    }
  1226→  }
  1227→
  1228→  const isLarge = text.length > 200
  1229→
  1230→  return (
  1231→    <div className="space-y-1.5">
  1232→      {isLarge ? (
  1233→        <Textarea
  1234→          value={text}
  1235→          onChange={(e) => handle(e.target.value)}
  1236→          rows={Math.min(12, Math.max(4, Math.ceil(text.length / 60)))}
  1237→          className="font-mono text-[11px] resize-y leading-relaxed"
  1238→        />
  1239→      ) : (
  1240→        <Textarea
  1241→          value={text}
  1242→          onChange={(e) => handle(e.target.value)}
  1243→          rows={Math.min(6, Math.max(2, Math.ceil(text.split("\n").length * 1.2)))}
  1244→          className="font-mono text-[11px] resize-y leading-relaxed"
  1245→        />
  1246→      )}
  1247→      {error ? (
  1248→        <div className="flex items-start gap-1.5 text-[10px] text-red-400 font-mono">
  1249→          <AlertCircle className="h-3 w-3 mt-0.5 shrink-0" />
  1250→          <span className="break-all">{error}</span>
  1251→        </div>
  1252→      ) : (
  1253→        <div className="flex items-center gap-1.5 text-[10px] text-emerald-400/80 font-mono">
  1254→          <CheckCircle2 className="h-3 w-3" />
  1255→          <span>Valid JSON</span>
  1256→        </div>
  1257→      )}
  1258→    </div>
  1259→  )
  1260→}
