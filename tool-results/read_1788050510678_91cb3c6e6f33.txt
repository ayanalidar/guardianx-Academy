  1014→interface Review {

  1023→function ReviewsSection({ courseId, isEnrolled }: { courseId: string; isEnrolled: boolean }) {
  1024→  const { user } = useUser()
  1025→  const qc = useQueryClient()
  1026→  const [showForm, setShowForm] = React.useState(false)
  1027→  const [rating, setRating] = React.useState(5)
  1028→  const [hoverRating, setHoverRating] = React.useState(0)
  1029→  const [title, setTitle] = React.useState("")

  1032→  const { data, isLoading } = useQuery<{ reviews: Review[]; avgRating: number; totalReviews: number; distribution: { star: number; count: number }[] }>({
  1033→    queryKey: ["reviews", courseId],

  1035→  })
  1036→
  1037→  const submitMutation = useMutation({
  1038→    mutationFn: () => api(`/api/courses/${courseId}/reviews`, {
  1039→      method: "POST",
  1040→      body: JSON.stringify({ rating, title, content }),
  1041→    }),
  1042→    onSuccess: () => {
  1043→      qc.invalidateQueries({ queryKey: ["reviews", courseId] })
  1044→      setShowForm(false); setTitle(""); setContent(""); setRating(5)

  1047→    onError: (e: any) => toast.error(e.message),
  1048→  })
  1049→
  1050→  const reviews = data?.reviews ?? []
  1051→  const avg = data?.avgRating ?? 0
