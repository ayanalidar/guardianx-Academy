   390→function ContentEditor() {
   391→  const { data: coursesData } = useQuery<{ courses: InstructorCourse[] }>({

   393→    queryFn: () => api("/api/instructor/courses"),
   394→  })
   395→  const courses = coursesData?.courses ?? []
   396→  const [selectedCourseId, setSelectedCourseId] = React.useState("")
   397→  const qc = useQueryClient()
   398→
   399→  React.useEffect(() => {
   400→    if (!selectedCourseId && courses[0]) setSelectedCourseId(courses[0].id)
   401→  }, [courses, selectedCourseId])

   403→  if (courses.length === 0) return null

   405→  return (

   435→  const [editingLesson, setEditingLesson] = React.useState<string | null>(null)
   436→  const [newLessonModuleId, setNewLessonModuleId] = React.useState<string | null>(null)

   438→  const [showNewModule, setShowNewModule] = React.useState(false)
   439→
   440→  const deleteLesson = useMutation({
   441→    mutationFn: (lessonId: string) => api(`/api/instructor/lessons/${lessonId}`, { method: "DELETE" }),
   442→    onSuccess: () => {
   443→      qc.invalidateQueries({ queryKey: ["course", courseId] })
   444→      qc.invalidateQueries({ queryKey: ["instructor", "courses"] })
   445→      toast.success("Lesson deleted")
   446→    },
   447→    onError: (e: any) => toast.error(e.message),
   448→  })
   449→
   450→  const deleteModule = useMutation({
   451→    mutationFn: (moduleId: string) => api(`/api/instructor/modules/${moduleId}`, { method: "DELETE" }),
   452→    onSuccess: () => {
   453→      qc.invalidateQueries({ queryKey: ["course", courseId] })
   454→      qc.invalidateQueries({ queryKey: ["instructor", "courses"] })
   455→      toast.success("Module deleted")
   456→    },
   457→    onError: (e: any) => toast.error(e.message),
   458→  })
   459→
   460→  if (!course) return <Skeleton className="h-64" />
   461→
   462→  return (
   463→    <div className="space-y-4">
   464→      {course.modules.map((m: any, mi: number) => {
   465→        const col = colorFor(course.color)
   466→        return (
   467→          <Card key={m.id} className="overflow-hidden">
   468→            <div className="p-4 border-b border-border bg-muted/20">
   469→              <div className="flex items-center gap-3">
   470→                <div className={cn("flex h-8 w-8 shrink-0 items-center justify-center rounded-lg font-mono text-sm font-bold", col.bg, col.border, "border", col.text)}>
   471→                  {String(mi + 1).padStart(2, "0")}
   472→                </div>
   473→                <div className="flex-1 min-w-0">
   474→                  <div className="font-medium text-sm truncate">{m.title}</div>
   475→                  <div className="text-xs text-muted-foreground">{m.lessons.length} lesson{m.lessons.length !== 1 ? "s" : ""}</div>
   476→                </div>
   477→                <Button size="sm" variant="outline" onClick={() => setEditingModuleId(editingModuleId === m.id ? null : m.id)}>
   478→                  <Pencil className="h-3.5 w-3.5 mr-1" /> Edit
   479→                </Button>
   480→                <Button size="sm" variant="outline" onClick={() => setNewLessonModuleId(newLessonModuleId === m.id ? null : m.id)}>
   481→                  <Plus className="h-3.5 w-3.5 mr-1" /> Lesson
   482→                </Button>
   483→                <button
   484→                  onClick={() => { if (confirm(`Delete module "${m.title}" and all its lessons?`)) deleteModule.mutate(m.id) }}
   485→                  className="p-1.5 rounded-lg text-muted-foreground hover:text-red-400 hover:bg-red-500/10 transition-colors"
   486→                  title="Delete module"
   487→                >
   488→                  <Trash2 className="h-3.5 w-3.5" />
   489→                </button>
   490→              </div>
   491→            </div>
   492→            <div className="divide-y divide-border">
   493→              {m.lessons.map((l: any) => (
   494→                <div key={l.id} className="p-3 flex items-center gap-3 hover:bg-accent/20 transition-colors">
   495→                  <FileText className="h-4 w-4 text-muted-foreground shrink-0" />
   496→                  <div className="flex-1 min-w-0">
   497→                    <div className="text-sm font-medium truncate">{l.title}</div>
   498→                    <div className="text-[10px] text-muted-foreground">{l.type} · {l.durationMin}m</div>

   501→                    <Pencil className="h-3 w-3 mr-1" /> Edit
   502→                  </Button>
   503→                  <button
   504→                    onClick={() => { if (confirm(`Delete "${l.title}"?`)) deleteLesson.mutate(l.id) }}
   505→                    className="p-1.5 rounded-lg text-muted-foreground hover:text-red-400 hover:bg-red-500/10 transition-colors"
   506→                    title="Delete lesson"
   507→                  >
   508→                    <Trash2 className="h-3.5 w-3.5" />
   509→                  </button>
   510→                </div>
   511→              ))}
   512→              {editingLesson && m.lessons.some((l: any) => l.id === editingLesson) && (
   513→                <LessonEditor
   514→                  lessonId={editingLesson}
   515→                  courseId={courseId}
   516→                  onClose={() => setEditingLesson(null)}
   517→                />
   518→              )}
   519→              {newLessonModuleId === m.id && (
   520→                <NewLessonForm moduleId={m.id} courseId={courseId} onClose={() => setNewLessonModuleId(null)} />
   521→              )}
   522→              {editingModuleId === m.id && (
   523→                <ModuleEditor moduleId={m.id} courseId={courseId} currentTitle={m.title} currentDescription={m.description} onClose={() => setEditingModuleId(null)} />
   524→              )}
   525→            </div>
   526→          </Card>
   527→        )
   528→      })}
   529→      {/* New Module button + form */}
   530→      {showNewModule ? (
   531→        <NewModuleForm courseId={courseId} onClose={() => setShowNewModule(false)} />
   532→      ) : (
   533→        <Button variant="outline" className="w-full border-dashed" onClick={() => setShowNewModule(true)}>
   534→          <Plus className="h-4 w-4 mr-1.5" /> Add Module
   535→        </Button>
   536→      )}
   537→    </div>
   538→  )
   539→}

   615→function LessonEditor({ lessonId, courseId, onClose }: { lessonId: string; courseId: string; onClose: () => void }) {

   617→  const { data } = useQuery<any>({
   618→    queryKey: ["lesson", lessonId],
   619→    queryFn: () => api(`/api/lessons/${lessonId}`),
   620→  })

   623→  const [content, setContent] = React.useState("")

   628→      setTitle(lesson.title)
   629→      setContent(lesson.content)
   630→      setDurationMin(lesson.durationMin)
   631→    }

   634→  const save = useMutation({
   635→    mutationFn: () => api(`/api/instructor/lessons/${lessonId}`, {

   640→      qc.invalidateQueries({ queryKey: ["course", courseId] })
   641→      qc.invalidateQueries({ queryKey: ["lesson", lessonId] })
   642→      qc.invalidateQueries({ queryKey: ["instructor", "courses"] })
   643→      toast.success("Lesson saved")
   644→      onClose()
   645→    },
   646→    onError: (e: any) => toast.error(e.message),
   647→  })

   660→        value={content}
   661→        onChange={(e) => setContent(e.target.value)}
   662→        placeholder="Lesson content (Markdown supported)..."
   663→        className="min-h-[180px] font-mono text-xs"
   664→      />
   665→      <div className="flex items-center gap-3">
   666→        <label className="text-xs text-muted-foreground">Duration (min):</label>
