   144→function relativeTime(iso: string): string {
   145→  const d = new Date(iso)
   146→  const diff = Date.now() - d.getTime()
   147→  const min = Math.floor(diff / 60000)
   148→  if (min < 1) return "just now"
   149→  if (min < 60) return `${min}m ago`
   150→  const hr = Math.floor(min / 60)
   151→  if (hr < 24) return `${hr}h ago`
   152→  const day = Math.floor(hr / 24)
   153→  if (day < 7) return `${day}d ago`
   154→  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" })
   155→}

   157→function formatClock(d: Date): string {
   158→  return d.toLocaleTimeString(undefined, {
   159→    hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false,
   160→  })
   161→}

   163→function deriveServices(lab: LabListItem): string[] {
   164→  const t = (lab.tags || "").toLowerCase()
   165→  const out: string[] = []
   166→  if (t.includes("ssh") || lab.commands?.includes("ssh")) out.push("SSH")
   167→  if (t.includes("http") || t.includes("web")) out.push("HTTP")
   168→  if (t.includes("ftp") || t.includes("smb")) out.push("FTP")
   169→  if (t.includes("dns") || t.includes("smtp")) out.push("DNS")
   170→  if (t.includes("mysql") || t.includes("sql")) out.push("MySQL")
   171→  if (out.length === 0) out.push("SSH", "HTTP")
   172→  return out.slice(0, 3)
   173→}

   175→function pseudoIp(seed: string): string {
   176→  let h = 0
   177→  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) | 0
   178→  const a = (Math.abs(h) % 200) + 10
   179→  const b = (Math.abs(h >> 8) % 200) + 10
   180→  return `10.10.${a % 250}.${b % 250}`
   181→}
