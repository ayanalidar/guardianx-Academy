    25→/* ============================================================
    26→   AdminCoursesView — simple form-based course management page.
    27→
    28→   - Table of all courses: title, category, level, price, status,
    29→     instructor, plus secondary info (rating, students, lessons).
    30→   - Create Course button → dialog with simple form (title,
    31→     shortName, description, category, level, durationHours,
    32→     price, instructorId dropdown, published checkbox).
    33→   - Edit button → same form, pre-filled.
    34→   - Delete button → confirm dialog (cascade delete).
    35→
    36→   Fetches:   /api/admin/courses
    37→   Mutates:   POST /api/admin/courses
    38→              PATCH /api/admin/courses/[id]
    39→              DELETE /api/admin/courses/[id]
    40→   Instructors dropdown is populated from /api/admin/instructors.
    41→   ============================================================ */
    42→
    43→/* ------------------------------ types ------------------------------ */
    44→interface AdminCourse {
    45→  id: string
    46→  slug: string
    47→  title: string
    48→  shortName: string
    49→  description: string
    50→  longDescription: string
    51→  category: string
    52→  level: string
    53→  durationHours: number
    54→  price: number
    55→  rating: number
    56→  studentsCount: number
    57→  thumbnail: string | null
    58→  color: string
    59→  tags: string
    60→  certBody: string | null
    61→  published: boolean
    62→  createdAt: string
    63→  updatedAt: string
    64→  instructor: { id: string; name: string; title: string | null; avatar: string | null }
    65→  moduleCount: number
    66→  lessonCount: number
    67→  labCount: number
    68→  enrollmentCount: number
    69→}
    70→
    71→interface InstructorOption {
    72→  id: string
    73→  name: string
    74→  email: string
    75→  title: string | null
    76→}
    77→
    78→/* ------------------------------ form ------------------------------ */
    79→interface CourseForm {
    80→  title: string
    81→  shortName: string
    82→  description: string
    83→  category: string
    84→  level: string
    85→  durationHours: string
    86→  price: string
    87→  instructorId: string
    88→  published: boolean
    89→}
    90→
    91→const CATEGORIES = [
    92→  "Ethical Hacking", "Networking", "Web Security", "System Administration",
    93→  "Security Management", "Identity & Access", "Cloud Security",
    94→  "DevSecOps", "Incident Response", "General",
    95→]
    96→
    97→const LEVELS = ["Beginner", "Intermediate", "Advanced"]
    98→
    99→function emptyForm(): CourseForm {
   100→  return {
   101→    title: "",
   102→    shortName: "",
   103→    description: "",
   104→    category: CATEGORIES[0]!,
   105→    level: LEVELS[0]!,
   106→    durationHours: "40",
   107→    price: "0",
   108→    instructorId: "",
   109→    published: true,
   110→  }
   111→}
   112→
   113→function formFromCourse(c: AdminCourse): CourseForm {
   114→  return {
   115→    title: c.title,
   116→    shortName: c.shortName,
   117→    description: c.description,
   118→    category: c.category,
   119→    level: c.level,
   120→    durationHours: String(c.durationHours),
   121→    price: String(c.price),
   122→    instructorId: c.instructor?.id ?? "",
   123→    published: c.published,
   124→  }
   125→}
   126→
   127→function validateForm(form: CourseForm, instructors: InstructorOption[]): string | null {
   128→  if (!form.title.trim()) return "Title is required"
   129→  if (!form.shortName.trim()) return "Short name (e.g. CEH) is required"
   130→  if (!form.instructorId) return "Instructor is required"
   131→  if (!instructors.some((i) => i.id === form.instructorId)) {
   132→    return "Please select a valid instructor from the list"
   133→  }
   134→  const dur = Number(form.durationHours)
   135→  if (!Number.isFinite(dur) || dur < 1) return "Duration must be a positive number"
   136→  const price = Number(form.price)
   137→  if (!Number.isFinite(price) || price < 0) return "Price must be a non-negative number"
   138→  return null
   139→}
   140→
   141→const LEVEL_BADGE: Record<string, string> = {
   142→  Beginner: "border-emerald-500/40 text-emerald-300 bg-emerald-500/10",
   143→  Intermediate: "border-cyan-500/40 text-cyan-300 bg-cyan-500/10",
   144→  Advanced: "border-violet-500/40 text-violet-300 bg-violet-500/10",
   145→}

   147→/* ------------------------------ view ------------------------------ */
   148→export function AdminCoursesView() {
   149→  const { navigate } = useAppStore()
   150→  const queryClient = useQueryClient()
   151→
   152→  const [search, setSearch] = React.useState("")
   153→  const [createOpen, setCreateOpen] = React.useState(false)
   154→  const [editOpen, setEditOpen] = React.useState(false)
   155→  const [editingCourse, setEditingCourse] = React.useState<AdminCourse | null>(null)
   156→  const [deleteOpen, setDeleteOpen] = React.useState(false)
   157→  const [deletingCourse, setDeletingCourse] = React.useState<AdminCourse | null>(null)
   158→  const [form, setForm] = React.useState<CourseForm>(emptyForm())
   159→  const [submitting, setSubmitting] = React.useState(false)
   160→  const [formError, setFormError] = React.useState<string | null>(null)
   161→
   162→  /* ------------------- DB queries ------------------- */
   163→  const { data, isLoading, isFetching, error } = useQuery<{ courses: AdminCourse[]; total: number }>({
   164→    queryKey: ["admin-courses"],
   165→    staleTime: 30_000,
   166→    gcTime: 5 * 60_000,
   167→    refetchOnWindowFocus: false,
   168→    queryFn: async () => {
   169→      const res = await fetch("/api/admin/courses")
   170→      if (!res.ok) {
   171→        if (res.status === 401) throw new Error("Unauthorized — please sign in as an admin")
   172→        if (res.status === 403) throw new Error("Forbidden — admin role required")
   173→        const j = await res.json().catch(() => ({}))
   174→        throw new Error(j.error || "Failed to load courses")
   175→      }
   176→      return res.json()
   177→    },
   178→  })
   179→
   180→  const { data: instructorData } = useQuery<{ instructors: InstructorOption[] }>({
   181→    queryKey: ["admin-instructors-mini"],
   182→    staleTime: 5 * 60_000,
   183→    queryFn: async () => {
   184→      const res = await fetch("/api/admin/instructors")
   185→      if (!res.ok) return { instructors: [] }
   186→      const j = await res.json()
   187→      const list: InstructorOption[] = (j.instructors ?? []).map((i: any) => ({
   188→        id: i.id,
   189→        name: i.name,
   190→        email: i.email,
   191→        title: i.title ?? null,
   192→      }))
   193→      return { instructors: list }
   194→    },
   195→  })
   196→
   197→  const courses = data?.courses ?? []
   198→  const instructors = instructorData?.instructors ?? []
   199→
   200→  // Client-side filter (admin course count is small enough)
   201→  const filtered = React.useMemo(() => {
   202→    const q = search.trim().toLowerCase()
   203→    if (!q) return courses
   204→    return courses.filter(
   205→      (c) =>
   206→        c.title.toLowerCase().includes(q) ||
   207→        c.shortName.toLowerCase().includes(q) ||
   208→        c.category.toLowerCase().includes(q),
   209→    )
   210→  }, [courses, search])
   211→
   212→  function refetch() {
   213→    queryClient.invalidateQueries({ queryKey: ["admin-courses"] })
   214→  }
   215→
   216→  function openCreate() {
   217→    setForm(emptyForm())
   218→    setFormError(null)
   219→    setCreateOpen(true)
   220→  }
   221→
   222→  function openEdit(c: AdminCourse) {
   223→    setEditingCourse(c)
   224→    setForm(formFromCourse(c))
   225→    setFormError(null)
   226→    setEditOpen(true)
   227→  }
   228→
   229→  function openDelete(c: AdminCourse) {
   230→    setDeletingCourse(c)
   231→    setDeleteOpen(true)
   232→  }
   233→
   234→  /* ------------------- mutations ------------------- */
   235→  async function handleSubmitCreate() {
   236→    const err = validateForm(form, instructors)
   237→    if (err) { setFormError(err); return }
   238→    setSubmitting(true)
   239→    setFormError(null)
   240→    try {
   241→      const res = await fetch("/api/admin/courses", {
   242→        method: "POST",
   243→        headers: { "Content-Type": "application/json" },
   244→        body: JSON.stringify({
   245→          title: form.title,
   246→          shortName: form.shortName,
   247→          description: form.description,
   248→          category: form.category,
   249→          level: form.level,
   250→          durationHours: Number(form.durationHours),
   251→          price: Number(form.price),
   252→          instructorId: form.instructorId,
   253→        }),
   254→      })
   255→      const j = await res.json()
   256→      if (!res.ok) throw new Error(j.error || "Failed to create course")
   257→      // The create endpoint hardcodes published=true; toggle if user unchecked it
   258→      if (!form.published && j.course?.id) {
   259→        await fetch(`/api/admin/courses/${j.course.id}`, {
   260→          method: "PATCH",
   261→          headers: { "Content-Type": "application/json" },
   262→          body: JSON.stringify({ published: false }),
   263→        })
   264→      }
   265→      toast.success(`Course "${form.shortName.toUpperCase()}" created`)
   266→      setCreateOpen(false)
   267→      refetch()
   268→    } catch (e: any) {
   269→      setFormError(e.message || "Failed to create course")
   270→    } finally {
   271→      setSubmitting(false)
   272→    }
   273→  }
   274→
   275→  async function handleSubmitEdit() {
   276→    if (!editingCourse) return
   277→    const err = validateForm(form, instructors)
   278→    if (err) { setFormError(err); return }
   279→    setSubmitting(true)
   280→    setFormError(null)
   281→    try {
   282→      const res = await fetch(`/api/admin/courses/${editingCourse.id}`, {
   283→        method: "PATCH",
   284→        headers: { "Content-Type": "application/json" },
   285→        body: JSON.stringify({
   286→          title: form.title,
   287→          shortName: form.shortName,
   288→          description: form.description,
   289→          category: form.category,
   290→          level: form.level,
   291→          durationHours: Number(form.durationHours),
   292→          price: Number(form.price),
   293→          instructorId: form.instructorId,
   294→          published: form.published,
   295→        }),
   296→      })
   297→      const j = await res.json()
   298→      if (!res.ok) throw new Error(j.error || "Failed to update course")
   299→      toast.success(`Course "${form.shortName.toUpperCase()}" updated`)
   300→      setEditOpen(false)
   301→      refetch()
   302→    } catch (e: any) {
   303→      setFormError(e.message || "Failed to update course")
   304→    } finally {
   305→      setSubmitting(false)
   306→    }
   307→  }
   308→
   309→  async function handleConfirmDelete() {
   310→    if (!deletingCourse) return
   311→    setSubmitting(true)
   312→    try {
   313→      const res = await fetch(`/api/admin/courses/${deletingCourse.id}`, { method: "DELETE" })
   314→      const j = await res.json()
   315→      if (!res.ok) throw new Error(j.error || "Failed to delete course")
   316→      toast.success(`Course "${deletingCourse.shortName}" deleted`)
   317→      setDeleteOpen(false)
   318→      refetch()
   319→    } catch (e: any) {
   320→      toast.error(e.message || "Failed to delete course")
   321→    } finally {
   322→      setSubmitting(false)
   323→    }
   324→  }
   325→
   326→  /* ------------------- render ------------------- */
   327→  return (
   328→    <div className="min-h-screen bg-background">
   329→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-6 lg:py-8 space-y-6">
   330→        {/* Header */}
   331→        <FadeInRow>
   332→          <div className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-4">
   333→            <div>
   334→              <button
   335→                onClick={() => navigate({ name: "admin" })}
   336→                className="inline-flex items-center gap-1.5 text-xs font-mono text-muted-foreground hover:text-amber-300 transition-colors mb-3"
   337→              >
   338→                <ArrowLeft className="h-3.5 w-3.5" /> BACK TO ADMIN
   339→              </button>
   340→              <h1 className="text-2xl lg:text-3xl font-bold tracking-tight flex items-center gap-3">
   341→                <span className="h-9 w-9 rounded-xl bg-amber-500/15 border border-amber-500/30 flex items-center justify-center">
   342→                  <BookOpen className="h-4.5 w-4.5 text-amber-300" />
   343→                </span>
   344→                Courses
   345→              </h1>
   346→              <p className="text-sm text-muted-foreground mt-1.5 max-w-2xl">
   347→                Create, edit, and publish courses. Assign instructors, set pricing, and manage the catalog from one place.
   348→              </p>
   349→            </div>
   350→            <Button
   351→              onClick={openCreate}
   352→              className="bg-amber-600 hover:bg-amber-500 text-amber-50 btn-premium h-10 px-5"
   353→            >
   354→              <Plus className="h-4 w-4 mr-2" /> Create Course
   355→            </Button>
   356→          </div>
   357→        </FadeInRow>
   358→
   359→        {/* Stats strip */}
   360→        <FadeInRow>
   361→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   362→            <StatTile
   363→              label="Total Courses"
   364→              value={courses.length}
   365→              icon={BookOpen}
   366→              tint="bg-amber-500/10 text-amber-300"
   367→            />
   368→            <StatTile
   369→              label="Published"
   370→              value={courses.filter((c) => c.published).length}
   371→              icon={CheckCircle2}
   372→              tint="bg-emerald-500/10 text-emerald-300"
   373→            />
   374→            <StatTile
   375→              label="Drafts"
   376→              value={courses.filter((c) => !c.published).length}
   377→              icon={Layers}
   378→              tint="bg-cyan-500/10 text-cyan-300"
   379→            />
   380→            <StatTile
   381→              label="Total Enrollments"
   382→              value={courses.reduce((acc, c) => acc + c.enrollmentCount, 0)}
   383→              icon={Users}
   384→              tint="bg-violet-500/10 text-violet-300"
   385→            />
   386→          </div>
   387→        </FadeInRow>
   388→
   389→        {/* Search bar */}
   390→        <FadeInRow>
   391→          <div className="relative max-w-md">
   392→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   393→            <Input
   394→              placeholder="Search by title, short name, or category..."
   395→              value={search}
   396→              onChange={(e) => setSearch(e.target.value)}
   397→              className="pl-9 bg-card/40 border-border/60"
   398→            />
   399→          </div>
   400→        </FadeInRow>
   401→
   402→        {/* Table card */}
   403→        <FadeInRow>
   404→          <Card className="bg-card/40 backdrop-blur-xl border-border/60 overflow-hidden p-0">
   405→            {/* Header row */}
   406→            <div className="flex items-center justify-between px-5 py-3.5 border-b border-border/60 bg-muted/20">
   407→              <h2 className="text-sm font-semibold">All Courses ({filtered.length})</h2>
   408→              {isFetching && !isLoading && (
   409→                <span className="text-[10px] font-mono text-muted-foreground/70 flex items-center gap-1">
   410→                  <Loader2 className="h-3 w-3 animate-spin" /> SYNCING
   411→                </span>
   412→              )}
   413→            </div>
   414→
   415→            {/* Column headers */}
   416→            <div className="hidden lg:grid grid-cols-12 gap-3 px-5 py-2.5 border-b border-border/40 text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   417→              <div className="col-span-4">Course</div>
   418→              <div className="col-span-2">Category</div>
   419→              <div className="col-span-1">Level</div>
   420→              <div className="col-span-1">Price</div>
   421→              <div className="col-span-2">Instructor</div>
   422→              <div className="col-span-1">Status</div>
   423→              <div className="col-span-1 text-right">Actions</div>
   424→            </div>
   425→
   426→            {isLoading ? (
   427→              <div className="p-5 space-y-3">
   428→                {Array.from({ length: 5 }).map((_, i) => (
   429→                  <Skeleton key={i} className="h-14 rounded-lg" />
   430→                ))}
   431→              </div>
   432→            ) : error ? (
   433→              <div className="p-12 text-center">
   434→                <AlertTriangle className="h-10 w-10 text-rose-400 mx-auto mb-3" />
   435→                <p className="font-medium mb-1">Failed to load courses</p>
   436→                <p className="text-sm text-muted-foreground">{(error as Error).message}</p>
   437→              </div>
   438→            ) : filtered.length === 0 ? (
   439→              <div className="p-12 text-center">
   440→                <BookOpen className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   441→                <p className="font-medium mb-1">{search ? "No courses match your search" : "No courses yet"}</p>
   442→                <p className="text-sm text-muted-foreground mb-4">
   443→                  {search
   444→                    ? "Try a different search term."
   445→                    : "Create your first course to populate the catalog."}
   446→                </p>
   447→                {!search && (
   448→                  <Button onClick={openCreate} className="bg-amber-600 hover:bg-amber-500 btn-premium h-9">
   449→                    <Plus className="h-4 w-4 mr-1.5" /> Create Course
   450→                  </Button>
   451→                )}
   452→              </div>
   453→            ) : (
   454→              <div className="divide-y divide-border/40 max-h-[640px] overflow-y-auto">
   455→                {filtered.map((c) => (
   456→                  <div
   457→                    key={c.id}
   458→                    className="grid grid-cols-1 lg:grid-cols-12 gap-3 px-5 py-3.5 hover:bg-muted/15 transition-colors items-center"
   459→                  >
   460→                    {/* Course */}
   461→                    <div className="lg:col-span-4 flex items-center gap-3 min-w-0">
   462→                      <div className="h-9 w-9 shrink-0 rounded-lg bg-violet-500/10 border border-violet-500/20 flex items-center justify-center">
   463→                        <BookOpen className="h-4 w-4 text-violet-300" />
   464→                      </div>
   465→                      <div className="min-w-0">
   466→                        <div className="font-medium text-sm truncate">{c.title}</div>
   467→                        <div className="flex items-center gap-2 text-[10px] text-muted-foreground">
   468→                          <span className="font-mono">{c.shortName}</span>
   469→                          <span>·</span>
   470→                          <span className="flex items-center gap-0.5">
   471→                            <Star className="h-2.5 w-2.5 text-amber-300" />
   472→                            {c.rating.toFixed(1)}
   473→                          </span>
   474→                          <span>·</span>
   475→                          <span className="flex items-center gap-0.5">
   476→                            <Users className="h-2.5 w-2.5" />
   477→                            {c.enrollmentCount}
   478→                          </span>
   479→                          <span>·</span>
   480→                          <span className="flex items-center gap-0.5">
   481→                            <Layers className="h-2.5 w-2.5" />
   482→                            {c.moduleCount}
   483→                          </span>
   484→                          <span>·</span>
   485→                          <span className="flex items-center gap-0.5">
   486→                            <Clock className="h-2.5 w-2.5" />
   487→                            {c.durationHours}h
   488→                          </span>
   489→                        </div>
   490→                      </div>
   491→                    </div>
   492→
   493→                    {/* Category */}
   494→                    <div className="lg:col-span-2 text-sm text-muted-foreground truncate">
   495→                      <span className="lg:hidden text-[10px] font-mono uppercase tracking-wider text-muted-foreground/70 mr-1">Category:</span>
   496→                      {c.category}
   497→                    </div>
   498→
   499→                    {/* Level */}
   500→                    <div className="lg:col-span-1">
   501→                      <Badge variant="outline" className={cn("text-[9px] uppercase", LEVEL_BADGE[c.level] ?? LEVEL_BADGE.Beginner)}>
   502→                        {c.level}
   503→                      </Badge>
   504→                    </div>
   505→
   506→                    {/* Price */}
   507→                    <div className="lg:col-span-1 text-sm font-medium tabular-nums flex items-center">
   508→                      <span className="lg:hidden text-[10px] font-mono uppercase tracking-wider text-muted-foreground/70 mr-1">Price:</span>
   509→                      {c.price === 0 ? (
   510→                        <span className="text-emerald-300">Free</span>
   511→                      ) : (
   512→                        <span className="flex items-center">
   513→                          <IndianRupee className="h-3 w-3" />
   514→                          {c.price.toLocaleString("en-IN")}
   515→                        </span>
   516→                      )}
   517→                    </div>
   518→
   519→                    {/* Instructor */}
   520→                    <div className="lg:col-span-2 text-sm truncate">
   521→                      <span className="lg:hidden text-[10px] font-mono uppercase tracking-wider text-muted-foreground/70 mr-1">Instructor:</span>
   522→                      {c.instructor?.name ?? "—"}
   523→                    </div>
   524→
   525→                    {/* Status */}
   526→                    <div className="lg:col-span-1">
   527→                      <Badge
   528→                        variant="outline"
   529→                        className={cn(
   530→                          "text-[9px] uppercase",
   531→                          c.published
   532→                            ? "border-emerald-500/40 text-emerald-300 bg-emerald-500/10"
   533→                            : "border-zinc-500/40 text-zinc-300 bg-zinc-500/10",
   534→                        )}
   535→                      >
   536→                        {c.published ? "Published" : "Draft"}
   537→                      </Badge>
   538→                    </div>
   539→
   540→                    {/* Actions */}
   541→                    <div className="lg:col-span-1 flex items-center justify-end gap-1">
   542→                      <Button
   543→                        size="sm"
   544→                        variant="ghost"
   545→                        onClick={() => openEdit(c)}
   546→                        className="h-7 w-7 p-0 text-muted-foreground hover:text-amber-300 hover:bg-amber-500/10"
   547→                        title="Edit"
   548→                      >
   549→                        <Pencil className="h-3.5 w-3.5" />
   550→                      </Button>
   551→                      <Button
   552→                        size="sm"
   553→                        variant="ghost"
   554→                        onClick={() => openDelete(c)}
   555→                        className="h-7 w-7 p-0 text-muted-foreground hover:text-rose-300 hover:bg-rose-500/10"
   556→                        title="Delete"
   557→                      >
   558→                        <Trash2 className="h-3.5 w-3.5" />
   559→                      </Button>
   560→                    </div>
   561→                  </div>
   562→                ))}
   563→              </div>
   564→            )}
   565→          </Card>
   566→        </FadeInRow>
   567→      </div>
   568→
   569→      {/* Create / Edit dialog */}
   570→      <CourseFormDialog
   571→        open={createOpen}
   572→        onOpenChange={setCreateOpen}
   573→        title="Create Course"
   574→        description="Add a new course to the catalog. The slug is auto-generated from the title."
   575→        form={form}
   576→        setForm={setForm}
   577→        instructors={instructors}
   578→        submitting={submitting}
   579→        error={formError}
   580→        onSubmit={handleSubmitCreate}
   581→        submitLabel="Create Course"
   582→      />
   583→      <CourseFormDialog
   584→        open={editOpen}
   585→        onOpenChange={setEditOpen}
   586→        title={`Edit Course ${editingCourse ? `"${editingCourse.shortName}"` : ""}`}
   587→        description="Update course details. Changes are visible immediately in the catalog."
   588→        form={form}
   589→        setForm={setForm}
   590→        instructors={instructors}
   591→        submitting={submitting}
   592→        error={formError}
   593→        onSubmit={handleSubmitEdit}
   594→        submitLabel="Save Changes"
   595→      />
   596→
   597→      {/* Delete alert */}
   598→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   599→        <AlertDialogContent className="bg-card border-border/60">
   600→          <AlertDialogHeader>
   601→            <AlertDialogTitle className="flex items-center gap-2">
   602→              <Trash2 className="h-4 w-4 text-rose-400" /> Delete course?
   603→            </AlertDialogTitle>
   604→            <AlertDialogDescription>
   605→              This will permanently delete <span className="font-mono font-bold text-foreground">{deletingCourse?.shortName}</span>
   606→              {" "}— <span className="text-foreground">{deletingCourse?.title}</span>. All modules, lessons,
   607→              enrollments, and certificates linked to this course will be cascade-deleted. This cannot be undone.
   608→            </AlertDialogDescription>
   609→          </AlertDialogHeader>
   610→          <AlertDialogFooter>
   611→            <AlertDialogCancel className="border-border/60" disabled={submitting}>Cancel</AlertDialogCancel>
   612→            <AlertDialogAction
   613→              onClick={handleConfirmDelete}
   614→              disabled={submitting}
   615→              className="bg-rose-600 hover:bg-rose-500 text-rose-50"
   616→            >
   617→              {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : "Delete"}
   618→            </AlertDialogAction>
   619→          </AlertDialogFooter>
   620→        </AlertDialogContent>
   621→      </AlertDialog>
   622→    </div>
   623→  )

   643→function FadeInRow({ children }: { children: React.ReactNode }) {
   644→  const [shown, setShown] = React.useState(false)
   645→  React.useEffect(() => {
   646→    const t = setTimeout(() => setShown(true), 50)
   647→    return () => clearTimeout(t)
   648→  }, [])
   649→  return (
   650→    <div
   651→      style={{
   652→        opacity: shown ? 1 : 0,
   653→        transform: shown ? "translateY(0)" : "translateY(6px)",
   654→        transition: "opacity 300ms ease-out, transform 300ms ease-out",
   655→      }}
   656→    >
   657→      {children}
   658→    </div>
   659→  )
   660→}
