   129→function BatchCarousel({ onEnroll }: { onEnroll: (batch: CourseBatch) => void }) {
   130→  const [currentIndex, setCurrentIndex] = React.useState(0)
   131→  const [isPaused, setIsPaused] = React.useState(false)
   132→
   133→  // Auto-rotate every 5s; pause on hover/focus
   134→  React.useEffect(() => {
   135→    if (isPaused) return
   136→    const timer = setInterval(() => {
   137→      setCurrentIndex((prev) => (prev + 1) % UPCOMING_BATCHES.length)
   138→    }, 5000)
   139→    return () => clearInterval(timer)
   140→  }, [isPaused])
   141→
   142→  const goToSlide = (index: number) => {
   143→    setCurrentIndex((index + UPCOMING_BATCHES.length) % UPCOMING_BATCHES.length)
   144→  }
   145→  const goToPrev = () => goToSlide(currentIndex - 1)
   146→  const goToNext = () => goToSlide(currentIndex + 1)
   147→
   148→  return (
   149→    <div
   150→      className="relative"
   151→      onMouseEnter={() => setIsPaused(true)}
   152→      onMouseLeave={() => setIsPaused(false)}
   153→      onFocus={() => setIsPaused(true)}
   154→      onBlur={() => setIsPaused(false)}
   155→    >
   156→      {/* Carousel viewport */}
   157→      <div className="relative overflow-hidden rounded-2xl">
   158→        <div
   159→          className="flex transition-transform duration-700 ease-[cubic-bezier(0.16,1,0.3,1)]"
   160→          style={{ transform: `translateX(-${currentIndex * 100}%)` }}
   161→        >
   162→          {UPCOMING_BATCHES.map((batch, i) => {
   163→            const modeStyle = BATCH_MODE_STYLES[batch.mode]
   164→            const seatsLeft = batch.seatsTotal - batch.seatsFilled
   165→            const fillPct = Math.round((batch.seatsFilled / batch.seatsTotal) * 100)
   166→            const ModeIcon = modeStyle.icon
   167→            return (
   168→              <div
   169→                key={batch.name}
   170→                className="w-full shrink-0 px-0.5"
   171→                aria-hidden={i !== currentIndex}
   172→                aria-label={`Batch ${i + 1} of ${UPCOMING_BATCHES.length}: ${batch.name}`}
   173→              >
   174→                <div className="relative rounded-2xl border border-border/60 bg-card shadow-lg overflow-hidden">
   175→                  {/* Atmospheric accent */}
   176→                  <div className="absolute top-0 right-0 w-[400px] h-[200px] bg-violet-600/5 blur-[100px] rounded-full pointer-events-none" />
   177→                  <div className="relative grid lg:grid-cols-12 gap-6 p-6 sm:p-8">
   178→                    {/* Left - batch identity */}
   179→                    <div className="lg:col-span-7">
   180→                      <div className="flex items-center gap-3 mb-4">
   181→                        <span className="text-[10px] font-mono text-violet-300 tracking-[0.3em]">
   182→                          UPCOMING BATCH
   183→                        </span>
   184→                        <span className="text-[10px] font-mono text-muted-foreground tracking-[0.2em]">
   185→                          {String(i + 1).padStart(2, "0")} / {String(UPCOMING_BATCHES.length).padStart(2, "0")}
   186→                        </span>
   187→                      </div>
   188→
   189→                      <h3 className="text-2xl sm:text-3xl font-bold tracking-tight mb-4 text-balance">
   190→                        {batch.name}
   191→                      </h3>
   192→
   193→                      <div className="flex flex-wrap items-center gap-2 mb-6">
   194→                        <Badge
   195→                          variant="outline"
   196→                          className={cn("text-[10px] font-mono tracking-wider", modeStyle.badge)}
   197→                        >
   198→                          <ModeIcon className="h-3 w-3 mr-1" />
   199→                          {batch.mode.toUpperCase()}
   200→                        </Badge>
   201→                        <Badge variant="outline" className="text-[10px] font-mono tracking-wider border-border/60 text-muted-foreground">
   202→                          <Clock className="h-3 w-3 mr-1" />
   203→                          {batch.schedule}
   204→                        </Badge>
   205→                      </div>
   206→
   207→                      {/* Key facts grid */}
   208→                      <div className="grid sm:grid-cols-2 gap-3">
   209→                        <div className="rounded-lg border border-border/40 bg-background/40 p-3">
   210→                          <div className="flex items-center gap-1.5 mb-1">
   211→                            <Calendar className="h-3 w-3 text-violet-300" />
   212→                            <span className="text-[9px] font-mono text-muted-foreground tracking-[0.2em]">STARTS</span>
   213→                          </div>
   214→                          <div className="text-sm font-semibold">{batch.startDate}</div>
   215→                        </div>
   216→                        <div className="rounded-lg border border-border/40 bg-background/40 p-3">

   231→                            SEATS AVAILABLE

   233→                          <span className="text-[10px] font-mono text-amber-300 tracking-[0.2em]">

   245→                            className={cn(
   246→                              "h-full rounded-full transition-all duration-700",

   270→      {/* Side nav buttons */}
   271→      <button
   272→        type="button"
   273→        onClick={goToPrev}

   276→      >
   277→        <ChevronLeft className="h-5 w-5" />
   278→      </button>
   279→      <button
   280→        type="button"
   281→        onClick={goToNext}
   282→        aria-label="Next batch"
   283→        className="absolute right-2 sm:-right-5 top-1/2 -translate-y-1/2 inline-flex h-10 w-10 items-center justify-center rounded-full border border-border/60 bg-card/80 backdrop-blur shadow-lg text-foreground hover:bg-violet-600 hover:text-white hover:border-violet-500 transition-all duration-300"
   284→      >
   285→        <ChevronRight className="h-5 w-5" />
   286→      </button>
   287→
   288→      {/* Dot indicators */}
   289→      <div className="flex items-center justify-center gap-2 mt-6">
   290→        {UPCOMING_BATCHES.map((b, i) => (
   291→          <button
   292→            key={b.name}
   293→            type="button"
   294→            onClick={() => goToSlide(i)}

   297→            className={cn(
   298→              "h-2 rounded-full transition-all duration-500",
   299→              i === currentIndex
   300→                ? "w-8 bg-violet-500"
   301→                : "w-2 bg-muted-foreground/30 hover:bg-muted-foreground/60"
   302→            )}
   303→          />
   304→        ))}
   305→      </div>
   306→    </div>
   307→  )

   310→export function CourseDetailView() {
   311→  const { view, navigate } = useAppStore()
   312→  const courseId = view.name === "course" ? view.courseId : ""
   313→  const { user } = useUser()
   314→  const qc = useQueryClient()
   315→  const heroRef = React.useRef(null)
   316→
   317→  const { data, isLoading } = useQuery<CourseDetail>({
   318→    queryKey: ["course", courseId],
   319→    queryFn: () => api(`/api/courses/${courseId}`),
   320→    enabled: !!courseId,
   321→  })
   322→
   323→  const enrollMutation = useMutation({
   324→    mutationFn: () => api(`/api/courses/${courseId}/enroll`, { method: "POST" }),
   325→    onSuccess: () => {
   326→      toast.success("Enrolled! Redirecting to My Learning…")
   327→      qc.invalidateQueries({ queryKey: ["course", courseId] })
   328→      qc.invalidateQueries({ queryKey: ["courses"] })
   329→      qc.invalidateQueries({ queryKey: ["me"] })
   330→      // Navigate to My Learning so the student can continue from the
   331→      // enrolled-courses list + Recommended Courses section.
   332→      setTimeout(() => navigate({ name: "learning" }), 600)
   333→    },
   334→    onError: (e: any) => toast.error(e.message),
   335→  })
   336→
   337→  // Fetch prerequisites for this course
   338→  const { data: prereqData } = useQuery<{ prerequisites: Prerequisite[] }>({
   339→    queryKey: ["course-prerequisites", courseId],
   340→    queryFn: () => api(`/api/courses/${courseId}/enroll`),
   341→    enabled: !!courseId,
   342→  })
   343→  const prerequisites = prereqData?.prerequisites ?? []
   344→
   345→  if (isLoading) {
   346→    return (
   347→      <div className="relative min-h-screen">
   348→        <div className="absolute inset-0 bg-mesh opacity-40 pointer-events-none" />

   353→            <Skeleton className="lg:col-span-2 h-[800px] rounded-2xl" />
   354→            <Skeleton className="h-[800px] rounded-2xl" />
   355→          </div>
   356→        </div>
   357→      </div>
   358→    )
   359→  }
   360→
   361→  if (!data) return null
   362→  const { course, enrollment, lessonProgress, progressPct, totalLessons, completedLessons } = data
   363→  const isEnrolled = !!enrollment
   364→
   365→  const goLesson = (lessonId: string) => {
   366→    if (!isEnrolled) {
   367→      toast.error("Enroll in this course to access lessons.")
   368→      return
   369→    }
   370→    navigate({ name: "lesson", lessonId, courseId })
   371→  }
   372→
   373→  // Build learning outcomes from longDescription / tags
   374→  const outcomes: string[] = []
   375→  if (course.longDescription) {
   376→    // First two sentences become outcomes preview
   377→    const sentences = course.longDescription.split(/\.\s+/).filter(Boolean).slice(0, 2)
   378→    sentences.forEach((s: string) => outcomes.push(s.trim() + "."))
   379→  }
   380→  if (course.tags) {
   381→    course.tags.split(",").slice(0, 3).forEach((t: string) => outcomes.push("Master " + t.trim().toLowerCase() + " fundamentals and real-world application."))
   382→  }
   383→  // Ensure we have at least 4 outcomes
   384→  while (outcomes.length < 4) {
   385→    outcomes.push("Develop practical, hands-on skills through GuardianX lab exercises.")
   386→  }
   387→
   388→  return (
   389→    <div className="relative min-h-screen">
   390→      {/* Atmospheric background */}
   391→      <div className="absolute inset-0 bg-mesh opacity-40 pointer-events-none" />
   392→      <div className="absolute inset-0 bg-grid opacity-15 pointer-events-none" />
   393→
   394→      <div className="relative z-10">
   395→        {/* ====================================================
   396→            HERO - cinematic course introduction
   397→            ==================================================== */}
   398→        <section ref={heroRef} className="relative overflow-hidden">
   399→          {/* Course hero image */}
   400→          <div className="absolute inset-0">
   401→            <img
   402→              src={getCourseImage(course)}
   403→              alt={course.title}
   404→              className="w-full h-full object-cover opacity-40"
   405→            />
   406→            <div className="absolute inset-0 bg-gradient-to-b from-background/60 via-background/80 to-background" />
   407→            <div className="absolute inset-0 bg-grid opacity-10" />
   408→          </div>
   409→          <div className="absolute top-1/3 -right-40 w-[600px] h-[600px] bg-violet-600/10 blur-[140px] rounded-full" />
   410→          <div className="absolute top-1/2 -left-40 w-[500px] h-[500px] bg-cyan-600/5 blur-[140px] rounded-full" />
