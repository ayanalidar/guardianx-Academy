   153→function ThemeToggle() {
   154→  const { theme, setTheme } = useTheme()
   155→  const [mounted, setMounted] = React.useState(false)
   156→  React.useEffect(() => setMounted(true), [])
   157→  if (!mounted) return <div className="h-9 w-9" />
   158→  return (
   159→    <Button
   160→      variant="ghost"
   161→      size="icon"
   162→      className="h-9 w-9"
   163→      onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
   164→      title="Toggle theme"
   165→      aria-label="Toggle dark/light theme"
   166→    >
   167→      {theme === "dark" ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
   168→    </Button>
   169→  )

   331→// ---- Command Palette (⌘K) ----
   332→function CommandPalette({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   333→  const { navigate } = useAppStore()
   334→  const [q, setQ] = React.useState("")
   335→  const [results, setResults] = React.useState<{ courses: any[]; labs: any[]; notes: any[] }>({ courses: [], labs: [], notes: [] })
   336→
   337→  React.useEffect(() => {
   338→    if (!q.trim()) {
   339→      setResults({ courses: [], labs: [], notes: [] })
   340→      return
   341→    }
   342→    const t = setTimeout(async () => {
   343→      try {
   344→        const r = await fetch(`/api/search?q=${encodeURIComponent(q)}`)
   345→        const data = await r.json()
   346→        setResults(data)
   347→      } catch {}
   348→    }, 150)
   349→    return () => clearTimeout(t)
   350→  }, [q])
   351→
   352→  React.useEffect(() => {
   353→    if (!open) setQ("")
   354→  }, [open])
   355→
   356→  const navItems = [
   357→    { label: "Dashboard", icon: LayoutDashboard, view: { name: "dashboard" } as const },
   358→    { label: "Course Catalog", icon: BookOpen, view: { name: "catalog" } as const },
   359→    { label: "My Learning", icon: GraduationCap, view: { name: "learning" } as const },
   360→    { label: "Cyber Labs", icon: FlaskConical, view: { name: "labs" } as const },

   389→            className="flex-1 bg-transparent outline-none text-sm placeholder:text-muted-foreground"
   390→          />
   391→          <kbd className="text-[10px] font-mono text-muted-foreground px-1.5 py-0.5 rounded border border-border">ESC</kbd>

   490→function timeAgo(date: string): string {
   491→  const diff = Date.now() - new Date(date).getTime()
   492→  const m = Math.floor(diff / 60000)
   493→  if (m < 1) return "just now"
   494→  if (m < 60) return `${m}m ago`
   495→  const h = Math.floor(m / 60)
   496→  if (h < 24) return `${h}h ago`
   497→  const d = Math.floor(h / 24)
   498→  if (d < 7) return `${d}d ago`
   499→  return new Date(date).toLocaleDateString()
   500→}
