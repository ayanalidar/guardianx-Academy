   108→function formatTimeAgo(iso?: string | null): string {
   109→  if (!iso) return "—"
   110→  const d = new Date(iso)
   111→  const diff = Date.now() - d.getTime()
   112→  const min = Math.floor(diff / 60_000)
   113→  if (min < 1) return "just now"
   114→  if (min < 60) return `${min}m ago`
   115→  const hr = Math.floor(min / 60)
   116→  if (hr < 24) return `${hr}h ago`
   117→  const day = Math.floor(hr / 24)
   118→  if (day < 30) return `${day}d ago`
   119→  return d.toLocaleDateString()
   120→}
   121→
   122→function detectKind(value: any): "string" | "stringArray" | "objectArray" | "object" | "primitive" {
   123→  if (typeof value === "string") return "string"
   124→  if (Array.isArray(value)) {
   125→    if (value.length === 0) return "stringArray"
   126→    if (value.every((v) => typeof v === "string")) return "stringArray"
   127→    if (value.every((v) => typeof v === "object" && v !== null)) return "objectArray"
   128→    if (value.every((v) => typeof v === "number" || typeof v === "boolean")) return "objectArray"
   129→    return "objectArray"
   130→  }
   131→  if (typeof value === "object" && value !== null) return "object"
   132→  if (typeof value === "number" || typeof value === "boolean") return "primitive"
   133→  return "string"
   134→}

   339→interface SectionData {
   340→  [key: string]: any
   341→}
   342→interface PageDataShape {
   343→  page: string
   344→  sections: Record<string, SectionData>
   345→  updatedAt?: string | null

   347→
   348→function PageEditor({ page }: { page: PageId }) {
   349→  const qc = useQueryClient()
   350→  const [draft, setDraft] = React.useState<Record<string, SectionData> | null>(null)
   351→  const [openSection, setOpenSection] = React.useState<string | null>(null)
   352→  const [resetDialogOpen, setResetDialogOpen] = React.useState(false)
   353→
   354→  const meta = PAGES.find((p) => p.id === page)!
   355→
   356→  const { data, isLoading, isError, refetch } = useQuery<PageDataShape>({
   357→    queryKey: ["cms-content", page],
   358→    queryFn: async () => {
   359→      const res = await fetch(`/api/cms/${page}`)
   360→      if (!res.ok) throw new Error("Failed to load content")
   361→      return res.json()
   362→    },
   363→    staleTime: 30_000,
   364→  })
   365→
   366→  // Reset draft whenever data changes (initial load + invalidations)
   367→  React.useEffect(() => {
   368→    if (data?.sections) {
   369→      setDraft(JSON.parse(JSON.stringify(data.sections)))
   370→    }
   371→  }, [data])
   372→
   373→  const saveMutation = useMutation({
   374→    mutationFn: async (payload: { items: Array<{ section: string; key: string; value: any }> }) => {
   375→      return api(`/api/cms/${page}`, {
   376→        method: "PUT",
   377→        body: JSON.stringify(payload),
   378→      })
   379→    },
   380→    onSuccess: () => {
   381→      toast.success("Changes saved · content is live")
   382→      qc.invalidateQueries({ queryKey: ["cms-content", page] })
   383→      // Also invalidate other pages (in case shared global content changed)
   384→      if (page === "global") {
   385→        qc.invalidateQueries({ queryKey: ["cms-content"] })
   386→      }
   387→    },
   388→    onError: (e: any) => {
   389→      toast.error(e?.message || "Failed to save")
   390→    },
   391→  })
   392→
   393→  const handleSave = () => {
   394→    if (!draft) return
   395→    const items: Array<{ section: string; key: string; value: any }> = []
   396→    for (const [section, keys] of Object.entries(draft)) {
   397→      for (const [key, value] of Object.entries(keys)) {
   398→        items.push({ section, key, value })
   399→      }
   400→    }
   401→    saveMutation.mutate({ items })
   402→  }
   403→
   404→  const handleReset = () => {
   405→    setResetDialogOpen(false)
   406→    if (data?.sections) {
   407→      setDraft(JSON.parse(JSON.stringify(data.sections)))
   408→      toast.info("Reverted to last saved version")
   409→    }
   410→  }
   411→
   412→  const isDirty = React.useMemo(() => {
   413→    if (!draft || !data?.sections) return false
   414→    return JSON.stringify(draft) !== JSON.stringify(data.sections)
   415→  }, [draft, data])
   416→
   417→  // Compute changes count
   418→  const changesCount = React.useMemo(() => {
   419→    if (!draft || !data?.sections) return 0
   420→    let n = 0
   421→    for (const [section, keys] of Object.entries(draft)) {
   422→      for (const [key, value] of Object.entries(keys)) {
   423→        const orig = data.sections[section]?.[key]
   424→        if (JSON.stringify(orig) !== JSON.stringify(value)) n++
   425→      }
   426→    }
   427→    return n
   428→  }, [draft, data])
   429→
   430→  const sectionList = draft ? Object.keys(draft).sort() : []
   431→
   432→  return (
   433→    <div className="space-y-4">
   434→      {/* === Top bar === */}
   435→      <Card className="bg-card shadow-lg border border-border p-4 lg:p-5">
   436→        <div className="flex flex-col md:flex-row md:items-center md:justify-between gap-3">
   437→          <div className="flex items-start gap-3">
   438→            <div className={cn(
   439→              "inline-flex items-center justify-center h-10 w-10 rounded-xl shrink-0",
   440→              "bg-amber-500/10"
   441→            )}>
   442→              <meta.icon className={cn("h-5 w-5", meta.accent)} />
   443→            </div>
   444→            <div className="min-w-0">
   445→              <div className="flex items-center gap-2 flex-wrap">
   446→                <h2 className="text-lg font-bold tracking-tight">{meta.label}</h2>
   447→                <Badge variant="outline" className="text-[9px] font-mono border-amber-500/30 text-amber-300 bg-amber-500/5">
   448→                  <Hash className="h-2.5 w-2.5 mr-1" />{page}
   449→                </Badge>
   450→                {sectionList.length > 0 && (
   451→                  <Badge variant="outline" className="text-[9px] font-mono">
   452→                    {sectionList.length} sections
   453→                  </Badge>
   454→                )}
   455→              </div>
   456→              <p className="text-xs text-muted-foreground mt-1">
   457→                {meta.description}
   458→              </p>
   459→              <p className="text-[10px] text-muted-foreground/70 font-mono mt-1">
   460→                Last saved: {formatTimeAgo(data?.updatedAt)}
   461→              </p>
   462→            </div>
   463→          </div>
   464→
   465→          <div className="flex items-center gap-2 flex-wrap">
   466→            <Button
   467→              variant="outline"
   468→              size="sm"
   469→              onClick={() => refetch()}
   470→              disabled={isLoading}
   471→              className="h-9"
   472→            >
   473→              <RefreshCw className={cn("h-3.5 w-3.5 mr-1.5", isLoading && "animate-spin")} />
   474→              Refresh
   475→            </Button>
   476→            <Button
   477→              variant="outline"
   478→              size="sm"
   479→              onClick={() => setResetDialogOpen(true)}
   480→              disabled={!isDirty || saveMutation.isPending}
   481→              className="h-9"
   482→            >
   483→              <RotateCcw className="h-3.5 w-3.5 mr-1.5" />
   484→              Reset
   485→            </Button>
   486→            <Button
   487→              size="sm"
   488→              onClick={handleSave}
   489→              disabled={!isDirty || saveMutation.isPending}
   490→              className="bg-amber-500 hover:bg-amber-400 text-amber-950 h-9"
   491→            >
   492→              {saveMutation.isPending ? (
   493→                <><Loader2 className="h-3.5 w-3.5 mr-1.5 animate-spin" /> Saving...</>
   494→              ) : (
   495→                <><Save className="h-3.5 w-3.5 mr-1.5" /> Save Changes</>
   496→              )}
   497→            </Button>
   498→          </div>
   499→        </div>
   500→
   501→        {/* Dirty indicator */}
   502→        {isDirty && (
   503→          <div className="mt-3 pt-3 border-t border-border flex items-center gap-2 text-xs">
   504→            <span className="inline-flex items-center gap-1.5 px-2 py-0.5 rounded-full bg-amber-500/10 border border-amber-500/30 text-amber-300 font-mono">
   505→              <span className="h-1.5 w-1.5 rounded-full bg-amber-400 pulse-dot" />
   506→              {changesCount} unsaved {changesCount === 1 ? "change" : "changes"}
   507→            </span>
   508→            <span className="text-muted-foreground">
   509→              Click "Save Changes" to publish.
   510→            </span>
   511→          </div>
   512→        )}
   513→      </Card>
   514→
   515→      {/* === Loading state === */}
   516→      {isLoading && <LoadingSkeleton />}
   517→
   518→      {/* === Error state === */}
   519→      {isError && (
   520→        <Card className="bg-card shadow-lg border border-red-500/30 p-6">
   521→          <div className="flex items-center gap-3">
   522→            <AlertCircle className="h-5 w-5 text-red-400" />
   523→            <div>
   524→              <div className="font-semibold text-sm">Failed to load content</div>
   525→              <div className="text-xs text-muted-foreground mt-0.5">
   526→                Check your connection or try refreshing.
   527→              </div>
   528→            </div>
   529→            <Button variant="outline" size="sm" onClick={() => refetch()} className="ml-auto">
   530→              Retry
   531→            </Button>
   532→          </div>
   533→        </Card>
   534→      )}
   535→
   536→      {/* === Sections === */}
   537→      {!isLoading && !isError && draft && sectionList.length === 0 && (
   538→        <Card className="bg-card shadow-lg border border-border p-12 text-center">

   765→function KeyEditorBody({
   766→  section,
   767→  keyName,
   768→  value,
   769→  kind,
   770→  showRaw,
   771→  onChange,
   772→}: {
   773→  section: string
   774→  keyName: string
   775→  value: any
   776→  kind: ReturnType<typeof detectKind>
   777→  showRaw: boolean
   778→  onChange: (v: any) => void
   779→}) {
   780→  // String
   781→  if (kind === "string") {
   782→    const isLong = (value as string).length > 60 || (value as string).includes("\n")
   783→    return isLong ? (
   784→      <Textarea
   785→        value={value as string}
   786→        onChange={(e) => onChange(e.target.value)}
   787→        rows={3}
   788→        className="resize-y text-xs"
   789→      />
   790→    ) : (
   791→      <Input
   792→        value={value as string}
   793→        onChange={(e) => onChange(e.target.value)}
   794→        className="h-9 text-xs"
   795→      />
   796→    )
   797→  }
   798→
   799→  // Number / boolean primitive
   800→  if (kind === "primitive") {
   801→    return (
   802→      <Input
   803→        value={String(value)}
   804→        onChange={(e) => {
   805→          const v = e.target.value
   806→          if (typeof value === "number") {
   807→            const n = Number(v)
   808→            onChange(Number.isNaN(n) ? 0 : n)
   809→          } else if (typeof value === "boolean") {
   810→            onChange(v === "true" || v === "1")
   811→          } else {
   812→            onChange(v)
   813→          }
   814→        }}
   815→        className="h-9 text-xs font-mono"
   816→      />
   817→    )
   818→  }
   819→
   820→  // String array
   821→  if (kind === "stringArray" && !showRaw) {
   822→    return <StringArrayEditor value={value as string[]} onChange={onChange} />
   823→  }
   824→
   825→  // Object array (with structured editor)
   826→  if (kind === "objectArray" && !showRaw) {
   827→    return <ObjectArrayEditor value={value as any[]} onChange={onChange} />
   828→  }
   829→
   830→  // Object (single)
   831→  if (kind === "object" && !showRaw) {
   832→    return <ObjectEditor value={value as Record<string, any>} onChange={onChange} />
   833→  }
   834→
   835→  // Fallback: raw JSON editor
   836→  return <JsonEditor value={value} onChange={onChange} />
   837→}

  1180→/* ============================================================
  1181→   JSON editor (raw, with validation)
  1182→   ============================================================ */
  1183→
  1184→function JsonEditor({
  1185→  value,
  1186→  onChange,
  1187→  compact = false,
  1188→}: {
  1189→  value: any
  1190→  onChange: (v: any) => void
  1191→  compact?: boolean
  1192→}) {
  1193→  const [text, setText] = React.useState<string>("")
  1194→  const [error, setError] = React.useState<string | null>(null)
  1195→
  1196→  // Sync text whenever value changes externally
  1197→  React.useEffect(() => {
  1198→    try {
  1199→      setText(JSON.stringify(value, null, compact ? 0 : 2))
  1200→      setError(null)
  1201→    } catch {
  1202→      setText(String(value ?? ""))
  1203→    }
  1204→  }, [value, compact])
  1205→
  1206→  const handle = (raw: string) => {
  1207→    setText(raw)
  1208→    if (raw.trim() === "") {
  1209→      setError(null)
  1210→      onChange(null)
  1211→      return
  1212→    }
  1213→    try {
  1214→      const parsed = JSON.parse(raw)
  1215→      setError(null)
  1216→      onChange(parsed)
  1217→    } catch (e: any) {
  1218→      setError(e?.message ?? "Invalid JSON")
  1219→      // Don't propagate the invalid value to parent
  1220→    }
  1221→  }
  1222→
  1223→  const isLarge = text.length > 200
  1224→
  1225→  return (
  1226→    <div className="space-y-1.5">
  1227→      {isLarge ? (
  1228→        <Textarea
  1229→          value={text}
  1230→          onChange={(e) => handle(e.target.value)}
  1231→          rows={Math.min(12, Math.max(4, Math.ceil(text.length / 60)))}
  1232→          className="font-mono text-[11px] resize-y leading-relaxed"
  1233→        />
  1234→      ) : (
  1235→        <Textarea
  1236→          value={text}
  1237→          onChange={(e) => handle(e.target.value)}
  1238→          rows={Math.min(6, Math.max(2, Math.ceil(text.split("\n").length * 1.2)))}
  1239→          className="font-mono text-[11px] resize-y leading-relaxed"
  1240→        />
  1241→      )}
  1242→      {error ? (
  1243→        <div className="flex items-start gap-1.5 text-[10px] text-red-400 font-mono">
  1244→          <AlertCircle className="h-3 w-3 mt-0.5 shrink-0" />
  1245→          <span className="break-all">{error}</span>
  1246→        </div>
  1247→      ) : (
  1248→        <div className="flex items-center gap-1.5 text-[10px] text-emerald-400/80 font-mono">
  1249→          <CheckCircle2 className="h-3 w-3" />
  1250→          <span>Valid JSON</span>
  1251→        </div>
  1252→      )}
  1253→    </div>
  1254→  )
  1255→}
