   292→/* ============================================================
   293→   StatCard — small solid premium card for stats strip

   295→function StatCard({
   296→  icon: Icon, label, value, color, tint, suffix,
   297→}: {
   298→  icon: typeof BookOpen; label: string; value: number; color: string; tint: string; suffix?: string
   299→}) {
   300→  return (
   301→    <div className="rounded-xl border border-border/60 bg-card shadow-lg p-5 flex items-center gap-4 hover:border-violet-500/30 transition-colors">
   302→      <div className={cn("inline-flex p-3 rounded-lg shrink-0", tint)}>
   303→        <Icon className={cn("h-5 w-5", color)} />
   304→      </div>
   305→      <div className="min-w-0">
   306→        <div className="text-2xl lg:text-3xl font-bold tabular-nums leading-none mb-1">

   310→      </div>
   311→    </div>
   312→  )
   313→}

   315→/* ============================================================
   316→   FilterChip — tiny removable filter pill
   317→   ============================================================ */
   318→function FilterChip({ label, onClear }: { label: string; onClear: () => void }) {
   319→  return (
   320→    <span className="inline-flex items-center gap-1.5 rounded-md border border-violet-500/30 bg-violet-500/10 px-2 py-1 text-[10px] font-mono text-violet-200">

   326→  )
