   414→/* ============================================================

   416→   ============================================================ */

   418→  // Word transformation: THREAT → DEFENSE
   419→  const threatOpacity = useTransform(progress, [0, 0.15, 0.35], [1, 1, 0])
   420→  const threatScale = useTransform(progress, [0, 0.35], [1, 1.3])
   421→  const threatBlur = useTransform(progress, [0, 0.35], [0, 20])
   422→  const defenseOpacity = useTransform(progress, [0.4, 0.6, 1], [0, 1, 1])
   423→  const defenseScale = useTransform(progress, [0.4, 0.6], [0.7, 1])
   424→
   425→  // Stages text
   426→  const stage1Opacity = useTransform(progress, [0.05, 0.15, 0.2], [0, 1, 0])
   427→  const stage2Opacity = useTransform(progress, [0.2, 0.3, 0.35], [0, 1, 0])

   430→
   431→  const vizScale = useTransform(progress, [0, 1], [1, 1.4])
   432→  const vizRotate = useTransform(progress, [0, 1], [0, 90])
   433→
   434→  return (
   435→    <div className="relative w-full h-full flex items-center justify-center">
   436→      {/* Pinned network visualization */}
   437→      <motion.div
   438→        style={{ scale: vizScale, rotate: vizRotate }}
   439→        className="absolute inset-0"
   440→      >

   443→
   444→      {/* Giant transforming word */}
   445→      <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
