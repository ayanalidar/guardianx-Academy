   111→export function AchievementsView() {
   112→  const { navigate } = useAppStore()
   113→  const { user } = useUser()
   114→  const isAdmin = user?.role === "ADMIN"
   115→
   116→  const [tierFilter, setTierFilter] = React.useState<TierFilterValue>("all")
   117→
   118→  const { data, isLoading } = useQuery<AchievementData>({
   119→    queryKey: ["achievements"],
   120→    queryFn: () => api("/api/achievements"),
   121→  })
   122→
   123→  // ----- admin create-achievement dialog state -----
   124→  const [createOpen, setCreateOpen] = React.useState(false)
   125→  const queryClient = useQueryClient()
   126→  const createMutation = useMutation({
   127→    mutationFn: async (payload: any) => api("/api/admin/achievements", {
   128→      method: "POST",
   129→      body: JSON.stringify(payload),
   130→    }),
   131→    onSuccess: () => {
   132→      toast.success("Achievement created")
   133→      queryClient.invalidateQueries({ queryKey: ["achievements"] })
   134→      queryClient.invalidateQueries({ queryKey: ["admin-achievements"] })
   135→      setCreateOpen(false)
   136→    },
   137→    onError: (err: any) => {
   138→      toast.error(err?.message ?? "Failed to create achievement")
   139→    },
   140→  })
   141→
   142→  // ----- tier filter (declared before any early returns to satisfy the
   143→  // React rules-of-hooks — hooks must run in the same order every render) -----
   144→  const achievementsList = data?.achievements ?? []
   145→  const earnedCount = data?.earnedCount ?? 0
   146→  const totalCount = data?.totalCount ?? 0
   147→
   148→  const filteredAchievements = React.useMemo(() => {
   149→    if (tierFilter === "all") return achievementsList
   150→    if (tierFilter === "earned") return achievementsList.filter((a) => a.earned)
   151→    if (tierFilter === "locked") return achievementsList.filter((a) => !a.earned)
   152→    return achievementsList.filter((a) => a.tier === tierFilter)
   153→  }, [achievementsList, tierFilter])
   154→
   155→  // Per-tier earned counts for filter chips
   156→  const tierChipCounts = React.useMemo(() => {
   157→    const out: Record<string, { earned: number; total: number }> = {
   158→      platinum: { earned: 0, total: 0 },
   159→      gold: { earned: 0, total: 0 },
   160→      silver: { earned: 0, total: 0 },
   161→      bronze: { earned: 0, total: 0 },
   162→    }
   163→    for (const a of achievementsList) {
   164→      if (!out[a.tier]) continue
   165→      out[a.tier].total++
   166→      if (a.earned) out[a.tier].earned++
   167→    }
   168→    return out
   169→  }, [achievementsList])
   170→
   171→  if (isLoading) {
   172→    return (
   173→      <div className="relative min-h-screen">
   174→        <div className="absolute inset-0 bg-mesh opacity-40 pointer-events-none" />
   175→        <div className="relative z-10 mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-12">
   176→          <Skeleton className="h-24 w-2/3 mb-8" />
   177→          <div className="grid sm:grid-cols-3 gap-8 mb-12">
   178→            {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-28" />)}
   179→          </div>
   180→          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
   181→            {Array.from({ length: 9 }).map((_, i) => <Skeleton key={i} className="h-40" />)}
   182→          </div>
   183→        </div>
   184→      </div>
   185→    )
   186→  }
   187→  if (!data) return null
   188→
   189→  const { xp, level, levelInfo, rank, streak, achievements, activities, heatmap, leaderboard } = data
   190→
   191→  // Group by tier (earned first within each tier)
   192→  const tierGroups = TIER_ORDER
   193→    .map((tier) => ({
   194→      tier,
   195→      config: TIER_CONFIG[tier],
   196→      items: filteredAchievements
   197→        .filter((a) => a.tier === tier)
   198→        .sort((a, b) => Number(b.earned) - Number(a.earned)),
   199→    }))
   200→    .filter((g) => g.items.length > 0)
   201→
   202→  // XP earned from achievements
   203→  const xpFromAchievements = achievements
   204→    .filter((a) => a.earned)
   205→    .reduce((sum, a) => sum + (a.xp || 0), 0)
   206→
   207→  // Completion rate
   208→  const completionRate = totalCount > 0 ? Math.round((earnedCount / totalCount) * 100) : 0
   209→
   210→  const filterChips: { value: TierFilterValue; label: string; count?: string }[] = [
   211→    { value: "all", label: "All", count: `${totalCount}` },
   212→    { value: "earned", label: "Earned", count: `${earnedCount}` },
   213→    { value: "locked", label: "Locked", count: `${totalCount - earnedCount}` },
   214→    ...TIER_ORDER.map((t) => ({
   215→      value: t as TierFilterValue,
   216→      label: TIER_CONFIG[t].label,
   217→      count: `${tierChipCounts[t].earned}/${tierChipCounts[t].total}`,
   218→    })),
   219→  ]
   220→
   221→  return (
   222→    <div className="relative min-h-screen">
   223→      {/* Atmospheric background */}
   224→      <div className="absolute inset-0 bg-mesh opacity-40 pointer-events-none" />
   225→      <div className="absolute top-0 right-0 w-[600px] h-[500px] bg-violet-600/6 blur-[140px] rounded-full pointer-events-none" />
   226→      <div className="absolute bottom-1/3 left-0 w-[400px] h-[400px] bg-amber-500/4 blur-[120px] rounded-full pointer-events-none" />
   227→
   228→      <div className="relative z-10 mx-auto max-w-7xl px-4 sm:px-6 lg:px-8 py-8 lg:py-12">
   229→        {/* ====================================================
   230→            HEADER - oversized headline + status pill
   231→            ==================================================== */}
   232→        <ScrollReveal>
   233→          <div className="flex items-center gap-3 mb-6">
   234→            <span className="h-1.5 w-1.5 rounded-full bg-violet-400 pulse-dot" />
   235→            <span className="text-[10px] font-mono text-violet-300/80 tracking-[0.3em]">
   236→              RANK {rank.toUpperCase()} · LEVEL {level}
   237→            </span>
   238→          </div>
   239→        </ScrollReveal>
   240→
   241→        <ScrollReveal delay={0.1}>
   242→          <h1 className="text-[clamp(2.5rem,8vw,5.5rem)] font-bold leading-[0.92] tracking-[-0.04em] mb-4 text-balance">
   243→            <TextReveal text="Your" />{" "}
   244→            <span className="text-gradient-premium">
   245→              <TextReveal text="achievements." delay={0.2} />
   246→            </span>
   247→          </h1>
   248→        </ScrollReveal>
   249→
   250→        <ScrollReveal delay={0.3}>
   251→          <div className="flex flex-wrap items-start gap-4 mb-12">
   252→            <p className="text-muted-foreground max-w-xl flex-1 text-base lg:text-lg leading-relaxed">
   253→              Every milestone is a checkpoint. Every badge, a piece of your story as a defender.
   254→            </p>
   255→            {isAdmin && (
   256→              <Button
   257→                onClick={() => setCreateOpen(true)}
   258→                className="gap-2"
   259→                size="sm"
   260→              >
   261→                <Plus className="h-4 w-4" /> New Achievement
   262→              </Button>
   263→            )}
   264→          </div>
   265→        </ScrollReveal>
   266→
   267→        {/* ====================================================
   268→            STATS STRIP - border-left editorial, NO cards
   269→            ==================================================== */}
   270→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-8 mb-12">
   271→          {[
   272→            { label: "Earned", value: earnedCount, suffix: ` / ${totalCount}`, accent: "border-violet-500/50", color: "text-violet-300", icon: Trophy },
   273→            { label: "XP from achievements", value: xpFromAchievements, accent: "border-cyan-500/50", color: "text-cyan-300", icon: Zap },
   274→            { label: "Completion rate", value: completionRate, suffix: "%", accent: "border-emerald-500/50", color: "text-emerald-300", icon: Target },
   275→            { label: "Day streak", value: streak, accent: "border-amber-500/50", color: "text-amber-300", icon: Flame },
   276→          ].map((s, i) => (
   277→            <ScrollReveal key={s.label} delay={0.4 + i * 0.08}>
   278→              <div className={cn("border-l pl-5", s.accent)}>
   279→                <s.icon className={cn("h-4 w-4 mb-3", s.color)} />
   280→                <div className="text-4xl lg:text-5xl font-bold tracking-[-0.03em] mb-1">
   281→                  <Counter value={s.value} suffix={s.suffix ?? ""} />
   282→                </div>
   283→                <div className="text-[10px] text-muted-foreground uppercase tracking-[0.2em]">{s.label}</div>
   284→              </div>
   285→            </ScrollReveal>
   286→          ))}
   287→        </div>
   288→
   289→        {/* ====================================================
   290→            LEVEL PROGRESS BAR - open, premium
   291→            ==================================================== */}
   292→        <ScrollReveal delay={0.5}>
   293→          <div className="mb-12 max-w-3xl">
   294→            <div className="flex items-center justify-between mb-3">
   295→              <span className="text-[10px] font-mono text-muted-foreground tracking-[0.3em]">
   296→                LEVEL {level} PROGRESSION
   297→              </span>
   298→              <span className="text-xs font-mono text-violet-300">
   299→                {levelInfo.currentLevelXp} / {levelInfo.nextLevelXp} XP
   300→              </span>
   301→            </div>
   302→            <Progress value={levelInfo.progress} className="h-1" />
   303→            <div className="mt-2 text-[10px] text-muted-foreground font-mono">
   304→              {levelInfo.nextLevelXp - levelInfo.currentLevelXp} XP TO LEVEL {level + 1}
   305→            </div>
   306→          </div>
   307→        </ScrollReveal>
   308→
   309→        {/* ====================================================
   310→            TIER FILTER CHIPS
   311→            ==================================================== */}
   312→        <ScrollReveal delay={0.05}>
   313→          <div className="mb-8 flex flex-wrap items-center gap-2">
   314→            <span className="text-[10px] font-mono text-muted-foreground tracking-[0.3em] uppercase mr-2">
   315→              Filter
   316→            </span>
   317→            {filterChips.map((c) => {
   318→              const active = tierFilter === c.value
   319→              const isTier = TIER_ORDER.includes(c.value as any)
   320→              const tierAccent = isTier ? TIER_CONFIG[c.value]?.text : null
   321→              return (
   322→                <button
   323→                  key={c.value}
   324→                  onClick={() => setTierFilter(c.value)}
   325→                  className={cn(
   326→                    "group inline-flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-[11px] font-mono uppercase tracking-wider transition-all",
   327→                    active
   328→                      ? "border-violet-500/40 bg-violet-500/10 text-violet-200 shadow-[0_0_24px_-8px] shadow-violet-500/30"
   329→                      : "border-border/60 bg-card/30 text-muted-foreground hover:text-foreground hover:border-border",
   330→                  )}
   331→                  aria-pressed={active}
   332→                >
   333→                  {isTier && (
   334→                    <span className={cn("text-base leading-none", tierAccent)}>
   335→                      {TIER_CONFIG[c.value]?.glyph}
   336→                    </span>
   337→                  )}
   338→                  <span>{c.label}</span>
   339→                  {c.count && (
   340→                    <span className={cn(
   341→                      "text-[9px] px-1.5 py-0.5 rounded-full",
   342→                      active ? "bg-violet-500/20 text-violet-200" : "bg-muted/40 text-muted-foreground",
   343→                    )}>
   344→                      {c.count}
   345→                    </span>
   346→                  )}
   347→                </button>
   348→              )
   349→            })}
   350→          </div>
   351→        </ScrollReveal>
   352→
   353→        {/* ====================================================
   354→            TIER-GROUPED ACHIEVEMENTS - editorial sections
   355→            ==================================================== */}
   356→        {tierGroups.length === 0 ? (
   357→          <div className="rounded-2xl border border-dashed border-border/60 p-12 text-center text-muted-foreground text-sm">
   358→            No achievements match this filter.
   359→          </div>
   360→        ) : (
   361→          <div className="space-y-20">
   362→            {tierGroups.map((group, gi) => (
   363→              <section key={group.tier} id={`tier-${group.tier}`}>
   364→                {/* Tier section header */}
   365→                <ScrollReveal>
   366→                  <div className="flex items-end justify-between mb-8 pb-4 border-b border-border/60">
   367→                    <div>
   368→                      <p className="text-[10px] font-mono text-muted-foreground tracking-[0.3em] mb-2">
   369→                        {String(gi + 1).padStart(2, "0")} - TIER {group.config.label.toUpperCase()}
   370→                      </p>
   371→                      <h2 className="flex items-baseline gap-3">
   372→                        <span className={cn("text-[clamp(2rem,5vw,3.5rem)] font-bold leading-none tracking-[-0.03em]", group.config.text)}>
   373→                          {group.config.label}
   374→                        </span>
   375→                        <span className={cn("text-[clamp(3rem,8vw,6rem)] font-bold leading-none", group.config.ghost)}>
   376→                          {group.config.glyph}
   377→                        </span>
   378→                      </h2>
   379→                    </div>
   380→                    <div className="text-right">
   381→                      <div className="text-2xl font-bold tabular-nums">
   382→                        {group.items.filter((i) => i.earned).length}
   383→                        <span className="text-muted-foreground/40"> / {group.items.length}</span>
   384→                      </div>
   385→                      <div className="text-[10px] text-muted-foreground uppercase tracking-[0.2em] mt-1">unlocked</div>
   386→                    </div>
   387→                  </div>
   388→                </ScrollReveal>
   389→
   390→                {/* Staggered grid */}
   391→                <Stagger className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4" staggerChildren={0.06}>
   392→                  {group.items.map((a) => (
   393→                    <StaggerItem key={a.code}>
   394→                      <AchievementTile a={a} tier={group.tier} />
   395→                    </StaggerItem>
   396→                  ))}
   397→                </Stagger>
   398→              </section>
   399→            ))}
   400→          </div>
   401→        )}
   402→
   403→        {/* ====================================================
   404→            ACTIVITY + LEADERBOARD + HEATMAP - sidebar grid
   405→            ==================================================== */}
   406→        <div className="mt-24 grid lg:grid-cols-3 gap-6">
   407→          {/* Heatmap */}
   408→          <ScrollReveal>
   409→            <div className="rounded-2xl border border-border/60 bg-card/30 p-6 h-full">
   410→              <div className="flex items-center gap-2 mb-5">
   411→                <TrendingUp className="h-4 w-4 text-emerald-300" />
   412→                <h3 className="text-sm font-semibold tracking-tight">Weekly Activity</h3>
   413→              </div>
   414→              <div className="grid grid-cols-7 gap-1.5 mb-3">
   415→                {heatmap.map((d) => {
   416→                  const dayName = ["Su", "Mo", "Tu", "We", "Th", "Fr", "Sa"][new Date(d.date + "T00:00:00").getDay()]
   417→                  const intensity = d.count === 0 ? 0 : Math.min(d.count, 4)
   418→                  return (
   419→                    <div key={d.date} className="flex flex-col items-center gap-1">
   420→                      <div className="text-[9px] text-muted-foreground font-mono">{dayName}</div>
   421→                      <div
   422→                        className={cn(
   423→                          "w-full aspect-square rounded-md border transition-all",
   424→                          intensity === 0 && "bg-muted/20 border-border",
   425→                          intensity === 1 && "bg-violet-500/20 border-violet-500/30",
   426→                          intensity === 2 && "bg-violet-500/40 border-violet-500/40",
   427→                          intensity === 3 && "bg-violet-500/60 border-violet-500/50",
   428→                          intensity >= 4 && "bg-violet-500/80 border-violet-500/60",
   429→                        )}
   430→                        title={`${d.date}: ${d.count} activities, ${d.xp} XP`}
   431→                      />
   432→                      <div className="text-[9px] text-muted-foreground font-mono">{d.xp}</div>
   433→                    </div>
   434→                  )
   435→                })}
   436→              </div>
   437→              <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   438→                <span>Less</span>
   439→                <div className="flex gap-0.5">
   440→                  <div className="h-2.5 w-2.5 rounded-sm bg-muted/20" />
   441→                  <div className="h-2.5 w-2.5 rounded-sm bg-violet-500/20" />
   442→                  <div className="h-2.5 w-2.5 rounded-sm bg-violet-500/40" />
   443→                  <div className="h-2.5 w-2.5 rounded-sm bg-violet-500/60" />
   444→                  <div className="h-2.5 w-2.5 rounded-sm bg-violet-500/80" />
   445→                </div>
   446→                <span>More</span>
   447→              </div>
   448→            </div>
   449→          </ScrollReveal>
   450→
   451→          {/* Recent XP activity */}
   452→          <ScrollReveal delay={0.1}>
   453→            <div className="rounded-2xl border border-border/60 bg-card/30 p-6 h-full">
   454→              <div className="flex items-center gap-2 mb-5">
   455→                <Zap className="h-4 w-4 text-amber-300" />
   456→                <h3 className="text-sm font-semibold tracking-tight">Recent XP</h3>
   457→              </div>
   458→              <ScrollArea className="h-64 pr-2">
   459→                <div className="space-y-2">
   460→                  {activities.length === 0 ? (
   461→                    <p className="text-xs text-muted-foreground text-center py-8">No activity yet. Start learning!</p>
   462→                  ) : (
   463→                    activities.map((a, i) => (
   464→                      <div key={i} className="flex items-center gap-3 p-2 rounded-lg bg-muted/20 border border-border/40">

   467→                          <div className="text-xs font-medium capitalize">{a.type.replace(/_/g, " ")}</div>

   474→                    ))
   475→                  )}
   476→                </div>
   477→              </ScrollArea>
   478→            </div>
   479→          </ScrollReveal>
   480→
   481→          {/* Mini leaderboard */}
   482→          <ScrollReveal delay={0.2}>
   483→            <div className="rounded-2xl border border-border/60 bg-card/30 p-6 h-full">
   484→              <div className="flex items-center justify-between mb-5">
   485→                <div className="flex items-center gap-2">

   499→                        "flex items-center gap-2 p-2 rounded-lg transition-colors border",
   500→                        u.isMe ? "bg-violet-500/10 border-violet-500/30" : "border-transparent hover:bg-muted/30",
   501→                      )}
   502→                    >
   503→                      <div className={cn(
   504→                        "flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold font-mono",
   505→                        u.rank === 1 && "bg-amber-400/20 text-amber-300",
   506→                        u.rank === 2 && "bg-slate-300/20 text-slate-300",
   507→                        u.rank === 3 && "bg-orange-600/20 text-orange-400",
   508→                        u.rank > 3 && "bg-muted text-muted-foreground",
   509→                      )}>
   510→                        {u.rank}
   511→                      </div>
   512→                      <div className="flex-1 min-w-0">
   513→                        <div className="text-xs font-medium truncate">
   514→                          {u.name}
   515→                          {u.isMe && <span className="text-violet-300 ml-1">(You)</span>}
   516→                        </div>
   517→                        <div className="text-[10px] text-muted-foreground truncate">{u.rankTitle}</div>
   518→                      </div>
   519→                      <div className="text-right">
   520→                        <div className="text-xs font-mono font-bold text-violet-300">{u.xp.toLocaleString()}</div>
   521→                        <div className="text-[9px] text-muted-foreground">Lv {u.level}</div>
   522→                      </div>
   523→                    </div>
   524→                  ))}
   525→                </div>
   526→              </ScrollArea>
   527→            </div>
   528→          </ScrollReveal>
   529→        </div>
   530→      </div>
   531→
   532→      {isAdmin && (
   533→        <CreateAchievementDialog
   534→          open={createOpen}
   535→          onOpenChange={setCreateOpen}
   536→          onSubmit={(payload) => createMutation.mutate(payload)}
   537→          submitting={createMutation.isPending}
   538→        />
   539→      )}
   540→    </div>
   541→  )
   542→}

   548→function AchievementTile({ a, tier }: { a: any; tier: string }) {

   550→  const col = colorFor(a.color)
   551→  const cfg = TIER_CONFIG[tier] ?? TIER_CONFIG.bronze
   552→  const progressPct = a.progress && a.progress.target > 0
   553→    ? Math.min(100, Math.round((a.progress.current / a.progress.target) * 100))
   554→    : 0

   567→        )}
   568→      >
   569→        {/* Gradient background wash - earned only */}
   570→        {a.earned && (

   575→          className="absolute top-0 right-0 w-16 h-16 pointer-events-none opacity-50"

   578→          }}
   579→        />

   608→              <span className="mr-1">{cfg.glyph}</span>
   609→              {cfg.label}
   610→            </Badge>
   611→          </div>
   612→
   613→          {/* Title + description */}
   614→          <h4 className={cn(
   615→            "text-base font-semibold tracking-tight mb-1",
   616→            !a.earned && "text-muted-foreground",
   617→          )}>

   633→                </span>
   634→              </div>
   635→              <Progress value={progressPct} className="h-1.5" />
   636→            </div>
   637→          )}
   638→
   639→          {/* Bottom row: XP + earned date */}
   640→          <div className="flex items-center justify-between pt-3 border-t border-border/40">
   641→            <span className={cn(
   642→              "text-xs font-mono flex items-center gap-1",
   643→              a.earned ? "text-emerald-300" : "text-muted-foreground/50",

   654→          </div>
   655→        </div>
   656→      </div>
   657→    </CursorGlow>
