```tsx
const { data } = useQuery<{ ... } | null>({
  queryKey: ["<unique-key>"],
  queryFn: async () => {
    try {
      const res = await fetch("/api/<endpoint>")
      if (!res.ok) return null
      return res.json()
    } catch {
      return null
    }
  },
  staleTime: 60_000,

const value = data?.field ?? FALLBACK_ARRAY
```
