    32→// Format milliseconds as Mm Ss or Hh Mm
    33→function formatDuration(ms: number): string {
    34→  const totalSec = Math.floor(ms / 1000)
    35→  const h = Math.floor(totalSec / 3600)
    36→  const m = Math.floor((totalSec % 3600) / 60)
    37→  const s = totalSec % 60
    38→  if (h > 0) return `${h}h ${m}m`
    39→  if (m > 0) return `${m}m ${s}s`
    40→  return `${s}s`
    41→}

   452→    queryFn: () => api("/api/labs/stats"),

   454→

   552→            </div>
   553→          </div>

   555→      </div>
   556→    </div>
   557→  )
   558→}
