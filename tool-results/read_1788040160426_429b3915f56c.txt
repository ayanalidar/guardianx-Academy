   160→function timeAgo(iso: string): string {
   161→  const diff = Date.now() - new Date(iso).getTime()
   162→  const s = Math.floor(diff / 1000)
   163→  if (s < 60) return "just now"
   164→  const m = Math.floor(s / 60)
   165→  if (m < 60) return `${m}m ago`
   166→  const h = Math.floor(m / 60)
   167→  if (h < 24) return `${h}h ago`
   168→  const d = Math.floor(h / 24)
   169→  if (d < 30) return `${d}d ago`
   170→  return new Date(iso).toLocaleDateString()
   171→}
   172→
   173→function dueLabel(iso: string): { text: string; date: Date } {
   174→  const date = new Date(iso)
   175→  return {
   176→    text: date.toLocaleDateString(undefined, {
   177→      month: "short",
   178→      day: "numeric",
   179→      hour: "2-digit",
   180→      minute: "2-digit",
   181→    }),
   182→    date,
   183→  }
   184→}
   185→
   186→function statusBadge(status: SubmissionStatus, overdue: boolean) {
   187→  if (overdue && status === "missing") {
   188→    return (
   189→      <Badge className="bg-rose-500/15 text-rose-400 border border-rose-500/30">
   190→        <AlertTriangle className="h-3 w-3 mr-1" /> Overdue
   191→      </Badge>
   192→    )
   193→  }
   194→  switch (status) {
   195→    case "missing":
   196→      return (
   197→        <Badge variant="outline" className="text-amber-400 border-amber-500/30 bg-amber-500/10">
   198→          <Clock className="h-3 w-3 mr-1" /> Pending
   199→        </Badge>
   200→      )
   201→    case "submitted":
   202→      return (
   203→        <Badge className="bg-cyan-500/15 text-cyan-400 border border-cyan-500/30">
   204→          <Send className="h-3 w-3 mr-1" /> Submitted
   205→        </Badge>
   206→      )
   207→    case "resubmitted":
   208→      return (
   209→        <Badge className="bg-cyan-500/15 text-cyan-400 border border-cyan-500/30">
   210→          <Send className="h-3 w-3 mr-1" /> Resubmitted
   211→        </Badge>
   212→      )
   213→    case "returned":
   214→      return (
   215→        <Badge className="bg-amber-500/15 text-amber-400 border border-amber-500/30">
   216→          <AlertTriangle className="h-3 w-3 mr-1" /> Returned
   217→        </Badge>
   218→      )
   219→    case "graded":
   220→      return (
   221→        <Badge className="bg-emerald-500/15 text-emerald-400 border border-emerald-500/30">
   222→          <CheckCircle2 className="h-3 w-3 mr-1" /> Graded
   223→        </Badge>
   224→      )
   225→    default:
   226→      return null
   227→  }
   228→}

   234→type FilterTab = "all" | "pending" | "submitted" | "graded" | "overdue"

   236→export function AssignmentsView() {
   237→  const { navigate } = useAppStore()
   238→  const qc = useQueryClient()
   239→  const [tab, setTab] = React.useState<FilterTab>("all")
   240→  const [selectedId, setSelectedId] = React.useState<string | null>(null)
   241→  const [peerReviewOpen, setPeerReviewOpen] = React.useState<string | null>(null)
   242→
   243→  const { data, isLoading } = useQuery<{ assignments: AssignmentItem[]; stats: AssignmentStats }>({
   244→    queryKey: ["me", "assignments"],
   245→    queryFn: () => api("/api/me/assignments"),
   246→  })
   247→
   248→  const assignments = data?.assignments ?? []
   249→
   250→  function bucketFor(a: AssignmentItem): Exclude<FilterTab, "all"> | null {
   251→    if (a.overdue) return "overdue"
   252→    if (a.status === "graded") return "graded"
   253→    if (a.status === "submitted" || a.status === "resubmitted" || a.status === "returned")
   254→      return "submitted"
   255→    if (a.status === "missing") return "pending"
   256→    return null
   257→  }
   258→
   259→  const filtered =
   260→    tab === "all" ? assignments : assignments.filter((a) => bucketFor(a) === tab)
   261→
   262→  // Compute stat-card counts (Pending = missing && !overdue, Submitted = submitted/resubmitted/returned, Graded, Overdue)
   263→  const pendingCount =
   264→    assignments.filter((a) => a.status === "missing" && !a.overdue).length
   265→  const submittedCount = assignments.filter(
   266→    (a) => a.status === "submitted" || a.status === "resubmitted" || a.status === "returned"
   267→  ).length
   268→  const gradedCount = assignments.filter((a) => a.status === "graded").length
   269→  const overdueCount = assignments.filter((a) => a.overdue).length
   270→
   271→  const statCards = [
   272→    {
   273→      key: "pending",
   274→      label: "Pending",
   275→      value: pendingCount,
   276→      icon: Clock,
   277→      accent: "text-amber-400",
   278→      bg: "bg-amber-500/10",
   279→      border: "border-amber-500/20",
   280→    },
   281→    {
   282→      key: "submitted",
   283→      label: "Submitted",
   284→      value: submittedCount,
   285→      icon: Send,
   286→      accent: "text-cyan-400",
   287→      bg: "bg-cyan-500/10",
   288→      border: "border-cyan-500/20",
   289→    },
   290→    {
   291→      key: "graded",
   292→      label: "Graded",
   293→      value: gradedCount,
   294→      icon: CheckCircle2,
   295→      accent: "text-emerald-400",
   296→      bg: "bg-emerald-500/10",
   297→      border: "border-emerald-500/20",
   298→    },
   299→    {
   300→      key: "overdue",
   301→      label: "Overdue",
   302→      value: overdueCount,
   303→      icon: AlertTriangle,
   304→      accent: "text-rose-400",
   305→      bg: "bg-rose-500/10",
   306→      border: "border-rose-500/20",
   307→    },
   308→  ]
   309→
   310→  return (
   311→    <div className="space-y-6">
   312→      {/* Header */}
   313→      <header className="space-y-1">
   314→        <h1 className="text-3xl font-bold flex items-center gap-2">
   315→          <ClipboardList className="h-7 w-7 text-emerald-400" /> Assignments
   316→        </h1>
   317→        <p className="text-muted-foreground">
   318→          Your pending and graded work across all enrolled courses.
   319→        </p>
   320→      </header>
   321→
   322→      {/* Stats row */}
   323→      <section className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   324→        {isLoading
   325→          ? Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28" />)
   326→          : statCards.map((s) => (
   327→              <Card
   328→                key={s.key}
   329→                className={cn(
   330→                  "p-5 relative overflow-hidden card-hover border",
   331→                  s.border
   332→                )}
   333→              >
   334→                <div className={cn("absolute inset-0 opacity-30 bg-grid", s.bg)} />
   335→                <div className="relative z-10">
   336→                  <div className="flex items-center justify-between mb-2">
   337→                    <s.icon className={cn("h-5 w-5", s.accent)} />
   338→                    <span className="text-3xl font-bold font-mono">{s.value}</span>
   339→                  </div>
   340→                  <div className="text-xs text-muted-foreground uppercase tracking-wider">
   341→                    {s.label}
   342→                  </div>
   343→                </div>
   344→              </Card>
   345→            ))}
   346→      </section>
   347→
   348→      {/* Filter tabs */}
   349→      <Tabs value={tab} onValueChange={(v) => setTab(v as FilterTab)}>
   350→        <TabsList className="grid w-full grid-cols-2 sm:grid-cols-5 h-auto">
   351→          <TabsTrigger value="all" className="py-2">All</TabsTrigger>
   352→          <TabsTrigger value="pending" className="py-2">Pending</TabsTrigger>
   353→          <TabsTrigger value="submitted" className="py-2">Submitted</TabsTrigger>
   354→          <TabsTrigger value="graded" className="py-2">Graded</TabsTrigger>
   355→          <TabsTrigger value="overdue" className="py-2 text-rose-400 data-[state=active]:text-rose-300">
   356→            Overdue
   357→          </TabsTrigger>
   358→        </TabsList>
   359→
   360→        <TabsContent value={tab} className="mt-4">
   361→          {isLoading ? (
   362→            <div className="space-y-3">
   363→              {Array.from({ length: 4 }).map((_, i) => (
   364→                <Skeleton key={i} className="h-28" />
   365→              ))}
   366→            </div>
   367→          ) : filtered.length === 0 ? (
   368→            <Card className="p-12 text-center border-dashed">
   369→              <ClipboardList className="h-12 w-12 text-muted-foreground mx-auto mb-4" />
   370→              <h3 className="font-semibold mb-1">No assignments here</h3>
   371→              <p className="text-sm text-muted-foreground mb-4">
   372→                {tab === "all"
   373→                  ? "You don't have any assignments across enrolled courses yet."
   374→                  : `Nothing in the "${tab}" bucket right now.`}
   375→              </p>
   376→              <Button onClick={() => navigate({ name: "catalog" })}>
   377→                <BookOpen className="h-4 w-4 mr-1.5" /> Browse Courses
   378→              </Button>
   379→            </Card>
   380→          ) : (
   381→            <div className="space-y-3">
   382→              {filtered.map((a) => (
   383→                <AssignmentCard
   384→                  key={a.id}
   385→                  assignment={a}
   386→                  onOpen={() => setSelectedId(a.id)}
   387→                  onOpenPeerReview={() => setPeerReviewOpen(a.id)}
   388→                />
   389→              ))}
   390→            </div>
   391→          )}
   392→        </TabsContent>
   393→      </Tabs>
   394→
   395→      {/* Submission / detail dialog */}
   396→      <AssignmentDialog
   397→        assignmentId={selectedId}
   398→        onClose={() => setSelectedId(null)}
   399→        onSubmitted={() => {
   400→          qc.invalidateQueries({ queryKey: ["me", "assignments"] })
   401→        }}
   402→      />
   403→
   404→      {/* Peer review dialog */}
   405→      <PeerReviewDialog
   406→        assignmentId={peerReviewOpen}
   407→        onClose={() => setPeerReviewOpen(null)}
   408→        onReviewed={() => {
   409→          qc.invalidateQueries({ queryKey: ["me", "assignments"] })
   410→        }}
   411→      />
   412→    </div>
   413→  )
   414→}

   503→
   504→function AssignmentDialog({
   505→  assignmentId,
   506→  onClose,
   507→  onSubmitted,
   508→}: {
   509→  assignmentId: string | null
   510→  onClose: () => void
   511→  onSubmitted: () => void
   512→}) {
   513→  const qc = useQueryClient()
   514→  const [content, setContent] = React.useState("")
   515→  const [fileUrl, setFileUrl] = React.useState("")
   516→
   517→  const { data, isLoading } = useQuery<{ assignment: AssignmentDetail }>({
   518→    queryKey: ["assignment", assignmentId],
   519→    queryFn: () => api(`/api/assignments/${assignmentId}`),
   520→    enabled: !!assignmentId,
   521→  })
   522→
   523→  // Reset form when assignment changes
   524→  React.useEffect(() => {
   525→    if (assignmentId) {
   526→      setContent("")
   527→      setFileUrl("")
   528→    }
   529→  }, [assignmentId])
   530→
   531→  const assignment = data?.assignment
   532→
   533→  const submitMutation = useMutation({
   534→    mutationFn: (body: { content?: string; fileUrl?: string }) =>
   535→      api(`/api/assignments/${assignmentId}/submit`, {
   536→        method: "POST",
   537→        body: JSON.stringify(body),
   538→      }),
   539→    onSuccess: () => {
   540→      toast.success("Submission received!", {
   541→        description: "Your instructor will grade it shortly.",
   542→      })
   543→      qc.invalidateQueries({ queryKey: ["assignment", assignmentId] })
   544→      onSubmitted()
   545→      onClose()
   546→    },
   547→    onError: (e: Error) => toast.error("Submission failed", { description: e.message }),
   548→  })
   549→
   550→  const canSubmit =
   551→    content.trim().length > 0 || fileUrl.trim().length > 0
   552→
   553→  return (
   554→    <Dialog open={!!assignmentId} onOpenChange={(o) => !o && onClose()}>
   555→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   556→        <DialogHeader>
   557→          <DialogTitle className="flex items-center gap-2">
   558→            <FileText className="h-5 w-5 text-emerald-400" />
   559→            {isLoading ? "Loading…" : assignment?.title}
   560→          </DialogTitle>
   561→          <DialogDescription>
   562→            {assignment
   563→              ? `${assignment.pointsPossible} points · Due ${dueLabel(assignment.dueDate).text}`
   564→              : "Assignment details"}
   565→          </DialogDescription>
   566→        </DialogHeader>
   567→
   568→        {isLoading || !assignment ? (
   569→          <div className="space-y-3 py-2">
   570→            <Skeleton className="h-4 w-3/4" />
   571→            <Skeleton className="h-4 w-2/3" />
   572→            <Skeleton className="h-20 w-full" />
   573→          </div>
   574→        ) : (
   575→          <div className="space-y-4 py-2">
   576→            {/* Description */}
   577→            <section>
   578→              <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   579→                Description
   580→              </Label>
   581→              <p className="text-sm mt-1 whitespace-pre-wrap">
   582→                {assignment.description || "No description provided."}
   583→              </p>
   584→            </section>
   585→
   586→            {/* Instructions */}
   587→            {assignment.instructions && (
   588→              <section>
   589→                <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   590→                  Instructions
   591→                </Label>
   592→                <div className="text-sm mt-1 whitespace-pre-wrap bg-card/50 border border-border rounded-lg p-3 max-h-64 overflow-y-auto">
   593→                  {assignment.instructions}
   594→                </div>
   595→              </section>
   596→            )}
   597→
   598→            {/* Submission type / rules */}
   599→            <section className="grid sm:grid-cols-3 gap-3 text-xs">
   600→              <div className="bg-card/50 border border-border rounded-lg p-3">
   601→                <div className="text-muted-foreground">Submission Type</div>
   602→                <div className="font-medium capitalize">{assignment.submissionType}</div>
   603→              </div>
   604→              <div className="bg-card/50 border border-border rounded-lg p-3">
   605→                <div className="text-muted-foreground">Late submissions</div>
   606→                <div className="font-medium">
   607→                  {assignment.allowLate ? `Allowed (-${assignment.latePenalty}%)` : "Not allowed"}
   608→                </div>
   609→              </div>
   610→              <div className="bg-card/50 border border-border rounded-lg p-3">
   611→                <div className="text-muted-foreground">Peer review</div>
   612→                <div className="font-medium">
   613→                  {assignment.enablePeerReview
   614→                    ? `${assignment.peerReviewCount} reviews`
   615→                    : "Disabled"}
   616→                </div>
   617→              </div>
   618→            </section>
   619→
   620→            {/* Submission form */}
   621→            <section className="space-y-3 border-t border-border pt-4">
   622→              <Label className="text-xs uppercase tracking-wider text-muted-foreground">
   623→                Submit your work
   624→              </Label>
   625→              <div className="space-y-2">
   626→                <Label htmlFor="submit-content">Content / Answer</Label>
   627→                <Textarea
   628→                  id="submit-content"
   629→                  placeholder="Type or paste your submission here…"
   630→                  value={content}
   631→                  onChange={(e) => setContent(e.target.value)}
   632→                  className="min-h-[120px]"
   633→                />
   634→              </div>
   635→              <div className="space-y-2">
   636→                <Label htmlFor="submit-file">File URL (optional)</Label>
   637→                <div className="flex gap-2">
   638→                  <Input
   639→                    id="submit-file"
   640→                    placeholder="https://drive.google.com/…"
   641→                    value={fileUrl}
   642→                    onChange={(e) => setFileUrl(e.target.value)}
   643→                  />
   644→                  <Link2 className="h-4 w-4 text-muted-foreground self-center" />
   645→                </div>
   646→              </div>
   647→            </section>
   648→          </div>
   649→        )}
   650→
   651→        <DialogFooter>
   652→          <DialogClose asChild>
   653→            <Button variant="ghost">Cancel</Button>
   654→          </DialogClose>
   655→          <Button
   656→            disabled={!canSubmit || submitMutation.isPending || !assignment}
   657→            onClick={() =>
   658→              submitMutation.mutate({
   659→                content: content.trim() || undefined,
   660→                fileUrl: fileUrl.trim() || undefined,
   661→              })
   662→            }
   663→          >
   664→            {submitMutation.isPending ? "Submitting…" : "Submit"}
   665→            <Send className="h-3.5 w-3.5 ml-1.5" />
   666→          </Button>
   667→        </DialogFooter>
   668→      </DialogContent>
   669→    </Dialog>
   670→  )
   671→}

   674→/* Peer Review Dialog                                                 */
   675→/* ------------------------------------------------------------------ */
   676→
   677→function PeerReviewDialog({
   678→  assignmentId,
   679→  onClose,
   680→  onReviewed,
   681→}: {
   682→  assignmentId: string | null
   683→  onClose: () => void
   684→  onReviewed: () => void
   685→}) {
   686→  const qc = useQueryClient()
   687→  const [activeSubmissionId, setActiveSubmissionId] = React.useState<string | null>(null)
   688→  const [rating, setRating] = React.useState(0)
   689→  const [hoverRating, setHoverRating] = React.useState(0)
   690→  const [feedback, setFeedback] = React.useState("")
   691→
   692→  const { data, isLoading } = useQuery<PeerReviewResponse>({
   693→    queryKey: ["assignment", assignmentId, "peer-reviews"],
   694→    queryFn: () => api(`/api/assignments/${assignmentId}/peer-reviews`),
   695→    enabled: !!assignmentId,
   696→  })
   697→
   698→  React.useEffect(() => {
   699→    if (assignmentId) {
   700→      setActiveSubmissionId(null)
   701→      setRating(0)
   702→      setHoverRating(0)
   703→      setFeedback("")
   704→    }
   705→  }, [assignmentId])
   706→
   707→  const reviewMutation = useMutation({
   708→    mutationFn: (body: { rating: number; feedback: string }) =>
   709→      api(`/api/submissions/${activeSubmissionId}/peer-review`, {
   710→        method: "POST",
   711→        body: JSON.stringify(body),
   712→      }),
   713→    onSuccess: () => {
   714→      toast.success("Peer review submitted!", {
   715→        description: "Thanks for helping a peer improve.",
   716→      })
   717→      qc.invalidateQueries({ queryKey: ["assignment", assignmentId, "peer-reviews"] })
   718→      setActiveSubmissionId(null)
   719→      setRating(0)
   720→      setHoverRating(0)
   721→      setFeedback("")
   722→      onReviewed()
   723→    },
   724→    onError: (e: Error) => toast.error("Review failed", { description: e.message }),
   725→  })
   726→
   727→  const toReview = data?.toReview ?? []
   728→  const completed = data?.completed ?? []
   729→  const progress = data?.progress
   730→
   731→  return (
   732→    <Dialog open={!!assignmentId} onOpenChange={(o) => !o && onClose()}>
   733→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   734→        <DialogHeader>
   735→          <DialogTitle className="flex items-center gap-2">
   736→            <Users className="h-5 w-5 text-cyan-400" /> Peer Review
   737→          </DialogTitle>
   738→          <DialogDescription>
   739→            {data?.assignment?.title ?? "Review your peers' submissions"}
   740→          </DialogDescription>
   741→        </DialogHeader>
   742→
   743→        {isLoading ? (
   744→          <div className="space-y-3 py-2">
   745→            <Skeleton className="h-6 w-1/2" />
   746→            <Skeleton className="h-24 w-full" />
   747→            <Skeleton className="h-24 w-full" />
   748→          </div>
   749→        ) : data?.needsOwnSubmission ? (
   750→          <div className="py-6 text-center">
   751→            <Lock className="h-10 w-10 text-muted-foreground mx-auto mb-3" />
   752→            <p className="font-medium mb-1">Submit your own work first</p>
   753→            <p className="text-sm text-muted-foreground">{data.message}</p>
   754→          </div>
   755→        ) : (
   756→          <div className="space-y-4 py-2">
   757→            {/* Progress */}
   758→            {progress && (
   759→              <div className="bg-card/50 border border-border rounded-lg p-3 flex items-center justify-between text-sm">
   760→                <span className="text-muted-foreground">Progress</span>
   761→                <span className="font-mono font-bold text-cyan-400">
   762→                  {progress.completed} / {progress.target}
   763→                </span>
   764→              </div>
   765→            )}
   766→
   767→            {/* Submissions to review */}
   768→            {toReview.length === 0 && completed.length === 0 ? (
   769→              <div className="py-6 text-center text-sm text-muted-foreground">
   770→                No peer submissions are available to review right now.
   771→              </div>
   772→            ) : null}
   773→
   774→            {toReview.length > 0 && (
   775→              <section>
   776→                <Label className="text-xs uppercase tracking-wider text-muted-foreground mb-2 block">
   777→                  To review ({toReview.length})
   778→                </Label>
   779→                <div className="space-y-2">
   780→                  {toReview.map((s) => {
   781→                    const active = activeSubmissionId === s.id
   782→                    return (
   783→                      <Card
   784→                        key={s.id}
   785→                        className={cn(
   786→                          "p-3 cursor-pointer transition-colors",
   787→                          active
   788→                            ? "border-cyan-500/40 bg-cyan-500/5"
   789→                            : "hover:border-border"
   790→                        )}
   791→                        onClick={() => {
   792→                          setActiveSubmissionId(s.id)
   793→                          setRating(0)
   794→                          setHoverRating(0)
   795→                          setFeedback("")
   796→                        }}
   797→                      >
   798→                        <div className="flex items-center gap-3">
   799→                          <Avatar className="h-8 w-8 border border-border">
   800→                            {s.user.avatar ? (
   801→                              <AvatarImage src={s.user.avatar} alt={s.user.name} />
   802→                            ) : null}
   803→                            <AvatarFallback className="bg-cyan-500/10 text-cyan-400 text-xs">
   804→                              {s.user.name
   805→                                .split(" ")
   806→                                .map((n) => n[0])
   807→                                .join("")
   808→                                .slice(0, 2)}
   809→                            </AvatarFallback>
   810→                          </Avatar>
   811→                          <div className="flex-1 min-w-0">
   812→                            <div className="text-sm font-medium truncate">{s.user.name}</div>
   813→                            <div className="text-xs text-muted-foreground">
   814→                              {timeAgo(s.submittedAt)}
   815→                              {s.late && (
   816→                                <span className="ml-2 text-rose-400">· Late</span>
   817→                              )}
   818→                            </div>
   819→                          </div>
   820→                          <Button size="sm" variant={active ? "default" : "outline"}>
   821→                            {active ? "Reviewing" : "Review"}
   822→                          </Button>
   823→                        </div>
   824→                        {active && (
   825→                          <div className="mt-3 space-y-3 border-t border-border pt-3">
   826→                            {s.content && (
   827→                              <div className="text-sm whitespace-pre-wrap bg-background/50 rounded p-2 max-h-40 overflow-y-auto">
   828→                                {s.content}
   829→                              </div>
   830→                            )}
   831→                            {s.fileUrl && (
   832→                              <a
   833→                                href={s.fileUrl}
   834→                                target="_blank"
   835→                                rel="noreferrer"
   836→                                className="inline-flex items-center gap-1.5 text-xs text-emerald-400 hover:underline"
   837→                              >
   838→                                <Link2 className="h-3 w-3" /> View attached file
   839→                              </a>
   840→                            )}
   841→
   842→                            {/* Rating */}
   843→                            <div>
   844→                              <Label className="text-xs text-muted-foreground">Rating</Label>
   845→                              <div className="flex items-center gap-1 mt-1">
   846→                                {[1, 2, 3, 4, 5].map((n) => (
   847→                                  <button
   848→                                    key={n}
   849→                                    type="button"
   850→                                    aria-label={`${n} stars`}
   851→                                    onMouseEnter={() => setHoverRating(n)}
   852→                                    onMouseLeave={() => setHoverRating(0)}
   853→                                    onClick={() => setRating(n)}
   854→                                    className="p-0.5"
   855→                                  >
   856→                                    <Star
   857→                                      className={cn(
   858→                                        "h-6 w-6 transition-colors",
   859→                                        (hoverRating || rating) >= n
   860→                                          ? "fill-amber-400 text-amber-400"
   861→                                          : "text-muted-foreground"
   862→                                      )}
   863→                                    />
   864→                                  </button>
   865→                                ))}
   866→                                <span className="ml-2 text-xs text-muted-foreground font-mono">
   867→                                  {rating > 0 ? `${rating}/5` : "Select"}
   868→                                </span>
   869→                              </div>
   870→                            </div>
   871→
   872→                            {/* Feedback */}
   873→                            <div>
   874→                              <Label htmlFor="peer-feedback" className="text-xs text-muted-foreground">
   875→                                Feedback
   876→                              </Label>
   877→                              <Textarea
   878→                                id="peer-feedback"
   879→                                placeholder="Provide constructive feedback…"
   880→                                value={feedback}
   881→                                onChange={(e) => setFeedback(e.target.value)}
   882→                                className="mt-1 min-h-[80px]"
   883→                              />
   884→                            </div>
   885→
   886→                            <div className="flex justify-end gap-2">
   887→                              <Button
   888→                                size="sm"
   889→                                disabled={
   890→                                  rating < 1 ||
   891→                                  reviewMutation.isPending
   892→                                }
   893→                                onClick={() =>
   894→                                  reviewMutation.mutate({ rating, feedback: feedback.trim() })
   895→                                }
   896→                              >
   897→                                {reviewMutation.isPending ? "Submitting…" : "Submit Review"}
   898→                                <Send className="h-3.5 w-3.5 ml-1.5" />
   899→                              </Button>
   900→                            </div>
   901→                          </div>
   902→                        )}
   903→                      </Card>
   904→                    )
   905→                  })}
   906→                </div>
   907→              </section>
   908→            )}
   909→
   910→            {/* Completed reviews */}
   911→            {completed.length > 0 && (
   912→              <section>
   913→                <Label className="text-xs uppercase tracking-wider text-muted-foreground mb-2 block">
   914→                  Completed ({completed.length})
   915→                </Label>
   916→                <div className="space-y-2">
   917→                  {completed.map((r) => (
   918→                    <Card key={r.id} className="p-3 opacity-80">
   919→                      <div className="flex items-center gap-2 mb-1">
   920→                        <Avatar className="h-6 w-6">
   921→                          <AvatarFallback className="bg-emerald-500/10 text-emerald-400 text-[10px]">
   922→                            {r.submission.user.name
   923→                              .split(" ")
   924→                              .map((n) => n[0])
   925→                              .join("")
   926→                              .slice(0, 2)}
   927→                          </AvatarFallback>
   928→                        </Avatar>
   929→                        <span className="text-xs font-medium">{r.submission.user.name}</span>
   930→                        <span className="text-xs text-muted-foreground">
   931→                          · {timeAgo(r.submittedAt)}
   932→                        </span>
   933→                        <Badge className="ml-auto bg-amber-500/15 text-amber-400 border border-amber-500/30">
   934→                          <Star className="h-3 w-3 mr-1 fill-amber-400" /> {r.rating}/5
   935→                        </Badge>
   936→                      </div>
   937→                      {r.feedback && (
   938→                        <p className="text-xs text-muted-foreground">{r.feedback}</p>
   939→                      )}
   940→                    </Card>
   941→                  ))}
   942→                </div>
   943→              </section>
   944→            )}
   945→          </div>
   946→        )}
   947→
   948→        <DialogFooter>
   949→          <DialogClose asChild>
   950→            <Button variant="ghost">Close</Button>
   951→          </DialogClose>
   952→        </DialogFooter>
   953→      </DialogContent>
   954→    </Dialog>
   955→  )
   956→}
