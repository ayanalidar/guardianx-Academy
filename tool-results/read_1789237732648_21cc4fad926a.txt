   477→function EditorView({ courseId, onBack }: { courseId: string; onBack: () => void }) {
   478→  const qc = useQueryClient()
   479→  const { data, isLoading, isError, error } = useQuery<any>({
   480→    queryKey: ["course-studio", courseId],
   481→    queryFn: () => api(`/api/course-studio/${courseId}`),
   482→    enabled: !!courseId,
   483→  })
   484→
   485→  // Local working copy of the config (so the editor feels instant)
   486→  const [config, setConfig] = React.useState<CourseConfig | null>(null)
   487→  const [title, setTitle] = React.useState("")
   488→  const [status, setStatus] = React.useState<string>("draft")
   489→  const [version, setVersion] = React.useState(1)
   490→
   491→  const [selectedModuleId, setSelectedModuleId] = React.useState<string | null>(null)
   492→  const [selectedLessonId, setSelectedLessonId] = React.useState<string | null>(null)
   493→  const [rightTab, setRightTab] = React.useState<"preview" | "json" | "settings">("preview")
   494→  const [dirty, setDirty] = React.useState(false)
   495→  const [publishOpen, setPublishOpen] = React.useState(false)
   496→
   497→  // Sync server data → local state once
   498→  React.useEffect(() => {
   499→    if (data?.course) {
   500→      const cfg = (data.course.config || {}) as CourseConfig
   501→      setConfig(cfg)
   502→      setTitle(data.course.title)
   503→      setStatus(data.course.status)
   504→      setVersion(data.course.version)
   505→      setDirty(false)
   506→      // Auto-select first module/lesson
   507→      if (cfg.modules?.length && !selectedModuleId) {
   508→        setSelectedModuleId(cfg.modules[0].id)
   509→        if (cfg.modules[0].lessons?.length && !selectedLessonId) {
   510→          setSelectedLessonId(cfg.modules[0].lessons[0].id)
   511→        }
   512→      }
   513→    }
   514→  }, [data])
   515→
   516→  const patchMutation = useMutation({
   517→    mutationFn: (patch: any) =>
   518→      api(`/api/course-studio/${courseId}`, {
   519→        method: "PATCH",
   520→        body: JSON.stringify(patch),
   521→      }),
   522→    onSuccess: (res) => {
   523→      setDirty(false)
   524→      setStatus(res.course.status)
   525→      setVersion(res.course.version)
   526→      qc.invalidateQueries({ queryKey: ["course-studio-list"] })
   527→      toast.success("Saved")
   528→    },
   529→    onError: (e: any) => toast.error(e.message || "Save failed"),
   530→  })
   531→
   532→  const publishMutation = useMutation({
   533→    mutationFn: () =>
   534→      api(`/api/course-studio/${courseId}`, {
   535→        method: "POST",
   536→        body: JSON.stringify({ force: true }),
   537→      }),
   538→    onSuccess: (res: any) => {
   539→      toast.success(
   540→        `Published! ${res.moduleCount} modules · ${res.lessonCount} lessons · v${res.version}`
   541→      )
   542→      setPublishOpen(false)
   543→      setStatus("published")
   544→      setVersion(res.version)
   545→      qc.invalidateQueries({ queryKey: ["course-studio", courseId] })
   546→      qc.invalidateQueries({ queryKey: ["course-studio-list"] })
   547→    },
   548→    onError: (e: any) => {
   549→      toast.error(e.message || "Publish failed")
   550→      setPublishOpen(false)
   551→    },
   552→  })
   553→
   554→  // ----- Mutators on the local config -----
   555→  const updateConfig = React.useCallback(
   556→    (updater: (prev: CourseConfig) => CourseConfig) => {
   557→      setConfig((prev) => (prev ? updater(prev) : prev))
   558→      setDirty(true)
   559→    },
   560→    []
   561→  )
   562→
   563→  const handleSave = () => {
   564→    if (!config) return
   565→    patchMutation.mutate({ config, title })
   566→  }
   567→
   568→  if (isLoading) {
   569→    return (
   570→      <div className="space-y-4">
   571→        <Skeleton className="h-16 w-full" />
   572→        <div className="grid lg:grid-cols-[280px_1fr_320px] gap-4">
   573→          <Skeleton className="h-96" />
   574→          <Skeleton className="h-96" />
   575→          <Skeleton className="h-96" />
   576→        </div>
   577→      </div>
   578→    )
   579→  }
   580→
   581→  if (isError || !config) {
   582→    return (
   583→      <Card className="p-8 text-center border-rose-500/30">
   584→        <p className="font-medium text-rose-400 mb-2">Failed to load course</p>
   585→        <p className="text-sm text-muted-foreground mb-4">
   586→          {(error as Error)?.message}
   587→        </p>
   588→        <Button onClick={onBack} variant="outline">
   589→          <ArrowLeft className="h-4 w-4 mr-2" /> Back to list
   590→        </Button>
   591→      </Card>
   592→    )
   593→  }
   594→
   595→  const selectedModule = config.modules.find((m) => m.id === selectedModuleId) || null
   596→  const selectedLesson =
   597→    selectedModule?.lessons.find((l) => l.id === selectedLessonId) || null
   598→
   599→  const totalLessons = config.modules.reduce(
   600→    (a, m) => a + m.lessons.length,
   601→    0
   602→  )
   603→
   604→  // ----- Module / lesson operations -----
   605→  const addModule = () => {
   606→    const m: ModuleCfg = {
   607→      id: uid("m"),
   608→      title: `Module ${config.modules.length + 1}`,
   609→      description: "",
   610→      lessons: [],
   611→    }
   612→    updateConfig((prev) => ({ ...prev, modules: [...prev.modules, m] }))
   613→    setSelectedModuleId(m.id)
   614→    setSelectedLessonId(null)
   615→  }
   616→
   617→  const deleteModule = (id: string) => {
   618→    updateConfig((prev) => ({
   619→      ...prev,
   620→      modules: prev.modules.filter((m) => m.id !== id),
   621→    }))
   622→    if (selectedModuleId === id) {
   623→      setSelectedModuleId(null)
   624→      setSelectedLessonId(null)
   625→    }
   626→  }
   627→
   628→  const moveModule = (id: string, dir: -1 | 1) => {
   629→    updateConfig((prev) => {
   630→      const idx = prev.modules.findIndex((m) => m.id === id)
   631→      if (idx < 0) return prev
   632→      const newIdx = idx + dir
   633→      if (newIdx < 0 || newIdx >= prev.modules.length) return prev
   634→      const arr = [...prev.modules]
   635→      ;[arr[idx], arr[newIdx]] = [arr[newIdx], arr[idx]]
   636→      return { ...prev, modules: arr }
   637→    })
   638→  }
   639→
   640→  const addLesson = (moduleId: string) => {
   641→    const l: LessonCfg = {
   642→      id: uid("l"),
   643→      title: "New Lesson",
   644→      type: "reading",
   645→      content: "",
   646→      durationMin: 15,
   647→      preview: false,
   648→    }
   649→    updateConfig((prev) => ({
   650→      ...prev,
   651→      modules: prev.modules.map((m) =>
   652→        m.id === moduleId ? { ...m, lessons: [...m.lessons, l] } : m
   653→      ),
   654→    }))
   655→    setSelectedModuleId(moduleId)
   656→    setSelectedLessonId(l.id)
   657→  }
   658→
   659→  const deleteLesson = (moduleId: string, lessonId: string) => {
   660→    updateConfig((prev) => ({
   661→      ...prev,
   662→      modules: prev.modules.map((m) =>
   663→        m.id === moduleId
   664→          ? { ...m, lessons: m.lessons.filter((l) => l.id !== lessonId) }
   665→          : m
   666→      ),
   667→    }))
   668→    if (selectedLessonId === lessonId) setSelectedLessonId(null)
   669→  }
   670→
   671→  const moveLesson = (moduleId: string, lessonId: string, dir: -1 | 1) => {
   672→    updateConfig((prev) => ({
   673→      ...prev,
   674→      modules: prev.modules.map((m) => {
   675→        if (m.id !== moduleId) return m
   676→        const idx = m.lessons.findIndex((l) => l.id === lessonId)
   677→        if (idx < 0) return m
   678→        const newIdx = idx + dir
   679→        if (newIdx < 0 || newIdx >= m.lessons.length) return m
   680→        const arr = [...m.lessons]
   681→        ;[arr[idx], arr[newIdx]] = [arr[newIdx], arr[idx]]
   682→        return { ...m, lessons: arr }
   683→      }),
   684→    }))
   685→  }
   686→
   687→  const updateModule = (moduleId: string, patch: Partial<ModuleCfg>) => {
   688→    updateConfig((prev) => ({
   689→      ...prev,
   690→      modules: prev.modules.map((m) =>
   691→        m.id === moduleId ? { ...m, ...patch } : m
   692→      ),
   693→    }))
   694→  }
   695→
   696→  const updateLesson = (
   697→    moduleId: string,
   698→    lessonId: string,
   699→    patch: Partial<LessonCfg>
   700→  ) => {
   701→    updateConfig((prev) => ({
   702→      ...prev,
   703→      modules: prev.modules.map((m) =>
   704→        m.id === moduleId
   705→          ? {
   706→              ...m,
   707→              lessons: m.lessons.map((l) =>
   708→                l.id === lessonId ? { ...l, ...patch } : l
   709→              ),
   710→            }
   711→          : m
   712→      ),
   713→    }))
   714→  }
   715→
   716→  return (
   717→    <div className="space-y-4">
   718→      {/* Top bar */}
   719→      <div className="flex flex-col sm:flex-row sm:items-center gap-3 justify-between">
   720→        <div className="flex items-center gap-3 min-w-0">
   721→          <Button
   722→            variant="outline"
   723→            size="sm"
   724→            onClick={onBack}
   725→            className="shrink-0"
   726→          >
   727→            <ArrowLeft className="h-3.5 w-3.5" />
   728→          </Button>
   729→          <div className="min-w-0">
   730→            <div className="flex items-center gap-2 flex-wrap">
   731→              <Input
   732→                value={title}
   733→                onChange={(e) => {
   734→                  setTitle(e.target.value)
   735→                  setDirty(true)
   736→                }}
   737→                className="h-8 w-full sm:w-72 font-semibold text-base px-2"
   738→              />
   739→              <Badge
   740→                variant="outline"
   741→                className={cn(
   742→                  "text-[10px]",
   743→                  status === "published"
   744→                    ? "text-emerald-400 border-emerald-500/30"
   745→                    : status === "review"
   746→                    ? "text-amber-400 border-amber-500/30"
   747→                    : "text-cyan-400 border-cyan-500/30"
   748→                )}
   749→              >
   750→                {status} · v{version}
   751→              </Badge>
   752→              {dirty && (
   753→                <Badge variant="outline" className="text-[10px] text-amber-300">
   754→                  <CircleDot className="h-2.5 w-2.5 mr-1 animate-pulse" /> unsaved
   755→                </Badge>
   756→              )}
   757→            </div>
   758→            <p className="text-[11px] text-muted-foreground mt-1">
   759→              {config.modules.length} modules · {totalLessons} lessons ·{" "}
   760→              {config.durationHours}h
   761→            </p>
   762→          </div>
   763→        </div>
   764→        <div className="flex items-center gap-2 shrink-0">
   765→          <Button
   766→            variant="outline"
   767→            size="sm"
   768→            onClick={handleSave}
   769→            disabled={patchMutation.isPending || !dirty}
   770→            className="border-violet-500/30 text-violet-300 hover:bg-violet-500/10"
   771→          >
   772→            <Save className="h-3.5 w-3.5 mr-1.5" /> Save
   773→          </Button>
   774→          <Button
   775→            size="sm"
   776→            onClick={() => setPublishOpen(true)}
   777→            disabled={publishMutation.isPending}
   778→            className="bg-emerald-600 hover:bg-emerald-500 btn-premium"
   779→          >
   780→            <Rocket className="h-3.5 w-3.5 mr-1.5" />
   781→            {publishMutation.isPending ? "Publishing…" : "Publish"}
   782→          </Button>
   783→        </div>
   784→      </div>
   785→
   786→      {/* Three-pane editor */}
   787→      <div className="grid lg:grid-cols-[300px_1fr_340px] gap-4">
   788→        {/* LEFT - outline */}
   789→        <Card className="bg-card shadow-lg p-3 flex flex-col max-h-[80vh] overflow-hidden">
   790→          <div className="flex items-center justify-between px-2 py-2 mb-1">
   791→            <h3 className="text-sm font-semibold flex items-center gap-2">
   792→              <Layers className="h-4 w-4 text-violet-400" /> Outline
   793→            </h3>
   794→            <Button
   795→              size="sm"
   796→              variant="ghost"
   797→              className="h-7 px-2 text-violet-300 hover:bg-violet-500/10"
   798→              onClick={addModule}
   799→            >
   800→              <Plus className="h-3.5 w-3.5 mr-1" /> Module
   801→            </Button>
   802→          </div>
   803→          <div className="flex-1 overflow-y-auto pr-1 space-y-2">
   804→            {config.modules.length === 0 && (
   805→              <div className="text-center py-8 text-xs text-muted-foreground">
   806→                No modules yet.
   807→                <br />
   808→                Click <b>+ Module</b> to start.
   809→              </div>
   810→            )}
   811→            {config.modules.map((m, mIdx) => {
   812→              const expanded = selectedModuleId === m.id
   813→              return (
   814→                <div
   815→                  key={m.id}
   816→                  className={cn(
   817→                    "rounded-lg border transition-colors",
   818→                    expanded
   819→                      ? "border-violet-500/40 bg-violet-500/5"
   820→                      : "border-border hover:border-violet-500/20"
   821→                  )}
   822→                >
   823→                  <div className="flex items-center gap-1 px-2 py-2">
   824→                    <button
   825→                      onClick={() =>
   826→                        setSelectedModuleId(expanded ? null : m.id)
   827→                      }
   828→                      className="flex items-center gap-2 flex-1 min-w-0 text-left"
   829→                    >
   830→                      <GripVertical className="h-3.5 w-3.5 text-muted-foreground/60 shrink-0" />
   831→                      <div className="min-w-0">
   832→                        <div className="text-xs font-mono text-muted-foreground">
   833→                          M{mIdx + 1}
   834→                        </div>
   835→                        <div className="text-sm font-medium truncate">
   836→                          {m.title || "Untitled"}
   837→                        </div>
   838→                      </div>
   839→                    </button>
   840→                    <div className="flex items-center gap-0.5">
   841→                      <button
   842→                        onClick={() => moveModule(m.id, -1)}
   843→                        disabled={mIdx === 0}
   844→                        className="p-1 rounded hover:bg-accent disabled:opacity-30"
   845→                        title="Move up"
   846→                      >
   847→                        <ChevronUp className="h-3 w-3" />
   848→                      </button>
   849→                      <button
   850→                        onClick={() => moveModule(m.id, 1)}
   851→                        disabled={mIdx === config.modules.length - 1}
   852→                        className="p-1 rounded hover:bg-accent disabled:opacity-30"
   853→                        title="Move down"
   854→                      >
   855→                        <ChevronDown className="h-3 w-3" />
   856→                      </button>
   857→                      <button
   858→                        onClick={() => deleteModule(m.id)}
   859→                        className="p-1 rounded hover:bg-rose-500/10 text-rose-400"
   860→                        title="Delete module"
   861→                      >
   862→                        <Trash2 className="h-3 w-3" />
   863→                      </button>
   864→                    </div>
   865→                  </div>
   866→                  {expanded && (
   867→                    <div className="px-2 pb-2 space-y-1 border-t border-border/60 mt-1 pt-2">
   868→                      {m.lessons.length === 0 && (
   869→                        <div className="text-[10px] text-muted-foreground text-center py-2">
   870→                          No lessons yet
   871→                        </div>
   872→                      )}
   873→                      {m.lessons.map((l, lIdx) => {
   874→                        const meta = LESSON_TYPE_META[l.type]
   875→                        const Icon = meta.icon
   876→                        const active = selectedLessonId === l.id
   877→                        return (
   878→                          <div
   879→                            key={l.id}
   880→                            className={cn(
   881→                              "group flex items-center gap-1 px-1.5 py-1 rounded text-xs",
   882→                              active
   883→                                ? "bg-violet-500/15 text-violet-200"
   884→                                : "hover:bg-accent/40"
   885→                            )}
   886→                          >
   887→                            <button
   888→                              onClick={() => {
   889→                                setSelectedModuleId(m.id)
   890→                                setSelectedLessonId(l.id)
   891→                              }}
   892→                              className="flex items-center gap-1.5 flex-1 min-w-0 text-left"
   893→                            >
   894→                              <Icon className={cn("h-3 w-3 shrink-0", meta.color)} />
   895→                              <span className="truncate">
   896→                                <span className="text-muted-foreground mr-1">
   897→                                  {lIdx + 1}.
   898→                                </span>
   899→                                {l.title || "Untitled"}
   900→                              </span>
   901→                              {l.preview && (
   902→                                <Badge
   903→                                  variant="outline"
   904→                                  className="text-[8px] py-0 px-1 ml-1 text-amber-300"
   905→                                >
   906→                                  FREE
   907→                                </Badge>
   908→                              )}
   909→                            </button>
   910→                            <div className="opacity-0 group-hover:opacity-100 flex items-center gap-0.5 transition-opacity">
   911→                              <button
   912→                                onClick={() => moveLesson(m.id, l.id, -1)}
   913→                                disabled={lIdx === 0}
   914→                                className="p-0.5 rounded hover:bg-accent disabled:opacity-30"
   915→                              >
   916→                                <ChevronUp className="h-2.5 w-2.5" />
   917→                              </button>
   918→                              <button
   919→                                onClick={() => moveLesson(m.id, l.id, 1)}
   920→                                disabled={lIdx === m.lessons.length - 1}
   921→                                className="p-0.5 rounded hover:bg-accent disabled:opacity-30"
   922→                              >
   923→                                <ChevronDown className="h-2.5 w-2.5" />
   924→                              </button>
   925→                              <button
   926→                                onClick={() => deleteLesson(m.id, l.id)}
   927→                                className="p-0.5 rounded hover:bg-rose-500/10 text-rose-400"
   928→                              >
   929→                                <Trash2 className="h-2.5 w-2.5" />
   930→                              </button>
   931→                            </div>
   932→                          </div>
   933→                        )
   934→                      })}
   935→                      <Button
   936→                        size="sm"
   937→                        variant="ghost"
   938→                        className="w-full h-7 text-[11px] text-violet-300 hover:bg-violet-500/10"
   939→                        onClick={() => addLesson(m.id)}
   940→                      >
   941→                        <Plus className="h-3 w-3 mr-1" /> Add Lesson
   942→                      </Button>
   943→                    </div>
   944→                  )}
   945→                </div>
   946→              )
   947→            })}
   948→          </div>
   949→        </Card>
   950→
   951→        {/* CENTER - editor */}
   952→        <Card className="bg-card shadow-lg p-5 max-h-[80vh] overflow-y-auto">
   953→          {!selectedModule ? (
   954→            <EmptyEditor
   955→              icon={Layers}
   956→              title="Select a module"
   957→              sub="Pick a module from the outline to edit its title, description, and lessons."
   958→            />
   959→          ) : !selectedLesson ? (
   960→            <ModuleEditor
   961→              module={selectedModule}
   962→              onUpdate={(patch) => updateModule(selectedModule.id, patch)}
   963→              onAddLesson={() => addLesson(selectedModule.id)}
   964→            />
   965→          ) : (
   966→            <LessonEditor
   967→              module={selectedModule}
   968→              lesson={selectedLesson}
   969→              onUpdateLesson={(patch) =>
   970→                updateLesson(selectedModule.id, selectedLesson.id, patch)
   971→              }
   972→              onBackToModule={() => setSelectedLessonId(null)}
   973→            />
   974→          )}
   975→        </Card>
   976→
   977→        {/* RIGHT - preview / json / settings */}
   978→        <Card className="bg-card shadow-lg p-0 max-h-[80vh] overflow-hidden flex flex-col">
   979→          <div className="flex border-b border-border">
   980→            <TabBtn
   981→              active={rightTab === "preview"}
   982→              onClick={() => setRightTab("preview")}
   983→              icon={Eye}
   984→              label="Preview"
   985→            />
   986→            <TabBtn
   987→              active={rightTab === "json"}
   988→              onClick={() => setRightTab("json")}
   989→              icon={Code2}
   990→              label="JSON"
   991→            />
   992→            <TabBtn
   993→              active={rightTab === "settings"}
   994→              onClick={() => setRightTab("settings")}
   995→              icon={Settings2}
   996→              label="Settings"
   997→            />
   998→          </div>
   999→          <div className="flex-1 overflow-y-auto">
  1000→            {rightTab === "preview" && (
  1001→              <PreviewPane
  1002→                config={config}
  1003→                selectedModuleId={selectedModuleId}
  1004→                selectedLessonId={selectedLessonId}
  1005→              />
  1006→            )}
  1007→            {rightTab === "json" && (
  1008→              <JsonPane
  1009→                config={config}
  1010→                onImport={(cfg) => {
  1011→                  setConfig(cfg)
  1012→                  setDirty(true)
  1013→                  toast.success("Config imported from JSON")
  1014→                }}
  1015→              />
  1016→            )}
  1017→            {rightTab === "settings" && (
  1018→              <SettingsPane
  1019→                config={config}
  1020→                onUpdate={(patch) => {
  1021→                  updateConfig((prev) => ({ ...prev, ...patch }))
  1022→                }}
  1023→              />
  1024→            )}
  1025→          </div>
  1026→          {/* Publish footer */}
  1027→          <div className="border-t border-border p-3 bg-card/50">
  1028→            <Button
  1029→              onClick={() => setPublishOpen(true)}
  1030→              disabled={publishMutation.isPending}
  1031→              className="w-full bg-emerald-600 hover:bg-emerald-500 btn-premium"
  1032→            >
  1033→              <Rocket className="h-4 w-4 mr-2" />
  1034→              {publishMutation.isPending ? "Publishing…" : "Publish Course"}
  1035→            </Button>
  1036→            <p className="text-[10px] text-muted-foreground text-center mt-2">
  1037→              {status === "published"
  1038→                ? "Re-publishing updates the live course."
  1039→                : "Publishing creates a live Course + modules + lessons."}
  1040→            </p>
  1041→          </div>
  1042→        </Card>
  1043→      </div>
  1044→
  1045→      {/* Publish confirm */}
  1046→      <AlertDialog open={publishOpen} onOpenChange={setPublishOpen}>
  1047→        <AlertDialogContent>
  1048→          <AlertDialogHeader>
  1049→            <AlertDialogTitle>Publish &quot;{title}&quot;?</AlertDialogTitle>
  1050→            <AlertDialogDescription>
  1051→              This will {status === "published" ? "update the existing" : "create a new"}{" "}
  1052→              live Course with{" "}
  1053→              <b className="text-foreground">
  1054→                {config.modules.length} modules
  1055→              </b>{" "}
  1056→              and{" "}
  1057→              <b className="text-foreground">{totalLessons} lessons</b>.
  1058→              Students will be able to enroll immediately.
  1059→            </AlertDialogDescription>
  1060→          </AlertDialogHeader>
  1061→          <AlertDialogFooter>
  1062→            <AlertDialogCancel>Cancel</AlertDialogCancel>
  1063→            <AlertDialogAction
  1064→              onClick={() => publishMutation.mutate()}
  1065→              className="bg-emerald-600 hover:bg-emerald-500"
  1066→            >
  1067→              <Rocket className="h-4 w-4 mr-2" /> Confirm Publish
  1068→            </AlertDialogAction>
  1069→          </AlertDialogFooter>
  1070→        </AlertDialogContent>
  1071→      </AlertDialog>
  1072→    </div>
  1073→  )
  1074→}

  1079→function ModuleEditor({

  1085→  onUpdate: (patch: Partial<ModuleCfg>) => void
  1086→  onAddLesson: () => void

  1135→              >
